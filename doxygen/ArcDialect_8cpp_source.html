58/// Registered hook to materialize a single constant operation from a given
59/// attribute value with the desired resultant type. This method should use
60/// the provided builder to create the operation without changing the
61/// insertion position. The generated operation is expected to be constant
62/// like, i.e. single result, zero operands, non side-effecting, etc. On
63/// success, this hook should return the value generated to represent the
64/// constant value. Otherwise, it should return null on failure.
65Operation *ArcDialect::materializeConstant(OpBuilder &builder, Attribute value,
66 Type type, Location loc) {
67 // Integer constants.
68 if (auto intType = dyn_cast<IntegerType>(type))
69 if (auto attrValue = dyn_cast<IntegerAttr>(value))
70 return hw::ConstantOp::create(builder, loc, type, attrValue);
71
72 // Parameter expressions materialize into hw.param.value.
73 auto *parentOp = builder.getBlock()->getParentOp();
74 auto curModule = dyn_cast<hw::HWModuleOp>(parentOp);
75 if (!curModule)
76 curModule = parentOp->getParentOfType<hw::HWModuleOp>();
77 if (curModule && isValidParameterExpression(value, curModule))
78 return hw::ParamValueOp::create(builder, loc, type, value);
79
80 return nullptr;
81}
