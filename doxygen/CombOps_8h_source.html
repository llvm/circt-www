120// Check if the operand is zext() and return the extension bits:
121// zext = comb.concat(0, baseValue)
122template <typename SubType>
123struct ZextByMatcher {
124 SubType lhs;
125 ZextByMatcher(SubType lhs) : lhs(std::move(lhs)) {}
126 bool match(Operation *op) {
127 // Check if operand is a concat operation
128 auto concatOp = dyn_cast<ConcatOp>(op);
129 if (!concatOp)
130 return false;
131
132 auto operands = concatOp.getOperands();
133 // ConcatOp must have at least 2 operands: (sign_bits, base_value_1,...)
134 if (operands.size() < 2)
135 return false;
136
137 auto constOp = operands[0].getDefiningOp<hw::ConstantOp>();
138 if (!constOp || !constOp.getValue().isZero())
139 return false;
140
141 // Match the most significant argument of the concat
142 return mlir::detail::matchOperandOrValueAtIndex(op, 0, lhs);
143 }
144};

152// Check if the operand has a replicated extension and return the extension
153// bits:
154// ext = comb.replicate(bit, width-baseWidth) or hw.constant -1
155// replExt = comb.concat(ext, baseValue_1, ...)
156template <typename SubType>
157struct ReplExtMatcher {
158 SubType lhs;
159 ReplExtMatcher(SubType lhs) : lhs(std::move(lhs)) {}
160 bool match(Operation *op) {
161 auto concatOp = dyn_cast<ConcatOp>(op);
162 if (!concatOp)
163 return false;
164
165 auto operands = concatOp.getOperands();
166 if (operands.size() < 2)
167 return false;
168
169 auto replicateOp = operands[0].getDefiningOp<ReplicateOp>();
170 auto constOp = operands[0].getDefiningOp<hw::ConstantOp>();
171 bool isBitReplicate =
172 replicateOp &&
173 replicateOp.getInput().getType().getIntOrFloatBitWidth() == 1;
174 bool isAllOnes = constOp && constOp.getValue().isAllOnes();
175 if (!isBitReplicate && !isAllOnes)
176 return false;
177
178 // Match the most significant argument of the concat.
179 return mlir::detail::matchOperandOrValueAtIndex(op, 0, lhs);
180 }
181};

189// Check if the operand is sext() and return the extension bits:
190// signBit = comb.extract(baseValue, width-1, 1)
191// ext = comb.replicate(signBit, width-baseWidth)
192// sext = comb.concat(ext, baseValue)
193// Also matches the single bit case:
194// sext = comb.concat(signBit, baseValue)
195template <typename SubType>
196struct SextByMatcher {
197 SubType lhs;
198 SextByMatcher(SubType lhs) : lhs(std::move(lhs)) {}
199 bool match(Operation *op) {
200 // Check if operand is a concat operation
201 auto concatOp = dyn_cast<ConcatOp>(op);
202 if (!concatOp)
203 return false;
204
205 auto operands = concatOp.getOperands();
206 // ConcatOp must have at least 2 operands: (sign_bits, base_value_1,...)
207 if (operands.size() < 2)
208 return false;
209
210 Value signBits = operands[0];
211 Value baseValue = operands[1];
212 auto baseWidth = baseValue.getType().getIntOrFloatBitWidth();
213
214 // Check if signBits is a replicate operation
215 auto replicateOp = dyn_cast_or_null<ReplicateOp>(signBits.getDefiningOp());
216 Value signBit = replicateOp ? replicateOp.getInput() : signBits;
217
218 // For the single bit case, check the bitwidth of signBit == 1
219 if (signBit.getType().getIntOrFloatBitWidth() != 1)
220 return false;
221
222 // Check if signBit is the msb of baseValue
223 auto extractOp = dyn_cast_or_null<ExtractOp>(signBit.getDefiningOp());
224 if (!extractOp)
225 return false;
226
227 if ((extractOp.getInput() != baseValue) ||
228 (extractOp.getLowBit() != baseWidth - 1))
229 return false;
230
231 // Match the most significant argument of the concat
232 return mlir::detail::matchOperandOrValueAtIndex(op, 0, lhs);
233 }
234};
