32namespace {
33// add(a1, a2, ...) -> add(compress(a1, a2, ...))
34struct CombAddOpConversion : OpConversionPattern<AddOp> {
35 using OpConversionPattern<AddOp>::OpConversionPattern;
36 LogicalResult
37 matchAndRewrite(AddOp op, OpAdaptor adaptor,
38 ConversionPatternRewriter &rewriter) const override {
39
40 auto width = op.getType().getIntOrFloatBitWidth();
41 // Skip a zero width value.
42 if (width == 0) {
43 rewriter.replaceOpWithNewOp<hw::ConstantOp>(op, op.getType(), 0);
44 return success();
45 }
46
47 // Reduce to two values (carry,save)
48 auto results = datapath::CompressOp::create(rewriter, op.getLoc(),
49 op.getOperands(), 2);
50 // carry+saved
51 rewriter.replaceOpWithNewOp<AddOp>(op, results.getResults(), true);
52 return success();
53 }
54};
55
56// mul(a,b) -> add(pp(a,b))
57// multi-input adder will be converted to a compressor by other pattern
58struct CombMulOpConversion : OpConversionPattern<MulOp> {
59 using OpConversionPattern<MulOp>::OpConversionPattern;
60 LogicalResult
61 matchAndRewrite(MulOp op, OpAdaptor adaptor,
62 ConversionPatternRewriter &rewriter) const override {
63 // TODO: support for variadic multipliers
64 if (adaptor.getInputs().size() != 2)
65 return failure();
66
67 auto width = op.getType().getIntOrFloatBitWidth();
68 // Create partial product rows - number of rows == width
69 auto pp = datapath::PartialProductOp::create(rewriter, op.getLoc(),
70 op.getInputs(), width);
71 // Sum partial products
72 rewriter.replaceOpWithNewOp<AddOp>(op, pp.getResults(), true);
73 return success();
74 }
75};

92static void
93populateCombToDatapathConversionPatterns(RewritePatternSet &patterns) {
94 patterns.add<CombAddOpConversion, CombMulOpConversion>(patterns.getContext());
95 patterns.add(comb::convertSubToAdd);
96}
97
98void ConvertCombToDatapathPass::runOnOperation() {
99 ConversionTarget target(getContext());
100
101 target.addLegalDialect<datapath::DatapathDialect, comb::CombDialect,
102 hw::HWDialect>();
103
104 // Permit 2-input adders and 3-input adds with a constant-1 carry-in operand
105 // (carry-propagate adders).
106 target.addDynamicallyLegalOp<comb::AddOp>([](comb::AddOp op) {
107 if (op.getNumOperands() <= 2)
108 return true;
109
110 if (op.getNumOperands() == 3) {
111 if (auto constOp = op.getOperand(2).getDefiningOp<hw::ConstantOp>())
112 return constOp.getValue().isOne();
113 }
114 return false;
115 });
116 // TODO: determine lowering of multi-input multipliers
117 target.addDynamicallyLegalOp<comb::MulOp>(
118 [](comb::MulOp op) { return op.getNumOperands() > 2; });
119 target.addIllegalOp<comb::SubOp>();
120
121 RewritePatternSet patterns(&getContext());
122 populateCombToDatapathConversionPatterns(patterns);
123
124 if (failed(mlir::applyPartialConversion(getOperation(), target,
125 std::move(patterns))))
126 return signalPassFailure();
127}
