23std::string MessageData::toHex() const {
24 std::ostringstream ss;
25 ss << std::hex;
26 for (size_t i = 0, e = data.size(); i != e; ++i) {
27 // Add spaces every 8 bytes.
28 if (i % 8 == 0 && i != 0)
29 ss << ' ';
30 // Add an extra space every 64 bytes.
31 if (i % 64 == 0 && i != 0)
32 ss << ' ';
33 ss << static_cast<unsigned>(data[i]);
34 }
35 return ss.str();
36}

69MessageData SegmentedMessageData::toMessageData() const {
70 size_t total = totalSize();
71 if (total == 0)
72 return MessageData();
73 std::vector<uint8_t> buf;
74 buf.reserve(total);
75 for (size_t i = 0, e = numSegments(); i < e; ++i) {
76 Segment seg = segment(i);
77 if (seg.size == 0)
78 continue;
79 buf.insert(buf.end(), seg.data, seg.data + seg.size);
80 }
81 return MessageData(std::move(buf));
82}

107Segment SegmentedMessageDataCursor::remainingSegment() const {
108 // Scan forward past empty segments without mutating cursor state.
109 size_t idx = segIdx;
110 size_t off = offset;
111 while (idx < msg.numSegments()) {
112 Segment seg = msg.segment(idx);
113 if (seg.size > off)
114 return {seg.data + off, seg.size - off, seg.region};
115 ++idx;
116 off = 0;
117 }
118 return {nullptr, 0};
119}

121void SegmentedMessageDataCursor::advance(size_t n) {
122 while (n > 0 && !done()) {
123 Segment seg = msg.segment(segIdx);
124 size_t left = seg.size - offset;
125 if (left == 0) {
126 // Skip empty segments.
127 ++segIdx;
128 offset = 0;
129 continue;
130 }
131 if (n < left) {
132 offset += n;
133 return;
134 }
135 n -= left;
136 ++segIdx;
137 offset = 0;
138 }
139}
