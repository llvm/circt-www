23std::string MessageData::toHex() const {
24 std::ostringstream ss;
25 ss << std::hex;
26 for (size_t i = 0, e = data.size(); i != e; ++i) {
27 // Add spaces every 8 bytes.
28 if (i % 8 == 0 && i != 0)
29 ss << ' ';
30 // Add an extra space every 64 bytes.
31 if (i % 64 == 0 && i != 0)
32 ss << ' ';
33 ss << static_cast<unsigned>(data[i]);
34 }
35 return ss.str();
36}

52size_t SegmentedMessageData::totalSize() const {
53 size_t total = 0;
54 for (size_t i = 0, e = numSegments(); i < e; ++i)

61MessageData SegmentedMessageData::toMessageData() const {
62 size_t total = totalSize();
63 if (total == 0)
64 return MessageData();
65 std::vector<uint8_t> buf;
66 buf.reserve(total);
67 for (size_t i = 0, e = numSegments(); i < e; ++i) {
68 Segment seg = segment(i);
69 if (seg.size == 0)
70 continue;
71 buf.insert(buf.end(), seg.data, seg.data + seg.size);
72 }
73 return MessageData(std::move(buf));
74}

99Segment SegmentedMessageDataCursor::remainingSegment() const {
100 // Scan forward past empty segments without mutating cursor state.
101 size_t idx = segIdx;
102 size_t off = offset;
103 while (idx < msg.numSegments()) {
104 Segment seg = msg.segment(idx);
105 if (seg.size > off)
106 return {seg.data + off, seg.size - off, seg.region};
107 ++idx;
108 off = 0;
109 }
110 return {nullptr, 0};
111}

113void SegmentedMessageDataCursor::advance(size_t n) {
114 while (n > 0 && !done()) {
115 Segment seg = msg.segment(segIdx);
116 size_t left = seg.size - offset;
117 if (left == 0) {
118 // Skip empty segments.
119 ++segIdx;
120 offset = 0;
121 continue;
122 }
123 if (n < left) {
124 offset += n;
125 return;
126 }
127 n -= left;
128 ++segIdx;
129 offset = 0;
130 }
131}
