24/// Build an index of children by AppID.
25static std::map<AppID, Instance *>
26buildIndex(const std::vector<std::unique_ptr<Instance>> &insts) {
27 std::map<AppID, Instance *> index;
28 for (auto &item : insts)
29 index[item->getID()] = item.get();

31}

33/// Build an index of ports by AppID.
34static std::map<AppID, BundlePort &>
35buildIndex(const std::vector<std::unique_ptr<BundlePort>> &ports) {
36 std::map<AppID, BundlePort &> index;
37 for (auto &item : ports)
38 index.emplace(item->getID(), *item);
39 return index;

59const HWModule *HWModule::resolveInst(const AppIDPath &path,
60 AppIDPath &lastLookup) const {
61 const HWModule *hwmodule = this;
62 for (auto &id : path) {
63 lastLookup.push_back(id);
64 auto childIter = hwmodule->childIndex.find(id);
65 if (childIter == hwmodule->childIndex.end())
66 return nullptr;
67 hwmodule = childIter->second;
68 }
69 return hwmodule;
70}

72BundlePort *HWModule::resolvePort(const AppIDPath &path,
73 AppIDPath &lastLookup) const {
74 AppID portID = path.back();
75 AppIDPath instPath = path;
76 instPath.pop_back();
77 const HWModule *hwmodule = resolveInst(instPath, lastLookup);
78 lastLookup.push_back(portID);
79 if (!hwmodule)
80 return nullptr;
81 const auto &ports = hwmodule->getPorts();
82 const auto &portIter = ports.find(portID);
83 if (portIter == ports.end())
84 return nullptr;
85 return &portIter->second;
86}
