70static StringAttr constructUniqueSymbol(Operation *tableOp,
71 StringRef proposedNameRef) {
72 SmallString<64> proposedName = proposedNameRef;
73
74 // Normalize the type name.
75 for (char &ch : proposedName) {
76 if (isalpha(ch) || isdigit(ch) || ch == '_')
77 continue;
78 ch = '_';
79 }
80
81 // Make sure that this symbol isn't taken. If it is, append a number and try
82 // again.
83 size_t baseLength = proposedName.size();
84 size_t tries = 0;
85 while (SymbolTable::lookupSymbolIn(tableOp, proposedName)) {
86 proposedName.resize(baseLength);
87 proposedName.append(llvm::utostr(++tries));
88 }
89
90 return StringAttr::get(tableOp->getContext(), proposedName);
91}

104/// Write an 'ExternModuleOp' to use a hand-coded SystemVerilog module. Said
105/// module implements pipeline stage, adding 1 cycle latency. This particular
106/// implementation is double-buffered and fully pipelines the reverse-flow ready
107/// signal.
108HWModuleExternOp ESIHWBuilder::declareStage(Operation *symTable,
109 PipelineStageOp stage) {
110 Type dataType = stage.innerType();
111 HWModuleExternOp &stageMod = declaredStage[dataType];
112 if (stageMod)
113 return stageMod;
114
115 // Since this module has parameterized widths on the a input and x output,
116 // give the extern declation a None type since nothing else makes sense.
117 // Will be refining this when we decide how to better handle parameterized
118 // types and ops.
119 size_t argn = 0;
120 size_t resn = 0;
121 llvm::SmallVector<PortInfo> ports = {
122 {{clk, getClockType(), ModulePort::Direction::Input}, argn++},
123 {{rst, getI1Type(), ModulePort::Direction::Input}, argn++}};
124
125 ports.push_back({{a, dataType, ModulePort::Direction::Input}, argn++});
126 ports.push_back(
127 {{aValid, getI1Type(), ModulePort::Direction::Input}, argn++});
128 ports.push_back(
129 {{aReady, getI1Type(), ModulePort::Direction::Output}, resn++});
130 ports.push_back({{x, dataType, ModulePort::Direction::Output}, resn++});
131
132 ports.push_back(
133 {{xValid, getI1Type(), ModulePort::Direction::Output}, resn++});
134 ports.push_back(
135 {{xReady, getI1Type(), ModulePort::Direction::Input}, argn++});
136
137 stageMod = HWModuleExternOp::create(
138 *this, constructUniqueSymbol(symTable, "ESI_PipelineStage"), ports,
139 "ESI_PipelineStage", getStageParameterList({}, {}));
140 return stageMod;
141}

144/// module contains a bi-directional Cosimulation DPI interface with valid/ready

146HWModuleExternOp
147ESIHWBuilder::declareCosimEndpointToHostModule(Operation *symTable) {
148 if (declaredCosimEndpointToHostModule)
149 return *declaredCosimEndpointToHostModule;
150
151 SmallVector<Attribute, 8> params;
152 params.push_back(
153 ParamDeclAttr::get("ENDPOINT_ID", NoneType::get(getContext())));
154 params.push_back(
155 ParamDeclAttr::get("TO_HOST_TYPE_ID", NoneType::get(getContext())));
156 params.push_back(ParamDeclAttr::get("TO_HOST_SIZE_BITS", getI32Type()));
157
158 auto dataInType = hw::IntType::get(hw::ParamDeclRefAttr::get(
159 StringAttr::get(getContext(), "TO_HOST_SIZE_BITS"),
160 getIntegerType(32, false)));
161 PortInfo ports[] = {
162 {{clk, getClockType(), ModulePort::Direction::Input}, 0},
163 {{rst, getI1Type(), ModulePort::Direction::Input}, 1},
164 {{dataInValid, getI1Type(), ModulePort::Direction::Input}, 2},
165 {{dataInReady, getI1Type(), ModulePort::Direction::Output}, 3},
166 {{dataIn, dataInType, ModulePort::Direction::Input}, 4}};
167
168 declaredCosimEndpointToHostModule = HWModuleExternOp::create(
169 *this, constructUniqueSymbol(symTable, "Cosim_Endpoint_ToHost"), ports,
170 "Cosim_Endpoint_ToHost", ArrayAttr::get(getContext(), params));
171 return *declaredCosimEndpointToHostModule;
172}
