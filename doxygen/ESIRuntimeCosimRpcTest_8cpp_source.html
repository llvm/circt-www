100 /// Read one message from `port` as the simulator would. Fails the test
101 /// (and returns an arbitrary value) rather than hanging if none arrives.
102 uint32_t consumeFrom(ReadChannelPort &port) {
103 std::future<MessageData> f = port.readAsync();
104 if (f.wait_for(10s) != std::future_status::ready) {
105 ADD_FAILURE() << "timed out waiting for a message";
106 return ~0u;
107 }
108 return *f.get().as<uint32_t>();
109 }
110};

112TEST_F(CosimRpcTest, UnconnectedServerPortHoldsFrames) {
113 connectClient();
114 // Frames for a port the simulator hasn't connected yet are held, not
115 // dropped, and handed over in order once it connects.
116 for (uint32_t i = 0; i < 4; ++i)
117 client->writeToServer("late", msg(i));
118 // Make sure they've arrived: the server handles frames in order, so once a
119 // later frame on another channel has been delivered, these have too.
120 client->writeToServer("in", msg(99));
121 EXPECT_EQ(consumeOne(), 99u);
122
123 lateToServer->connect();
124 for (uint32_t i = 0; i < 4; ++i)
125 EXPECT_EQ(consumeFrom(*lateToServer), i);
126
127 // And again after a disconnect.
128 lateToServer->disconnect();
129 client->writeToServer("late", msg(10));
130 client->writeToServer("in", msg(98));
131 EXPECT_EQ(consumeOne(), 98u);
132 lateToServer->connect();
133 EXPECT_EQ(consumeFrom(*lateToServer), 10u);
134}

160TEST_F(CosimRpcTest, RefusedParkedFrameDoesNotBlockTheRest) {
161 connectClient();
162 for (uint32_t i = 0; i < 3; ++i)
163 client->writeToServer("late", msg(i));
164 client->writeToServer("in", msg(99));
165 EXPECT_EQ(consumeOne(), 99u);
166
167 // A callback-mode consumer which refuses the first frame it's offered.
168 std::mutex receivedMutex;
169 std::vector<uint32_t> received;
170 bool refused = false;
171 lateToServer->connect([&](MessageData data) -> bool {
172 std::lock_guard<std::mutex> lock(receivedMutex);
173 if (!refused) {
174 refused = true;
175 return false;
176 }
177 received.push_back(*data.as<uint32_t>());
178 return true;
179 });
180 // The refused frame is dropped, as one arriving while connected would be;
181 // the ones behind it still get through.
182 {
183 std::lock_guard<std::mutex> lock(receivedMutex);
184 EXPECT_EQ(received, (std::vector<uint32_t>{1, 2}));
185 }
186 // The callback refers to locals.
187 lateToServer->disconnect();
188}
