61TEST(HostMemRegionTest, GetDeviceAddress) {
62 FakeRegion r(0x1000, 0xA000'0000, 0x100);
63 EXPECT_EQ(r.getDeviceAddress(at(0x1000), 0x100), 0xA000'0000u);
64 EXPECT_EQ(r.getDeviceAddress(at(0x10FF), 1), 0xA000'00FFu);
65 // Empty, starting before, ending past, starting past, and wrapping ranges.
66 EXPECT_FALSE(r.getDeviceAddress(at(0x1000), 0));
67 EXPECT_FALSE(r.getDeviceAddress(at(0x0FFF), 2));
68 EXPECT_FALSE(r.getDeviceAddress(at(0x10FF), 2));
69 EXPECT_FALSE(r.getDeviceAddress(at(0x1100), 1));
70 EXPECT_FALSE(r.getDeviceAddress(at(0x1080), SIZE_MAX));
71
72 constexpr uintptr_t maxPtr = std::numeric_limits<uintptr_t>::max();
73 // Host range at the top of the address space.
74 FakeRegion top(maxPtr - 0xF, 0x2000, 0x10);
75 EXPECT_EQ(top.getDeviceAddress(at(maxPtr - 0x7), 8), 0x2008u);
76 EXPECT_FALSE(top.getDeviceAddress(at(maxPtr - 0x7), 9));
77
78 // Device range at the top of the address space.
79 if constexpr (sizeof(uintptr_t) == sizeof(uint64_t)) {
80 FakeRegion devTop(0x1000, maxPtr - 0x3, 0x10);
81 EXPECT_EQ(devTop.getDeviceAddress(at(0x1000), 4), maxPtr - 0x3);
82 EXPECT_FALSE(devTop.getDeviceAddress(at(0x1000), 5));
83 EXPECT_FALSE(devTop.getDeviceAddress(at(0x1004), 1));
84 }

97/// A header with no region, then two 4-byte segments sharing one region which
98/// the message owns and hands over via take() after both have been taken.
99struct TestMessage : public SegmentedMessageData {
100 TestMessage(bool *destroyed = nullptr)
101 : header{0xA0, 0xA1},
102 region(std::make_unique<BufferRegion>(8, 0xD000'0000, destroyed)) {
103 base = static_cast<const uint8_t *>(region->getPtr());
104 }
105
106 size_t numSegments() const override { return 3; }
107 Segment segment(size_t idx) const override {
108 if (taken.at(idx))
109 throw std::runtime_error("segment has been taken");
110 if (idx == 0)
111 return {header.data(), header.size()};
112 return {base + (idx - 1) * 4, 4, region.get()};
113 }
114 std::unique_ptr<HostMemRegion> take(size_t idx) override {
115 if (idx == 0)
116 return nullptr;
117 if (taken.at(idx))
118 throw std::runtime_error("segment has been taken");
119 taken[idx] = true;
120 return taken[1] && taken[2] ? std::move(region) : nullptr;
121 }
122
123 std::vector<uint8_t> header;
124 std::unique_ptr<HostMemRegion> region;
125 const uint8_t *base;
126 std::vector<bool> taken = {false, false, false};
127};

129TEST(SegmentTest, CursorRemainingSegmentKeepsRegion) {
130 TestMessage msg;
131 SegmentedMessageDataCursor cursor(msg);
132 EXPECT_EQ(cursor.remainingSegment().region, nullptr);
133
134 // A partial advance keeps the region; the device address follows the data.
135 cursor.advance(3);
136 Segment s = cursor.remainingSegment();
137 EXPECT_EQ(s.data, msg.base + 1);
138 EXPECT_EQ(s.size, 3u);
139 EXPECT_EQ(s.region, msg.region.get());

141 EXPECT_EQ(cursor.remaining().data(), s.data);
142

145 cursor.advance(4);

147 EXPECT_EQ(cursor.remainingSegment().data, nullptr);

150TEST(SegmentTest, TakeHandsOverSharedRegionAfterLastUser) {
151 bool destroyed = false;
152 auto msg = std::make_unique<TestMessage>(&destroyed);
153 HostMemRegion *shared = msg->region.get();
154
155 // Segments without a region are unaffected by take().
156 EXPECT_EQ(msg->take(0), nullptr);
157 EXPECT_EQ(msg->segment(0).size, 2u);
158
159 // Taking the first user marks it taken but keeps the shared region.
160 EXPECT_EQ(msg->take(1), nullptr);
161 EXPECT_THROW(msg->segment(1), std::runtime_error);
162 EXPECT_THROW(msg->take(1), std::runtime_error);
163 EXPECT_THROW(msg->toMessageData(), std::runtime_error);
164 EXPECT_EQ(msg->segment(2).region, shared);
165
166 // A cursor created after a take() throws upon reaching the taken segment.
167 SegmentedMessageDataCursor cursor(*msg);
168 EXPECT_THROW(cursor.advance(3), std::runtime_error);
169
170 // Taking the last user hands over the region, which outlives the message.
171 std::unique_ptr<HostMemRegion> region = msg->take(2);
172 EXPECT_EQ(region.get(), shared);
173 msg.reset();
174 EXPECT_FALSE(destroyed);
175 region.reset();
176 EXPECT_TRUE(destroyed);
177
178 // An untaken region is freed with its message.
179 destroyed = false;
180 { TestMessage untaken(&destroyed); }
181 EXPECT_TRUE(destroyed);
182}
