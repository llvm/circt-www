46TEST(AcceleratorServiceThreadTest, TaskExceptionsAreLogged) {
47 Context ctxt(std::make_unique<RecordingLogger>());
48 auto ownedLogger = std::make_unique<RecordingLogger>();
49 RecordingLogger &logger = *ownedLogger;
50 std::atomic<int> throwingRuns = 0, otherRuns = 0;
51 {
52 AcceleratorServiceThread serviceThread(&ctxt);
53 // The context's current logger is used, even if it was replaced.
54 ctxt.setLogger(std::move(ownedLogger));
55 serviceThread.addTask([&]() {
56 if (throwingRuns++ == 0)
57 throw std::runtime_error("task failure");
58 });
59 serviceThread.addTask([&]() { ++otherRuns; });
60
61 // The thread carries on running both tasks.
62 auto deadline = std::chrono::steady_clock::now() + std::chrono::seconds(10);
63 while ((throwingRuns < 2 || otherRuns < 2) &&
64 std::chrono::steady_clock::now() < deadline)
65 std::this_thread::sleep_for(std::chrono::milliseconds(1));
66 }
67 EXPECT_GE(throwingRuns, 2);
68 EXPECT_GE(otherRuns, 2);
69 std::vector<std::string> errors = logger.getErrors();
70 ASSERT_EQ(errors.size(), 1u);
71 EXPECT_NE(errors[0].find("task failure"), std::string::npos);
72}
