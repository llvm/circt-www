40class Emitter {
41public:
42 Emitter(llvm::raw_ostream &os, const DenseSet<StringAttr> &unsupportedInstr)
43 : os(os), unsupportedInstr(unsupportedInstr) {}
44
45 LogicalResult emitFile(emit::FileOp fileOp) {
46 auto res = emitBlock(fileOp.getBody());
47 state.clear();
48 return res;
49 }
50
51 LogicalResult emitBlock(Block *block) {
52 for (auto &op : *block) {
53 if (op.hasTrait<OpTrait::ConstantLike>()) {
54 SmallVector<OpFoldResult> results;
55 if (failed(op.fold(results)))
56 return failure();
57
58 for (auto [val, res] : llvm::zip(op.getResults(), results)) {
59 auto attr = res.dyn_cast<Attribute>();
60 if (!attr)
61 return failure();
62
63 state[val] = attr;
64 }
65
66 continue;
67 }
68
69 if (auto implicitConstrOp =
70 dyn_cast<ImplicitConstraintOpInterface>(&op)) {
71 if (!implicitConstrOp.isConstraintMaterialized())
72 return op.emitError("implicit constraint not materialized");
73 }
74
75 auto res =
76 TypeSwitch<Operation *, LogicalResult>(&op)
77 .Case<InstructionOpInterface, LabelOp, CommentOp, SpaceOp,
78 StringDataOp, SegmentOp>([&](auto op) { return emit(op); })
79 .Default([](auto op) {
80 return op->emitError("emitter unknown RTG operation");
81 });
82
83 if (failed(res))
84 return failure();
85 }
86
87 return success();
88 }
89
90private:
91 LogicalResult emit(InstructionOpInterface instr) {
92 os << llvm::indent(4);
93 bool useBinary =
94 unsupportedInstr.contains(instr->getName().getIdentifier());
95
96 // TODO: we cannot just assume that double-slash is the way to do a line
97 // comment
98 if (useBinary)
99 os << "# ";
100
101 SmallVector<Attribute> operands;
102 for (auto operand : instr->getOperands()) {
103 if (isa<LabelType>(operand.getType()) && useBinary)
104 return instr->emitError("labels cannot be emitted as binary");
105
106 auto attr = state.lookup(operand);
107 if (!attr)
108 return failure();
109
110 operands.push_back(attr);
111 }
112
113 instr.printInstructionAssembly(os, operands);
114 os << "\n";
115
116 if (!useBinary)
117 return success();
118
119 os << llvm::indent(4);
120 // TODO: don't hardcode '.word'
121 os << ".word 0x";
122 instr.printInstructionBinary(os, operands);
123 os << "\n";
124
125 return success();
126 }
127
128 LogicalResult emit(LabelOp op) {
129 auto labelStr = cast<LabelAttr>(state[op.getLabel()]).getName();
130 if (op.getVisibility() == LabelVisibility::external) {
131 os << ".extern " << labelStr << "\n";
132 return success();
133 }
134
135 if (op.getVisibility() == LabelVisibility::global)
136 os << ".global " << labelStr << "\n";
137
138 os << labelStr << ":\n";
139 return success();
140 }
141
142 LogicalResult emit(CommentOp op) {
143 os << llvm::indent(4) << "# "
144 << cast<StringAttr>(state[op.getComment()]).getValue() << "\n";
145 return success();
146 }
147
148 LogicalResult emit(SpaceOp op) {
149 os << llvm::indent(4) << ".space "
150 << cast<IntegerAttr>(state[op.getSize()]).getValue() << "\n";
151 return success();
152 }
153
154 LogicalResult emit(StringDataOp op) {
155 os << llvm::indent(4) << ".asciz \"";
156 // NOTE: llvm::printEscapedString does not work because assemblers do not
157 // support the hex escapes (e.g., \0a instead of \n)
158 for (auto c : cast<StringAttr>(state[op.getData()]).getValue()) {
159 switch (c) {
160 case '\n':
161 os << "\\n";
162 break;
163 case '\t':
164 os << "\\t";
165 break;
166 case '\\':
167 os << "\\\\";
168 break;
169 case '"':
170 os << "\\\"";
171 break;
172 default:
173 os << c;
174 break;
175 }
176 }
177 os << "\"\n";
178 return success();
179 }
180
181 LogicalResult emit(SegmentOp op) {
182 os << "." << op.getKind() << "\n";
183 return emitBlock(op.getBody());
184 }
185
186private:
187 /// Output Stream.
188 llvm::raw_ostream &os;
189
190 /// Instructions to emit in binary.
191 const DenseSet<StringAttr> &unsupportedInstr;
192
193 /// Evaluated values.
194 DenseMap<Value, Attribute> state;
195};

199static void
200parseUnsupportedInstructionsFile(MLIRContext *ctxt,
201 const std::string &unsupportedInstructionsFile,
202 DenseSet<StringAttr> &unsupportedInstrs) {
203 if (!unsupportedInstructionsFile.empty()) {
204 std::ifstream input(unsupportedInstructionsFile, std::ios::in);
205 std::string token;
206 while (std::getline(input, token, ',')) {
207 auto trimmed = StringRef(token).trim();
208 if (!trimmed.empty())
209 unsupportedInstrs.insert(StringAttr::get(ctxt, trimmed));
210 }
211 }
212}

226void EmitRTGISAAssemblyPass::runOnOperation() {
227 // Get the set of instructions not supported by the assembler
228 DenseSet<StringAttr> unsupportedInstr;
229 for (const auto &instr : unsupportedInstructions)
230 unsupportedInstr.insert(StringAttr::get(&getContext(), instr));
231 parseUnsupportedInstructionsFile(
232 &getContext(), unsupportedInstructionsFile.getValue(), unsupportedInstr);
233
234 // Create the output file
235 auto filename = getOperation().getFileName();
236 std::unique_ptr<llvm::ToolOutputFile> file;
237 bool emitToFile = !filename.empty() && filename != "-";
238 if (emitToFile) {
239 file = createOutputFile(filename, std::string(),
240 [&]() { return getOperation().emitError(); });
241 if (!file)
242 return signalPassFailure();
243
244 file->keep();
245 }
246
247 Emitter emitter(emitToFile ? file->os()
248 : (filename.empty() ? llvm::errs() : llvm::outs()),
249 unsupportedInstr);
250 if (failed(emitter.emitFile(getOperation())))
251 return signalPassFailure();
252}
