43/// Base class for evaluator runtime values.
44/// Enables the shared_from_this functionality so Evaluator Value pointers can
45/// be passed through the CAPI and unwrapped back into C++ smart pointers with
46/// the appropriate reference count.
47class EvaluatorValue : public std::enable_shared_from_this<EvaluatorValue> {
48public:
49 // Implement LLVM RTTI.
50 enum class Kind { Attr, Object, List, Option, BasePath, Path };
51 EvaluatorValue(MLIRContext *ctx, Kind kind, Location loc)
52 : kind(kind), ctx(ctx), loc(loc) {}
53 Kind getKind() const { return kind; }
54 MLIRContext *getContext() const { return ctx; }
55
56 // Return true the value is fully evaluated.
57 // Unknown values are considered fully evaluated.
58 bool isFullyEvaluated() const { return fullyEvaluated; }
59 void markFullyEvaluated() {
60 assert(!fullyEvaluated && "should not mark twice");
61 fullyEvaluated = true;
62 }
63
64 /// Return true if the value itself is unknown. This does not recursively
65 /// inspect object fields or container payloads.
66 bool isUnknown() const { return unknown; }
67
68 /// Mark this value as unknown.
69 /// This also marks the value as fully evaluated if it isn't already, since
70 /// unknown values are considered fully evaluated. This maintains the
71 /// invariant that unknown implies fullyEvaluated.
72 void markUnknown() {
73 unknown = true;
74 if (!fullyEvaluated)
75 markFullyEvaluated();
76 }
77
78 /// Return the associated MLIR context.
79 MLIRContext *getContext() { return ctx; }
80
81 // Return a MLIR type which the value represents.
82 Type getType() const;
83
84 // Return the Location associated with the Value.
85 Location getLoc() const { return loc; }
86 // Set the Location associated with the Value.
87 void setLoc(Location l) { loc = l; }
88 // Set the Location, if it is unknown.
89 void setLocIfUnknown(Location l) {
90 if (isa<UnknownLoc>(loc))
91 loc = l;
92 }
93
94private:
95 const Kind kind;
96 MLIRContext *ctx;
97 Location loc;
98 bool fullyEvaluated = false;
99 bool unknown = false;
100};

440/// Helper to enable printing objects in Diagnostics.
441static inline mlir::Diagnostic &
442operator<<(mlir::Diagnostic &diag,
443 const evaluator::EvaluatorValue &evaluatorValue) {
444 if (auto *attr = llvm::dyn_cast<evaluator::AttributeValue>(&evaluatorValue))
445 diag << attr->getAttr();
446 else if (auto *object =
447 llvm::dyn_cast<evaluator::ObjectValue>(&evaluatorValue))
448 diag << "Object(" << object->getType() << ")";
449 else if (auto *list = llvm::dyn_cast<evaluator::ListValue>(&evaluatorValue))
450 diag << "List(" << list->getType() << ")";
451 else if (auto *option =
452 llvm::dyn_cast<evaluator::OptionValue>(&evaluatorValue))
453 diag << "Option(" << option->getType() << ")";
454 else if (llvm::isa<evaluator::BasePathValue>(&evaluatorValue))
455 diag << "BasePath()";
456 else if (llvm::isa<evaluator::PathValue>(&evaluatorValue))
457 diag << "Path()";
458 else
459 assert(false && "unhandled evaluator value");
460
461 // Add unknown marker if the value is unknown
462 if (evaluatorValue.isUnknown())
463 diag << " [unknown]";
464 return diag;
465}
