42/// This holds the name and type that describes the module's ports.
43struct PortInfo {
44 StringAttr name;
45 Type type;
46 Direction direction;
47 hw::InnerSymAttr sym = {};
48 Location loc = UnknownLoc::get(type.getContext());
49 AnnotationSet annotations = AnnotationSet(type.getContext());
50 Attribute domains;
51
52 StringRef getName() const { return name ? name.getValue() : ""; }
53
54 /// Return true if this is a simple output-only port. If you want the
55 /// direction of the port, use the \p direction parameter.
56 bool isOutput() const { return direction == Direction::Out && !isInOut(); }
57
58 /// Return true if this is a simple input-only port. If you want the
59 /// direction of the port, use the \p direction parameter.
60 bool isInput() const { return direction == Direction::In && !isInOut(); }
61
62 /// Return true if this is an inout port. This will be true if the port
63 /// contains either bi-directional signals or analog types.
64 /// Non-HW types (e.g., ref types) are never considered InOut.
65 bool isInOut() const { return isTypeInOut(type); }
66
67 /// Default constructors
68 PortInfo(StringAttr name, Type type, Direction dir, StringAttr symName = {},
69 std::optional<Location> location = {},
70 std::optional<AnnotationSet> annos = {},
71 std::optional<Attribute> domains = {})
72 : name(name), type(type), direction(dir) {
73 if (symName)
74 sym = hw::InnerSymAttr::get(symName);
75 if (location)
76 loc = *location;
77 if (annos)
78 annotations = *annos;
79 if (domains)
80 this->domains = *domains;
81 };
82 PortInfo(StringAttr name, Type type, Direction dir, hw::InnerSymAttr sym,
83 Location loc, AnnotationSet annos, Attribute domains)
84 : name(name), type(type), direction(dir), sym(sym), loc(loc),
85 annotations(annos), domains(domains) {}
86};
