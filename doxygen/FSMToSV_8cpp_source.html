33/// Get the port info of a FSM machine. Clock and reset port are also added.
34namespace {
35struct ClkRstIdxs {
36 size_t clockIdx;
37 size_t resetIdx;
38};
39} // namespace
40static ClkRstIdxs getMachinePortInfo(SmallVectorImpl<hw::PortInfo> &ports,
41 MachineOp machine, OpBuilder &b) {
42 // Get the port info of the machine inputs and outputs.
43 machine.getHWPortInfo(ports);
44 ClkRstIdxs specialPorts;
45
46 // Add clock port.
47 hw::PortInfo clock;
48 clock.name = b.getStringAttr("clk");
49 clock.dir = hw::ModulePort::Direction::Input;
50 clock.type = seq::ClockType::get(b.getContext());
51 clock.argNum = machine.getNumArguments();
52 ports.push_back(clock);
53 specialPorts.clockIdx = clock.argNum;
54
55 // Add reset port.
56 hw::PortInfo reset;
57 reset.name = b.getStringAttr("rst");
58 reset.dir = hw::ModulePort::Direction::Input;
59 reset.type = b.getI1Type();
60 reset.argNum = machine.getNumArguments() + 1;
61 ports.push_back(reset);
62 specialPorts.resetIdx = reset.argNum;
63
64 return specialPorts;
65}

67// Clones constants implicitly captured by the region, into the region.
68static void cloneConstantsIntoRegion(Region &region, OpBuilder &builder) {
69 // Values implicitly captured by the region.
70 llvm::SetVector<Value> captures;
71 getUsedValuesDefinedAbove(region, region, captures);
72
73 OpBuilder::InsertionGuard guard(builder);
74 builder.setInsertionPointToStart(&region.front());
75
76 // Clone ConstantLike operations into the region.
77 for (auto &capture : captures) {
78 Operation *op = capture.getDefiningOp();
79 if (!op || !op->hasTrait<OpTrait::ConstantLike>())
80 continue;
81
82 Operation *cloned = builder.clone(*op);
83 for (auto [orig, replacement] :
84 llvm::zip(op->getResults(), cloned->getResults()))
85 replaceAllUsesInRegionWith(orig, replacement, region);
86 }
87}

91class StateEncoding {
92 // An class for handling state encoding. The class is designed to
93 // abstract away how states are selected in case patterns, referred to as
94 // values, and used as selection signals for muxes.
95
96public:
97 StateEncoding(OpBuilder &b, hw::TypeScopeOp typeScope, MachineOp machine,
98 hw::HWModuleOp hwModule);
99
100 // Get the encoded value for a state.
101 Value encode(StateOp state);
102 // Get the state corresponding to an encoded value.
103 StateOp decode(Value value);
104
105 // Returns the type which encodes the state values.
106 Type getStateType() { return stateType; }
107
108 // Returns a case pattern which matches the provided state.
109 std::unique_ptr<sv::CasePattern> getCasePattern(StateOp state);
110
111protected:
112 // Creates a constant value in the module for the given encoded state
113 // and records the state value in the mappings. An inner symbol is
114 // attached to the wire to avoid it being optimized away.
115 // The constant can optionally be assigned behind a sv wire - doing so at this
116 // point ensures that constants don't end up behind "_GEN#" wires in the
117 // module.
118 void setEncoding(StateOp state, Value v, bool wire = false);
119
120 // A mapping between a StateOp and its corresponding encoded value.
121 SmallDenseMap<StateOp, Value> stateToValue;
122
123 // A mapping between an encoded value and its corresponding StateOp.
124 SmallDenseMap<Value, StateOp> valueToState;
125
126 // A mapping between an encoded value and the source value in the IR.
127 SmallDenseMap<Value, Value> valueToSrcValue;
128
129 // A typescope to emit the FSM enum type within.
130 hw::TypeScopeOp typeScope;
131
132 // The enum type for the states.
133 Type stateType;
134
135 OpBuilder &b;
136 MachineOp machine;
137 hw::HWModuleOp hwModule;
138};

140StateEncoding::StateEncoding(OpBuilder &b, hw::TypeScopeOp typeScope,
141 MachineOp machine, hw::HWModuleOp hwModule)
142 : typeScope(typeScope), b(b), machine(machine), hwModule(hwModule) {
143 Location loc = machine.getLoc();
144 llvm::SmallVector<Attribute> stateNames;
145
146 for (auto state : machine.getBody().getOps<StateOp>())
147 stateNames.push_back(b.getStringAttr(state.getName()));
148
149 // Create an enum typedef for the states.
150 Type rawEnumType =
151 hw::EnumType::get(b.getContext(), b.getArrayAttr(stateNames));
152
153 OpBuilder::InsertionGuard guard(b);
154 b.setInsertionPointToStart(&typeScope.getBodyRegion().front());
155 auto typedeclEnumType = hw::TypedeclOp::create(
156 b, loc, b.getStringAttr(hwModule.getName() + "_state_t"),
157 TypeAttr::get(rawEnumType), nullptr);
158
159 stateType = hw::TypeAliasType::get(
160 SymbolRefAttr::get(typeScope.getSymNameAttr(),
161 {FlatSymbolRefAttr::get(typedeclEnumType)}),
162 rawEnumType);
163
164 // And create enum values for the states
165 b.setInsertionPointToStart(&hwModule.getBody().front());
166 for (auto state : machine.getBody().getOps<StateOp>()) {
167 auto fieldAttr = hw::EnumFieldAttr::get(
168 loc, b.getStringAttr(state.getName()), stateType);
169 auto enumConstantOp = hw::EnumConstantOp::create(
170 b, loc, fieldAttr.getType().getValue(), fieldAttr);
171 setEncoding(state, enumConstantOp,
172 /*wire=*/true);
173 }
174}

176// Get the encoded value for a state.
177Value StateEncoding::encode(StateOp state) {
178 auto it = stateToValue.find(state);
179 assert(it != stateToValue.end() && "state not found");
180 return it->second;
181}
182// Get the state corresponding to an encoded value.
183StateOp StateEncoding::decode(Value value) {
184 auto it = valueToState.find(value);
185 assert(it != valueToState.end() && "encoded state not found");
186 return it->second;
187}

189// Returns a case pattern which matches the provided state.
190std::unique_ptr<sv::CasePattern> StateEncoding::getCasePattern(StateOp state) {
191 // Get the field attribute for the state - fetch it through the encoding.
192 auto fieldAttr =
193 cast<hw::EnumConstantOp>(valueToSrcValue[encode(state)].getDefiningOp())
194 .getFieldAttr();
195 return std::make_unique<sv::CaseEnumPattern>(fieldAttr);
196}

198void StateEncoding::setEncoding(StateOp state, Value v, bool wire) {
199 assert(stateToValue.find(state) == stateToValue.end() &&
200 "state already encoded");
201
202 Value encodedValue;
203 if (wire) {
204 auto loc = machine.getLoc();
205 auto stateType = getStateType();
206 auto stateEncodingWire = sv::RegOp::create(
207 b, loc, stateType, b.getStringAttr("to_" + state.getName()),
208 hw::InnerSymAttr::get(state.getNameAttr()));
209 sv::AssignOp::create(b, loc, stateEncodingWire, v);
210 encodedValue = sv::ReadInOutOp::create(b, loc, stateEncodingWire);
211 } else
212 encodedValue = v;
213 stateToValue[state] = encodedValue;
214 valueToState[encodedValue] = state;
215 valueToSrcValue[encodedValue] = v;
216}

218class MachineOpConverter {
219public:
220 MachineOpConverter(OpBuilder &builder, hw::TypeScopeOp typeScope,
221 MachineOp machineOp, FlatSymbolRefAttr headerName)
222 : machineOp(machineOp), typeScope(typeScope), b(builder),
223 headerName(headerName) {}
224
225 // Converts the machine op to a hardware module.
226 // 1. Creates a HWModuleOp for the machine op, with the same I/O as the FSM +
227 // clk/reset ports.
228 // 2. Creates a state register + encodings for the states visible in the
229 // machine.
230 // 3. Iterates over all states in the machine
231 // 3.1. Moves all `comb` logic into the body of the HW module
232 // 3.2. Records the SSA value(s) associated to the output ports in the state
233 // 3.3. iterates of the transitions of the state
234 // 3.3.1. Moves all `comb` logic in the transition guard/action regions to
235 // the body of the HW module.
236 // 3.3.2. Creates a case pattern for the transition guard
237 // 3.4. Creates a next-state value for the state based on the transition
238 // guards.
239 // 4. Assigns next-state values for the states in a case statement on the
240 // state reg.
241 // 5. Assigns the current-state outputs for the states in a case statement
242 // on the state reg.
243 LogicalResult dispatch();
244
245private:
246 struct StateConversionResult {
247 // Value of the next state output signal of the converted state.
248 Value nextState;
249 // Value of the output signals of the converted state.
250 llvm::SmallVector<Value> outputs;
251 };
252
253 using StateConversionResults = DenseMap<StateOp, StateConversionResult>;
254
255 // Converts a StateOp within this machine, and returns the value corresponding
256 // to the next-state output of the op.
257 FailureOr<StateConversionResult> convertState(StateOp state);
258
259 // Converts the outgoing transitions of a state and returns the value
260 // corresponding to the next-state output of the op.
261 // Transitions are priority encoded in the order which they appear in the
262 // state transition region.
263 FailureOr<Value> convertTransitions(StateOp currentState,
264 ArrayRef<TransitionOp> transitions);
265
266 // Moves operations from 'block' into module scope, failing if any op were
267 // deemed illegal. Returns the final op in the block if the op was a
268 // terminator. An optional 'exclude' filer can be provided to dynamically
269 // exclude some ops from being moved.
270 FailureOr<Operation *>
271 moveOps(Block *block,
272 llvm::function_ref<bool(Operation *)> exclude = nullptr);
273
274 struct CaseMuxItem;
275 using StateCaseMapping =
276 llvm::SmallDenseMap<StateOp,
277 std::variant<Value, std::shared_ptr<CaseMuxItem>>>;
278 struct CaseMuxItem {
279 // The target wire to be assigned.
280 sv::RegOp wire;
281
282 // The case select signal to be used.
283 Value select;
284
285 // A mapping between a state and an assignment within that state.
286 // An assignment can either be a value or a nested CaseMuxItem. The latter
287 // case will create nested case statements.
288 StateCaseMapping assignmentInState;
289
290 // An optional default value to be assigned before the case statement, if
291 // the case is not fully specified for all states.
292 std::optional<Value> defaultValue = {};
293 };
294
295 // Build an SV-based case mux for the given assignments. Assignments are
296 // merged into the same case statement. Caller is expected to ensure that the
297 // insertion point is within an `always_...` block.
298 void buildStateCaseMux(llvm::MutableArrayRef<CaseMuxItem> assignments);
299
300 // A handle to the state encoder for this machine.
301 std::unique_ptr<StateEncoding> encoding;
302
303 // A deterministic ordering of the states in this machine.
304 llvm::SmallVector<StateOp> orderedStates;
305
306 // A mapping from a fsm.variable op to its register.
307 llvm::SmallDenseMap<VariableOp, seq::CompRegOp> variableToRegister;
308
309 // A mapping from a state to variable updates performed during outgoing state
310 // transitions.
311 llvm::SmallDenseMap<
312 /*currentState*/ StateOp,
313 llvm::SmallDenseMap<
314 /*targetState*/ StateOp,
315 llvm::DenseMap</*targetVariable*/ VariableOp, /*targetValue*/ Value>>>
316 stateToVariableUpdates;
317
318 // A handle to the MachineOp being converted.
319 MachineOp machineOp;
320 // A handle to the HW ModuleOp being created.
321 hw::HWModuleOp hwModuleOp;
322
323 // A handle to the state register of the machine.
324 seq::CompRegOp stateReg;
325
326 // A typescope to emit the FSM enum type within.
327 hw::TypeScopeOp typeScope;
328
329 OpBuilder &b;
330
331 // The name of the header which contains the type scope for the machine.
332 FlatSymbolRefAttr headerName;
333};

335FailureOr<Operation *>
336MachineOpConverter::moveOps(Block *block,
337 llvm::function_ref<bool(Operation *)> exclude) {
338 for (auto &op : llvm::make_early_inc_range(*block)) {
339 if (!isa<comb::CombDialect, hw::HWDialect, fsm::FSMDialect>(
340 op.getDialect()))
341 return op.emitOpError()
342 << "is unsupported (op from the "
343 << op.getDialect()->getNamespace() << " dialect).";
344
345 if (exclude && exclude(&op))
346 continue;
347
348 if (op.hasTrait<OpTrait::IsTerminator>())
349 return &op;
350
351 op.moveBefore(hwModuleOp.getBodyBlock(), b.getInsertionPoint());
352 }
353 return nullptr;
354}

356void MachineOpConverter::buildStateCaseMux(
357 llvm::MutableArrayRef<CaseMuxItem> assignments) {
358
359 // Gather the select signal. All assignments are expected to use the same
360 // select signal.
361 Value select = assignments.front().select;
362 assert(llvm::all_of(
363 assignments,
364 [&](const CaseMuxItem &item) { return item.select == select; }) &&
365 "All assignments must use the same select signal.");
366
367 sv::CaseOp caseMux;
368 // Default assignments.
369 for (auto &assignment : assignments) {
370 if (assignment.defaultValue)
371 sv::BPAssignOp::create(b, assignment.wire.getLoc(), assignment.wire,
372 *assignment.defaultValue);
373 }
374
375 // Case assignments.
376 caseMux = sv::CaseOp::create(
377 b, machineOp.getLoc(), CaseStmtType::CaseStmt,
378 /*sv::ValidationQualifierTypeEnum::ValidationQualifierUnique, */ select,
379 /*numCases=*/machineOp.getNumStates() + 1, [&](size_t caseIdx) {
380 // Make Verilator happy for sized enums.
381 if (caseIdx == machineOp.getNumStates())
382 return std::unique_ptr<sv::CasePattern>(
383 new sv::CaseDefaultPattern(b.getContext()));
384 StateOp state = orderedStates[caseIdx];
385 return encoding->getCasePattern(state);
386 });
387
388 // Create case assignments.
389 for (auto assignment : assignments) {
390 OpBuilder::InsertionGuard g(b);
391 for (auto [caseInfo, stateOp] :
392 llvm::zip(caseMux.getCases(), orderedStates)) {
393 auto assignmentInState = assignment.assignmentInState.find(stateOp);
394 if (assignmentInState == assignment.assignmentInState.end())
395 continue;
396 b.setInsertionPointToEnd(caseInfo.block);
397 if (auto v = std::get_if<Value>(&assignmentInState->second); v) {
398 sv::BPAssignOp::create(b, machineOp.getLoc(), assignment.wire, *v);
399 } else {
400 // Nested case statement.
401 llvm::SmallVector<CaseMuxItem, 4> nestedAssignments;
402 nestedAssignments.push_back(
403 *std::get<std::shared_ptr<CaseMuxItem>>(assignmentInState->second));
404 buildStateCaseMux(nestedAssignments);
405 }
406 }
407 }
408}

410LogicalResult MachineOpConverter::dispatch() {
411 b.setInsertionPoint(machineOp);
412 auto loc = machineOp.getLoc();
413
414 // Clone all referenced constants into the machine body - constants may have
415 // been moved to the machine parent due to the lack of IsolationFromAbove.
416 cloneConstantsIntoRegion(machineOp.getBody(), b);
417
418 // 1) Get the port info of the machine and create a new HW module for it.
419 SmallVector<hw::PortInfo, 16> ports;
420 auto clkRstIdxs = getMachinePortInfo(ports, machineOp, b);
421 hwModuleOp =
422 hw::HWModuleOp::create(b, loc, machineOp.getSymNameAttr(), ports);
423 hwModuleOp->setAttr(emit::getFragmentsAttrName(),
424 b.getArrayAttr({headerName}));
425 b.setInsertionPointToStart(hwModuleOp.getBodyBlock());
426
427 // Replace all uses of the machine arguments with the arguments of the
428 // new created HW module.
429 for (auto args : llvm::zip(machineOp.getArguments(),
430 hwModuleOp.getBodyBlock()->getArguments())) {
431 auto machineArg = std::get<0>(args);
432 auto hwModuleArg = std::get<1>(args);
433 machineArg.replaceAllUsesWith(hwModuleArg);
434 }
435
436 auto clock = hwModuleOp.getBodyBlock()->getArgument(clkRstIdxs.clockIdx);
437 auto reset = hwModuleOp.getBodyBlock()->getArgument(clkRstIdxs.resetIdx);
438
439 // 2) Build state and variable registers.
440 encoding =
441 std::make_unique<StateEncoding>(b, typeScope, machineOp, hwModuleOp);
442 auto stateType = encoding->getStateType();
443
444 auto nextStateWire =
445 sv::RegOp::create(b, loc, stateType, b.getStringAttr("state_next"));
446 auto nextStateWireRead = sv::ReadInOutOp::create(b, loc, nextStateWire);
447 stateReg = seq::CompRegOp::create(
448 b, loc, nextStateWireRead, clock, reset,
449 /*reset value=*/encoding->encode(machineOp.getInitialStateOp()),
450 "state_reg");
451
452 llvm::DenseMap<VariableOp, sv::RegOp> variableNextStateWires;
453 for (auto variableOp : machineOp.front().getOps<fsm::VariableOp>()) {
454 auto initValueAttr = dyn_cast<IntegerAttr>(variableOp.getInitValueAttr());
455 if (!initValueAttr)
456 return variableOp.emitOpError() << "expected an integer attribute "
457 "for the initial value.";
458 Type varType = variableOp.getType();
459 auto varLoc = variableOp.getLoc();
460 auto varNextState = sv::RegOp::create(
461 b, varLoc, varType, b.getStringAttr(variableOp.getName() + "_next"));
462 auto varResetVal = hw::ConstantOp::create(b, varLoc, initValueAttr);
463 auto variableReg = seq::CompRegOp::create(
464 b, varLoc, sv::ReadInOutOp::create(b, varLoc, varNextState), clock,
465 reset, varResetVal, b.getStringAttr(variableOp.getName() + "_reg"));
466 variableToRegister[variableOp] = variableReg;
467 variableNextStateWires[variableOp] = varNextState;
468 // Postpone value replacement until all logic has been created.
469 // fsm::UpdateOp's require their target variables to refer to a
470 // fsm::VariableOp - if this is not the case, they'll throw an assert.
471 }
472
473 // Move any operations at the machine-level scope, excluding state ops, which
474 // are handled separately.
475 if (failed(moveOps(&machineOp.front(), [](Operation *op) {
476 return isa<fsm::StateOp, fsm::VariableOp>(op);
477 })))
478 return failure();
479
480 // 3) Convert states and record their next-state value assignments.
481 StateCaseMapping nextStateFromState;
482 StateConversionResults stateConvResults;
483 for (auto state : machineOp.getBody().getOps<StateOp>()) {
484 auto stateConvRes = convertState(state);
485 if (failed(stateConvRes))
486 return failure();
487
488 stateConvResults[state] = *stateConvRes;
489 orderedStates.push_back(state);
490 nextStateFromState[state] = {stateConvRes->nextState};
491 }
492
493 // 4/5) Create next-state assignments for each output.
494 llvm::SmallVector<CaseMuxItem, 4> outputCaseAssignments;
495 auto hwPortList = hwModuleOp.getPortList();
496 size_t portIndex = 0;
497 for (auto &port : hwPortList) {
498 if (!port.isOutput())
499 continue;
500 auto outputPortType = port.type;
501 CaseMuxItem outputAssignment;
502 outputAssignment.wire = sv::RegOp::create(
503 b, machineOp.getLoc(), outputPortType,
504 b.getStringAttr("output_" + std::to_string(portIndex)));
505 outputAssignment.select = stateReg;
506 for (auto &state : orderedStates)
507 outputAssignment.assignmentInState[state] = {
508 stateConvResults[state].outputs[portIndex]};
509
510 outputCaseAssignments.push_back(outputAssignment);
511 ++portIndex;
512 }
513
514 // Create next-state maps for the FSM variables.
515 llvm::DenseMap<VariableOp, CaseMuxItem> variableCaseMuxItems;
516 for (auto &[currentState, it] : stateToVariableUpdates) {
517 for (auto &[targetState, it2] : it) {
518 for (auto &[variableOp, targetValue] : it2) {
519 auto caseMuxItemIt = variableCaseMuxItems.find(variableOp);
520 if (caseMuxItemIt == variableCaseMuxItems.end()) {
521 // First time seeing this variable. Initialize the outer case
522 // statement. The outer case has a default assignment to the current
523 // value of the variable register.
524 variableCaseMuxItems[variableOp];
525 caseMuxItemIt = variableCaseMuxItems.find(variableOp);
526 assert(variableOp);
527 assert(variableNextStateWires.count(variableOp));
528 caseMuxItemIt->second.wire = variableNextStateWires[variableOp];
529 caseMuxItemIt->second.select = stateReg;
530 caseMuxItemIt->second.defaultValue =
531 variableToRegister[variableOp].getResult();
532 }
533
534 if (!std::get_if<std::shared_ptr<CaseMuxItem>>(
535 &caseMuxItemIt->second.assignmentInState[currentState])) {
536 // Initialize the inner case statement. This is an inner case within
537 // the current state, switching on the next-state value.
538 CaseMuxItem innerCaseMuxItem;
539 innerCaseMuxItem.wire = caseMuxItemIt->second.wire;
540 innerCaseMuxItem.select = nextStateWireRead;
541 caseMuxItemIt->second.assignmentInState[currentState] = {
542 std::make_shared<CaseMuxItem>(innerCaseMuxItem)};
543 }
544
545 // Append to the nested case mux for the variable, with a case select
546 // on the next-state signal.
547 // Append an assignment in the case that nextState == targetState.
548 auto &innerCaseMuxItem = std::get<std::shared_ptr<CaseMuxItem>>(
549 caseMuxItemIt->second.assignmentInState[currentState]);
550 innerCaseMuxItem->assignmentInState[targetState] = {targetValue};
551 }
552 }
553 }
554
555 // Materialize the case mux.
556 llvm::SmallVector<CaseMuxItem, 4> nextStateCaseAssignments;
557 nextStateCaseAssignments.push_back(
558 CaseMuxItem{nextStateWire, stateReg, nextStateFromState});
559 for (auto &[_, caseMuxItem] : variableCaseMuxItems)
560 nextStateCaseAssignments.push_back(caseMuxItem);
561 nextStateCaseAssignments.append(outputCaseAssignments.begin(),
562 outputCaseAssignments.end());
563
564 {
565 auto alwaysCombOp = sv::AlwaysCombOp::create(b, loc);
566 OpBuilder::InsertionGuard g(b);
567 b.setInsertionPointToStart(alwaysCombOp.getBodyBlock());
568 buildStateCaseMux(nextStateCaseAssignments);
569 }
570
571 // Replace variable values with their register counterparts.
572 for (auto &[variableOp, variableReg] : variableToRegister)
573 variableOp.getResult().replaceAllUsesWith(variableReg);
574
575 // Assing output ports.
576 llvm::SmallVector<Value> outputPortAssignments;
577 for (auto outputAssignment : outputCaseAssignments)
578 outputPortAssignments.push_back(
579 sv::ReadInOutOp::create(b, machineOp.getLoc(), outputAssignment.wire));
580
581 // Delete the default created output op and replace it with the output
582 // muxes.
583 auto *oldOutputOp = hwModuleOp.getBodyBlock()->getTerminator();
584 hw::OutputOp::create(b, loc, outputPortAssignments);
585 oldOutputOp->erase();
586
587 // Erase the original machine op.
588 machineOp.erase();
589
590 return success();
591}

593FailureOr<Value>
594MachineOpConverter::convertTransitions( // NOLINT(misc-no-recursion)
595 StateOp currentState, ArrayRef<TransitionOp> transitions) {
596 Value nextState;
597 if (transitions.empty()) {
598 // Base case
599 // State: transition to the current state.
600 nextState = encoding->encode(currentState);
601 } else {
602 // Recursive case - transition to a named state.
603 auto transition = cast<fsm::TransitionOp>(transitions.front());
604 nextState = encoding->encode(transition.getNextStateOp());
605
606 // Action conversion
607 if (transition.hasAction()) {
608 // Move any ops from the action region to the general scope, excluding
609 // variable update ops.
610 auto actionMoveOpsRes =
611 moveOps(&transition.getAction().front(),
612 [](Operation *op) { return isa<fsm::UpdateOp>(op); });
613 if (failed(actionMoveOpsRes))
614 return failure();
615
616 // Gather variable updates during the action.
617 DenseMap<fsm::VariableOp, Value> variableUpdates;
618 for (auto updateOp : transition.getAction().getOps<fsm::UpdateOp>()) {
619 VariableOp variableOp = updateOp.getVariableOp();
620 variableUpdates[variableOp] = updateOp.getValue();
621 }
622
623 stateToVariableUpdates[currentState][transition.getNextStateOp()] =
624 variableUpdates;
625 }
626
627 // Guard conversion
628 if (transition.hasGuard()) {
629 // Not always taken; recurse and mux between the targeted next state and
630 // the recursion result, selecting based on the provided guard.
631 auto guardOpRes = moveOps(&transition.getGuard().front());
632 if (failed(guardOpRes))
633 return failure();
634
635 auto guardOp = cast<ReturnOp>(*guardOpRes);
636 assert(guardOp && "guard should be defined");
637 auto guard = guardOp.getOperand();
638 auto otherNextState =
639 convertTransitions(currentState, transitions.drop_front());
640 if (failed(otherNextState))
641 return failure();
642 comb::MuxOp nextStateMux = comb::MuxOp::create(
643 b, transition.getLoc(), guard, nextState, *otherNextState, false);
644 nextState = nextStateMux;
645 }
646 }
647
648 assert(nextState && "next state should be defined");
649 return nextState;
650}

652FailureOr<MachineOpConverter::StateConversionResult>
653MachineOpConverter::convertState(StateOp state) {
654 MachineOpConverter::StateConversionResult res;
655
656 // 3.1) Convert the output region by moving the operations into the module
657 // scope and gathering the operands of the output op.
658 if (!state.getOutput().empty()) {
659 auto outputOpRes = moveOps(&state.getOutput().front());
660 if (failed(outputOpRes))
661 return failure();
662
663 OutputOp outputOp = cast<fsm::OutputOp>(*outputOpRes);
664 res.outputs = outputOp.getOperands(); // 3.2
665 }
666
667 SmallVector<TransitionOp> transitions;
668 for (auto &op : state.getTransitions().getOps()) {
669 if (auto transOp = dyn_cast<TransitionOp>(op)) {
670 transitions.push_back(transOp);
671 } else {
672 // Clone operations which are inside `transitions` region but outside
673 // `guard` region.
674 auto opClone = b.clone(op);
675 for (auto [i, res] : llvm::enumerate(op.getResults()))
676 res.replaceAllUsesWith(opClone->getResult(i));
677 }
678 }
679 // 3.3, 3.4) Convert the transitions and record the next-state value
680 // derived from the transitions being selected in a priority-encoded manner.
681 auto nextStateRes = convertTransitions(state, transitions);
682 if (failed(nextStateRes))
683 return failure();
684 res.nextState = *nextStateRes;
685 return res;
686}

692void FSMToSVPass::runOnOperation() {
693 auto module = getOperation();
694 auto loc = module.getLoc();
695 auto b = OpBuilder(module);
696
697 // Identify the machines to lower, bail out if none exist.
698 auto machineOps = llvm::to_vector(module.getOps<MachineOp>());
699 if (machineOps.empty()) {
700 markAllAnalysesPreserved();
701 return;
702 }
703
704 // Create a typescope shared by all of the FSMs. This typescope will be
705 // emitted in a single separate file to avoid polluting each output file with
706 // typedefs.
707 b.setInsertionPointToStart(module.getBody());
708 hw::TypeScopeOp typeScope =
709 hw::TypeScopeOp::create(b, loc, b.getStringAttr("fsm_enum_typedecls"));
710 typeScope.getBodyRegion().push_back(new Block());
711
712 auto file = emit::FileOp::create(b, loc, "fsm_enum_typedefs.sv", [&] {
713 emit::RefOp::create(b, loc,
714 FlatSymbolRefAttr::get(typeScope.getSymNameAttr()));
715 });
716 auto fragment = emit::FragmentOp::create(b, loc, "FSM_ENUM_TYPEDEFS", [&] {
717 sv::VerbatimOp::create(b, loc, "`include \"" + file.getFileName() + "\"");
718 });
719
720 auto headerName = FlatSymbolRefAttr::get(fragment.getSymNameAttr());
721
722 // Traverse all machines and convert.
723 for (auto machineOp : machineOps) {
724 MachineOpConverter converter(b, typeScope, machineOp, headerName);
725
726 if (failed(converter.dispatch())) {
727 signalPassFailure();
728 return;
729 }
730 }
731
732 // Traverse all machine instances and convert to hw instances.
733 llvm::SmallVector<HWInstanceOp> instances;
734 module.walk([&](HWInstanceOp instance) { instances.push_back(instance); });
735 for (auto instance : instances) {
736 auto fsmHWModule =
737 module.lookupSymbol<hw::HWModuleOp>(instance.getMachine());
738 assert(fsmHWModule &&
739 "FSM machine should have been converted to a hw.module");
740
741 b.setInsertionPoint(instance);
742 llvm::SmallVector<Value, 4> operands;
743 llvm::transform(instance.getOperands(), std::back_inserter(operands),
744 [&](auto operand) { return operand; });
745 auto hwInstance = hw::InstanceOp::create(
746 b, instance.getLoc(), fsmHWModule, b.getStringAttr(instance.getName()),
747 operands, nullptr);
748 instance.replaceAllUsesWith(hwInstance);
749 instance.erase();
750 }
751
752 assert(!typeScope.getBodyBlock()->empty() && "missing type decls");
753}

755} // end anonymous namespace
