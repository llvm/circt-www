26/// A named and typed field of an aggregate type.
27struct FieldInfo {
28 mlir::StringAttr name;
29 mlir::Type type;
30};
31
32inline bool operator==(const FieldInfo &a, const FieldInfo &b) {
33 return a.name == b.name && a.type == b.type;
34}
35inline llvm::hash_code hash_value(const FieldInfo &fi) {
36 return llvm::hash_combine(fi.name, fi.type);
37}

39/// Parse a list of named fields within `<>`, e.g. `<foo: i7, bar: i8>`. Field
40/// names may be bare keywords or quoted strings. The location and name of every

42/// `duplicates`, leaving it to the caller to report them.

53/// Expose the names and types of fields to the generic attribute and type
54/// walking and replacement infrastructure. This allows walkers to recurse into
55/// the fields of aggregate types, and replacers such as
56/// `mlir::AttrTypeReplacer` to replace types nested within the fields.
57template <>
58struct AttrTypeSubElementHandler<circt::FieldInfo> {
59 static void walk(const circt::FieldInfo &param,
60 AttrTypeImmediateSubElementWalker &walker) {
61 walker.walk(param.name);
62 walker.walk(param.type);
63 }
64 static circt::FieldInfo replace(const circt::FieldInfo &param,
65 AttrSubElementReplacements &attrRepls,
66 TypeSubElementReplacements &typeRepls) {
67 return {cast<StringAttr>(attrRepls.take_front(1)[0]),
68 typeRepls.take_front(1)[0]};
69 }
70};
71} // namespace mlir
