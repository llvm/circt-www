32/// A format descriptor, to be given to arcRuntimeFormat.
33///
34/// arcRuntimeFormat takes an array of FmtDescriptor and a variadic argument
35/// list. Each FmtDescriptor describes how to format the corresponding
36/// argument. The array is terminated by a FmtDescriptor with action Action_End.
37struct FmtDescriptor {
38 /// Default construction creates an end of string descriptor.
39 FmtDescriptor() {
40 std::memset(this, 0, sizeof(*this));
41 action = Action_End;
42 }
43
44 /// Creates a literal string descriptor.
45 ///
46 /// width: The width of the literal string in characters.
47 ///
48 /// The string itself will be passed as a variadic argument (const char*).
49 static FmtDescriptor createLiteral(int64_t width) {
50 FmtDescriptor d;
51 d.action = Action_Literal;
52 d.literal.width = width;
53 return d;
54 }
55
56 /// Creates a small literal string descriptor.
57 static FmtDescriptor createSmallLiteral(std::string_view str) {
58 assert(str.size() < sizeof(FmtDescriptor::smallLiteral.data));
59 FmtDescriptor d;
60 d.action = Action_LiteralSmall;
61 std::strncpy(d.smallLiteral.data, str.data(),
62 sizeof(d.smallLiteral.data) - 1);
63 d.smallLiteral.data[sizeof(d.smallLiteral.data) - 1] = '\0';
64 return d;
65 }
66
67 /// Creates an integer descriptor.
68 ///
69 /// bitwidth: The bitwidth of the integer value.
70 /// radix: The radix to use for formatting. Must be one of {2, 8, 10, 16}.
71 /// isLeftAligned: Whether the value is left aligned.
72 /// specifierWidth: The minumum width of the output in characters.
73 /// isUpperCase: Whether to use uppercase hex letters.
74 /// isSigned: Whether to treat the value as signed.
75 ///
76 /// The integer value will be passed as a variadic argument by *pointer*.
77 static FmtDescriptor createInt(int32_t bitwidth, int8_t radix,
78 bool isLeftAligned, int32_t specifierWidth,
79 char paddingChar, bool isUpperCase,
80 bool isSigned) {
81 FmtDescriptor d;
82 d.action = Action_Int;
83 d.intFmt.bitwidth = bitwidth;
84 d.intFmt.radix = radix;
85 d.intFmt.isLeftAligned = isLeftAligned;
86 d.intFmt.specifierWidth = specifierWidth;
87 d.intFmt.paddingChar = paddingChar;
88 d.intFmt.isUpperCase = isUpperCase;
89 d.intFmt.isSigned = isSigned;
90 return d;
91 }
92
93 /// Creates a char descriptor.
94 ///
95 /// The character value will be passed as a variadic argument by value.
96 static FmtDescriptor createChar() {
97 FmtDescriptor d;
98 d.action = Action_Char;
99 return d;
100 }
101
102 /// The action to take for this descriptor.
103 ///
104 /// We use uint64_t to ensure that the descriptor is always 16 bytes in size
105 /// with zero padding.
106 enum Action : uint64_t {
107 /// End of the format string, no action to take.
108 Action_End = 0,
109 /// Prints a literal string.
110 Action_Literal,
111 /// Prints a literal string (small string optimization).
112 Action_LiteralSmall,
113 /// Prints an integer.
114 Action_Int,
115 /// Prints a character (%c).
116 Action_Char,
117 };
118 Action action;

120 /// Integer formatting options.
121 struct IntFmt {
122 /// The bitwidth of the integer value.
123 int16_t bitwidth;
124 /// The minumum width of the output in characters.
125 int16_t specifierWidth;
126 /// The radix to use for formatting. Must be one of {2, 8, 10, 16}.
127 int8_t radix;
128 /// Padding character (NUL if no padding is desired).
129 char paddingChar;
130 /// Whether the value is left aligned.
131 bool isLeftAligned;
132 /// Whether to use uppercase hex letters.
133 bool isUpperCase;
134 /// Whether to treat the value as signed.
135 bool isSigned;
136 };
137
138 /// Literal string formatting options.
139 struct LiteralFmt {
140 /// The width of the literal string in characters. Note that the string
141 /// itself is passed as a variadic argument, and may contain NUL characters.
142 int64_t width;
143 };
144
145 /// Literal string (small string optimization).
146 struct SmallLiteral {
147 /// NUL-terminated string.
148 char data[8];
149 };
150
151 union {
152 LiteralFmt literal;
153 IntFmt intFmt;
154 SmallLiteral smallLiteral;
155 };
156};
