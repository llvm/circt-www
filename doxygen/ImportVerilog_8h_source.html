36/// Options that control how Verilog input files are parsed and processed.
37///
38/// See `slang::driver::Driver::Options` for inspiration. Also check out
39/// `Driver::addStandardArgs()` for some inspiration on how to expose these on
40/// the command line.
41struct ImportVerilogOptions {
42 /// Limit importing to linting, parsing, or elaboration only.
43 enum class Mode {
44 /// Only lint the input.
45 OnlyLint,
46 /// Only parse the input syntax and report parse diagnostics.
47 OnlyParse,
48 /// Parse and elaborate the input and convert it to Moore dialect ops,
49 /// without running any transformation passes.
50 OnlyImport,
51 /// Perform a full import and mapping to the core dialects.
52 Full
53 };
54 Mode mode = Mode::Full;
55
56 /// Generate debug information in the form of debug dialect ops in the IR.
57 bool debugInfo = false;
58
59 /// Interpret `always @(*)` as `always_comb`.
60 bool lowerAlwaysAtStarAsComb = true;
61
62 //===--------------------------------------------------------------------===//
63 // Include paths
64 //===--------------------------------------------------------------------===//
65
66 /// A list of include directories in which to search for files.
67 std::vector<std::string> includeDirs;
68
69 /// A list of system include directories in which to search for files.
70 std::vector<std::string> includeSystemDirs;
71
72 /// A list of library directories in which to search for missing modules.
73 std::vector<std::string> libDirs;
74
75 /// A list of extensions that will be used to search for library files.
76 std::vector<std::string> libExts;
77
78 /// A list of extensions that will be used to exclude files.
79 std::vector<std::string> excludeExts;
80
81 /// A list of preprocessor directives to be ignored.
82 std::vector<std::string> ignoreDirectives;
83
84 //===--------------------------------------------------------------------===//
85 // Preprocessing
86 //===--------------------------------------------------------------------===//
87
88 /// The maximum depth of included files before an error is issued.
89 std::optional<uint32_t> maxIncludeDepth;
90
91 /// A list of macros that should be defined in each compilation unit.
92 std::vector<std::string> defines;
93
94 /// A list of macros that should be undefined in each compilation unit.
95 std::vector<std::string> undefines;
96
97 /// If true, library files will inherit macro definitions from primary source
98 /// files.
99 std::optional<bool> librariesInheritMacros;
100
101 //===--------------------------------------------------------------------===//
102 // Compilation
103 //===--------------------------------------------------------------------===//
104
105 /// A string that indicates the default time scale to use for any design
106 /// elements that don't specify one explicitly.
107 std::optional<std::string> timeScale;
108
109 /// If true, allow various to be referenced before they are declared.
110 std::optional<bool> allowUseBeforeDeclare;
111
112 /// If true, ignore errors about unknown modules.
113 std::optional<bool> ignoreUnknownModules;
114
115 /// If non-empty, specifies the list of modules that should serve as the top
116 /// modules in the design. If empty, this will be automatically determined
117 /// based on which modules are unreferenced elsewhere.
118 std::vector<std::string> topModules;
119
120 /// A list of top-level module parameters to override, of the form
121 /// `<name>=<value>`.
122 std::vector<std::string> paramOverrides;
123
124 //===--------------------------------------------------------------------===//
125 // Diagnostics Control
126 //===--------------------------------------------------------------------===//
127
128 /// The maximum number of errors to print before giving up.
129 std::optional<uint32_t> errorLimit;
130
131 /// A list of warning options that will be passed to the DiagnosticEngine.
132 std::vector<std::string> warningOptions;
133
134 /// A list of paths in which to suppress warnings.
135 std::vector<std::string> suppressWarningsPaths;
136
137 /// If true, make paths for source file debug locations relative to the
138 /// current working directory. Otherwise, leave them unchanged.
139 bool makeLocationPathsProximate = true;
140
141 //===--------------------------------------------------------------------===//
142 // File lists
143 //===--------------------------------------------------------------------===//
144
145 /// If set to true, all source files will be treated as part of a single
146 /// compilation unit, meaning all of their text will be merged together.
147 std::optional<bool> singleUnit;
148
149 /// A list of library files to include in the compilation.
150 std::vector<std::string> libraryFiles;
151
152 /// A list of command files to process for compilation.
153 std::vector<std::string> commandFiles;
154
155 //===--------------------------------------------------------------------===//
156 // Misc slang options
157 //===--------------------------------------------------------------------===//
158 /// A list of arbitrary arguments to pass to the Slang CLI.
159 std::vector<std::string> slangArgs;
160};
