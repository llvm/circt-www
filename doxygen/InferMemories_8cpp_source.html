33struct InferMemoriesPass
34 : public arc::impl::InferMemoriesBase<InferMemoriesPass> {
35 using InferMemoriesBase::InferMemoriesBase;
36
37 void runOnOperation() override;
38
39 SmallVector<Operation *> opsToDelete;
40 SmallPtrSet<StringAttr, 2> schemaNames;
41 DenseMap<StringAttr, DictionaryAttr> memoryParams;
42};
43} // namespace
44
45void InferMemoriesPass::runOnOperation() {
46 auto module = getOperation();
47 opsToDelete.clear();
48 schemaNames.clear();
49 memoryParams.clear();
50
51 SymbolCache cache;
52 cache.addDefinitions(module);
53 Namespace names;
54 names.add(cache);
55
56 // Find the matching generator schemas.
57 for (auto schemaOp : module.getOps<hw::HWGeneratorSchemaOp>()) {
58 if (schemaOp.getDescriptor() == "FIRRTL_Memory") {
59 schemaNames.insert(schemaOp.getSymNameAttr());
60 opsToDelete.push_back(schemaOp);
61 }
62 }
63 LLVM_DEBUG(llvm::dbgs() << "Found " << schemaNames.size() << " schemas\n");
64
65 // Find generated ops using these schemas.
66 for (auto genOp : module.getOps<hw::HWModuleGeneratedOp>()) {
67 if (!schemaNames.contains(genOp.getGeneratorKindAttr().getAttr()))
68 continue;
69 memoryParams[genOp.getModuleNameAttr()] = genOp->getAttrDictionary();
70 opsToDelete.push_back(genOp);
71 }
72 LLVM_DEBUG(llvm::dbgs() << "Found " << memoryParams.size()
73 << " memory modules\n");
74
75 // Convert instances of the generated ops into dedicated memories.
76 unsigned numReplaced = 0;
77 module.walk([&](hw::InstanceOp instOp) {
78 auto it = memoryParams.find(instOp.getModuleNameAttr().getAttr());
79 if (it == memoryParams.end())
80 return;
81 ++numReplaced;
82 DictionaryAttr params = it->second;
83 auto width = params.getAs<IntegerAttr>("width").getValue().getZExtValue();
84 auto depth = params.getAs<IntegerAttr>("depth").getValue().getZExtValue();
85 auto maskGranAttr = params.getAs<IntegerAttr>("maskGran");
86 auto maskGran =
87 maskGranAttr ? maskGranAttr.getValue().getZExtValue() : width;
88 auto maskBits = width / maskGran;
89
90 auto writeLatency =
91 params.getAs<IntegerAttr>("writeLatency").getValue().getZExtValue();
92 auto readLatency =
93 params.getAs<IntegerAttr>("readLatency").getValue().getZExtValue();
94 if (writeLatency != 1) {
95 instOp.emitError("unsupported memory write latency ") << writeLatency;
96 return signalPassFailure();
97 }
98
99 // FIRRTL memories are currently underspecified. They have a single read
100 // latency, but it is unclear where within this latency the read of the
101 // underlying storage happens. The `HWMemSimImpl` pass implements memories
102 // such that the storage is probed at the very end of the latency, and that
103 // the probed value becomes available immediately. We keep the latencies
104 // configurable here, in the hopes that we'll improve our memory
105 // abstractions at some point.
106 unsigned readPreLatency = readLatency; // cycles before storage is read
107 unsigned readPostLatency = 0; // cycles to move read value to output
108
109 ImplicitLocOpBuilder builder(instOp.getLoc(), instOp);
110 auto wordType = builder.getIntegerType(width);
111 auto addressTy = dyn_cast<IntegerType>(instOp.getOperand(0).getType());
112 if (!addressTy) {
113 instOp.emitError("expected integer type for memory addressing, got ")
114 << addressTy;
115 return signalPassFailure();
116 }
117 auto memType = MemoryType::get(&getContext(), depth, wordType, addressTy);
118 auto memOp = MemoryOp::create(builder, memType);
119 if (tapMemories && !instOp.getInstanceName().empty())
120 memOp->setAttr("name", instOp.getInstanceNameAttr());
121
122 unsigned argIdx = 0;
123 unsigned resultIdx = 0;
124
125 auto applyLatency = [&](Value clock, Value data, unsigned latency) {
126 for (unsigned i = 0; i < latency; ++i)
127 data = seq::CompRegOp::create(builder, data, clock,
128 builder.getStringAttr(""), Value{},
129 Value{}, Value{}, hw::InnerSymAttr{});
130 return data;
131 };
132
133 SmallVector<std::tuple<Value, Value, SmallVector<Value>, bool, bool>>
134 writePorts;
135
136 // Use `<inst-name>/` as the prefix for all port taps.
137 SmallString<64> tapPrefix(instOp.getInstanceName());
138 if (!tapPrefix.empty())
139 tapPrefix.push_back('/');
140 auto tapPrefixBaseLen = tapPrefix.size();
141
142 auto tap = [&](Value value, const Twine &name) {
143 auto prefixedName = builder.getStringAttr(tapPrefix + "_" + name);
144 arc::TapOp::create(builder, value, prefixedName);
145 };
146
147 // Handle read ports.
148 auto numReadPorts =
149 params.getAs<IntegerAttr>("numReadPorts").getValue().getZExtValue();
150 for (unsigned portIdx = 0; portIdx != numReadPorts; ++portIdx) {
151 auto address = instOp.getOperand(argIdx++);
152 auto enable = instOp.getOperand(argIdx++);
153 auto clock = instOp.getOperand(argIdx++);
154 auto data = instOp.getResult(resultIdx++);
155
156 if (address.getType() != addressTy) {
157 instOp.emitOpError("expected ")
158 << addressTy << ", but got " << address.getType();
159 return signalPassFailure();
160 }
161
162 // Add port taps.
163 if (tapPorts) {
164 tapPrefix.resize(tapPrefixBaseLen);
165 (Twine("R") + Twine(portIdx)).toVector(tapPrefix);
166 tap(address, "addr");
167 tap(enable, "en");
168 tap(data, "data");
169 }
170
171 // Apply the latency before the underlying storage is accessed.
172 address = applyLatency(clock, address, readPreLatency);
173 enable = applyLatency(clock, enable, readPreLatency);
174
175 // Read the underlying storage. (The result of a disabled read port is
176 // undefined, currently we define it to be zero.)
177 Value readOp =
178 MemoryReadPortOp::create(builder, wordType, memOp, address);
179 Value zero = hw::ConstantOp::create(builder, wordType, 0);
180 readOp = comb::MuxOp::create(builder, enable, readOp, zero);
181
182 // Apply the latency after the underlying storage was accessed. (If the
183 // latency is 0, the memory read is combinatorial without any buffer.)
184 readOp = applyLatency(clock, readOp, readPostLatency);
185 data.replaceAllUsesWith(readOp);
186 }
187
188 // Handle read-write ports.
189 auto numReadWritePorts = params.getAs<IntegerAttr>("numReadWritePorts")
190 .getValue()
191 .getZExtValue();
192 for (unsigned portIdx = 0; portIdx != numReadWritePorts; ++portIdx) {
193 auto address = instOp.getOperand(argIdx++);
194 auto enable = instOp.getOperand(argIdx++);
195 auto clock = instOp.getOperand(argIdx++);
196 auto writeMode = instOp.getOperand(argIdx++);
197 auto writeData = instOp.getOperand(argIdx++);
198 auto writeMask = maskBits > 1 ? instOp.getOperand(argIdx++) : Value{};
199 auto readData = instOp.getResult(resultIdx++);
200
201 if (address.getType() != addressTy) {
202 instOp.emitOpError("expected ")
203 << addressTy << ", but got " << address.getType();
204 return signalPassFailure();
205 }
206
207 // Add port taps.
208 if (tapPorts) {
209 tapPrefix.resize(tapPrefixBaseLen);
210 (Twine("RW") + Twine(portIdx)).toVector(tapPrefix);
211 tap(address, "addr");
212 tap(enable, "en");
213 tap(writeMode, "wmode");
214 tap(writeData, "wdata");
215 if (writeMask)
216 tap(writeMask, "wmask");
217 tap(readData, "rdata");
218 }
219
220 auto c1_i1 = hw::ConstantOp::create(builder, builder.getI1Type(), 1);
221 auto notWriteMode = comb::XorOp::create(builder, writeMode, c1_i1);
222 Value readEnable = comb::AndOp::create(builder, enable, notWriteMode);
223
224 // Apply the latency before the underlying storage is accessed.
225 Value readAddress = applyLatency(clock, address, readPreLatency);
226 readEnable = applyLatency(clock, readEnable, readPreLatency);
227
228 // Read the underlying storage. (The result of a disabled read port is
229 // undefined, currently we define it to be zero.)
230 Value readOp =
231 MemoryReadPortOp::create(builder, wordType, memOp, readAddress);
232 Value zero = hw::ConstantOp::create(builder, wordType, 0);
233 readOp = comb::MuxOp::create(builder, readEnable, readOp, zero);
234
235 if (writeMask) {
236 unsigned maskWidth = cast<IntegerType>(writeMask.getType()).getWidth();
237 SmallVector<Value> toConcat;
238 for (unsigned i = 0; i < maskWidth; ++i) {
239 Value bit = comb::ExtractOp::create(builder, writeMask, i, 1);
240 Value replicated = comb::ReplicateOp::create(builder, bit, maskGran);
241 toConcat.push_back(replicated);
242 }
243 std::reverse(toConcat.begin(), toConcat.end()); // I hate concat
244 writeMask =
245 comb::ConcatOp::create(builder, writeData.getType(), toConcat);
246 }
247
248 // Apply the latency after the underlying storage was accessed. (If the
249 // latency is 0, the memory read is combinatorial without any buffer.)
250 readOp = applyLatency(clock, readOp, readPostLatency);
251 readData.replaceAllUsesWith(readOp);
252
253 auto writeEnable = comb::AndOp::create(builder, enable, writeMode);
254 SmallVector<Value> inputs({address, writeData, writeEnable});
255 if (writeMask)
256 inputs.push_back(writeMask);
257 writePorts.push_back({memOp, clock, inputs, true, !!writeMask});
258 }
259
260 // Handle write ports.
261 auto numWritePorts =
262 params.getAs<IntegerAttr>("numWritePorts").getValue().getZExtValue();
263 for (unsigned portIdx = 0; portIdx != numWritePorts; ++portIdx) {
264 auto address = instOp.getOperand(argIdx++);
265 auto enable = instOp.getOperand(argIdx++);
266 auto clock = instOp.getOperand(argIdx++);
267 auto data = instOp.getOperand(argIdx++);
268 auto mask = maskBits > 1 ? instOp.getOperand(argIdx++) : Value{};
269
270 if (address.getType() != addressTy) {
271 instOp.emitOpError("expected ")
272 << addressTy << ", but got " << address.getType();
273 return signalPassFailure();
274 }
275
276 // Add port taps.
277 if (tapPorts) {
278 tapPrefix.resize(tapPrefixBaseLen);
279 (Twine("W") + Twine(portIdx)).toVector(tapPrefix);
280 tap(address, "addr");
281 tap(enable, "en");
282 tap(data, "data");
283 if (mask)
284 tap(mask, "mask");
285 }
286
287 if (mask) {
288 unsigned maskWidth = cast<IntegerType>(mask.getType()).getWidth();
289 SmallVector<Value> toConcat;
290 for (unsigned i = 0; i < maskWidth; ++i) {
291 Value bit = comb::ExtractOp::create(builder, mask, i, 1);
292 Value replicated = comb::ReplicateOp::create(builder, bit, maskGran);
293 toConcat.push_back(replicated);
294 }
295 std::reverse(toConcat.begin(), toConcat.end()); // I hate concat
296 mask = comb::ConcatOp::create(builder, data.getType(), toConcat);
297 }
298 SmallVector<Value> inputs({address, data});
299 if (enable)
300 inputs.push_back(enable);
301 if (mask)
302 inputs.push_back(mask);
303 writePorts.push_back({memOp, clock, inputs, !!enable, !!mask});
304 }
305
306 // Create the actual write ports with a dependency arc to all read
307 // ports.
308 for (auto [memOp, clock, inputs, hasEnable, hasMask] : writePorts) {
309 auto ipSave = builder.saveInsertionPoint();
310 TypeRange types = ValueRange(inputs).getTypes();
311 builder.setInsertionPointToStart(module.getBody());
312 auto defOp = DefineOp::create(builder, names.newName("mem_write"),
313 builder.getFunctionType(types, types));
314 auto &block = defOp.getBody().emplaceBlock();
315 auto args = block.addArguments(
316 types, SmallVector<Location>(types.size(), builder.getLoc()));
317 builder.setInsertionPointToEnd(&block);
318 arc::OutputOp::create(builder, SmallVector<Value>(args));
319 builder.restoreInsertionPoint(ipSave);
320 MemoryWritePortOp::create(builder, memOp, defOp.getName(), inputs, clock,
321 hasEnable, hasMask);
322 }
323
324 opsToDelete.push_back(instOp);
325 });
326 LLVM_DEBUG(llvm::dbgs() << "Inferred " << numReplaced << " memories\n");
327
328 for (auto *op : opsToDelete)
329 op->erase();
330}
