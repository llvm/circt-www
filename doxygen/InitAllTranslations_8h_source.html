32// This function should be called before creating any MLIRContext if one
33// expects all the possible translations to be made available to the context
34// automatically.
35inline void registerAllTranslations() {
36 static bool initOnce = []() {
37 arc::registerArcModelInfoTranslation();
38 calyx::registerToCalyxTranslation();
39 firrtl::registerFromFIRFileTranslation();
40 firrtl::registerToFIRFileTranslation();
41 mlir::smt::registerExportSMTLIBTranslation();
42 ExportSystemC::registerExportSystemCTranslation();
43 debug::registerTranslations();
44 aiger::registerImportAIGERTranslation();
45 aiger::registerExportAIGERTranslation();
46 liberty::registerImportLibertyTranslation();
47 return true;
48 }();
49 (void)initOnce;
50}
51} // namespace circt
