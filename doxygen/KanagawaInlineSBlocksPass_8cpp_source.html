45class InlineSBlocksPattern
46 : public OpConversionPattern<kanagawa::StaticBlockOp> {
47public:
48 using OpConversionPattern<kanagawa::StaticBlockOp>::OpConversionPattern;
49 using OpAdaptor =
50 typename OpConversionPattern<kanagawa::StaticBlockOp>::OpAdaptor;
51
52 LogicalResult
53 matchAndRewrite(kanagawa::StaticBlockOp op, OpAdaptor adaptor,
54 ConversionPatternRewriter &rewriter) const override {
55 Location loc = op.getLoc();
56
57 // We only create block markers in case the sblock has attributes.
58 bool hasAttributes = !op->getAttrs().empty();
59
60 if (hasAttributes) {
61 // Start the inline block...
62 auto inlineStart =
63 kanagawa::InlineStaticBlockBeginOp::create(rewriter, loc);
64 inlineStart->setAttrs(op->getAttrs());
65 }
66
67 // Inline the sblock.
68 Block *sblockBody = op.getBodyBlock();
69 BlockReturnOp ret = cast<BlockReturnOp>(sblockBody->getTerminator());
70 rewriter.inlineBlockBefore(sblockBody, op->getNextNode());
71
72 // Replace the kanagawa.sblock return values with the values that were
73 // defined (returned from) within the sblock body, and erase the return op.
74 rewriter.replaceOp(op, ret.getRetValues());
75
76 if (hasAttributes) {
77 // Close the inline block
78 rewriter.setInsertionPoint(ret);
79 kanagawa::InlineStaticBlockEndOp::create(rewriter, loc);
80 }
81
82 rewriter.eraseOp(ret);
83 return success();
84 }
85};
86
87} // anonymous namespace

89void InlineSBlocksPass::runOnOperation() {
90 MethodOp parent = getOperation();
91 ConversionTarget target(getContext());
92 target.addIllegalOp<kanagawa::StaticBlockOp>();
93 // Mark everything but `kanagawa.sblock` as legal - we need a legalization
94 // pattern for any possible op that gets inlined from a block.
95 target.markUnknownOpDynamicallyLegal([](Operation *op) { return true; });
96 RewritePatternSet patterns(&getContext());
97 patterns.add<InlineSBlocksPattern>(&getContext());
98
99 if (failed(applyPartialConversion(parent, target, std::move(patterns))))
100 return signalPassFailure();
101}
