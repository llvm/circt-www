26/// Options which control the emission from CIRCT to Verilog.
27struct LoweringOptions {
28 /// Error callback type used to indicate errors parsing the options string.
29 using ErrorHandlerT = llvm::function_ref<void(llvm::Twine)>;
30
31 /// Create a LoweringOptions with the default values.
32 LoweringOptions() = default;
33
34 /// Create a LoweringOptions and read in options from a string,
35 /// overriding only the set options in the string.
36 LoweringOptions(llvm::StringRef options, ErrorHandlerT errorHandler);
37
38 /// Create a LoweringOptions with values loaded from an MLIR ModuleOp. This
39 /// loads a string attribute with the key `circt.loweringOptions`. If there is
40 /// an error parsing the attribute this will print an error using the
41 /// ModuleOp.
42 LoweringOptions(mlir::ModuleOp module);
43
44 /// Return the value of the `circt.loweringOptions` in the specified module
45 /// if present, or a null attribute if not.
46 static mlir::StringAttr getAttributeFrom(mlir::ModuleOp module);
47
48 /// Read in options from a string, overriding only the set options in the
49 /// string.
50 void parse(llvm::StringRef options, ErrorHandlerT callback);
51
52 /// Returns a string representation of the options.
53 std::string toString() const;
54
55 /// Write the verilog emitter options to a module's attributes.
56 void setAsAttribute(mlir::ModuleOp module);
57
58 /// Load any emitter options from the module. If there is an error validating
59 /// the attribute, this will print an error using the ModuleOp.
60 void parseFromAttribute(mlir::ModuleOp module);
61
62 /// If true, emits `sv.alwayscomb` as Verilog `always @(*)` statements.
63 /// Otherwise, print them as `always_comb`.
64 bool noAlwaysComb = false;
65
66 /// If true, expressions are allowed in the sensitivity list of `always`
67 /// statements, otherwise they are forced to be simple wires. Some EDA
68 /// tools rely on these being simple wires.
69 bool allowExprInEventControl = false;
70
71 /// If true, eliminate packed arrays for tools that don't support them (e.g.
72 /// Yosys).
73 bool disallowPackedArrays = false;
74
75 /// If true, eliminate packed struct assignments in favor of a wire +
76 /// assignments to the individual fields.
77 bool disallowPackedStructAssignments = false;
78
79 /// If true, do not emit SystemVerilog locally scoped "automatic" or logic
80 /// declarations - emit top level wire and reg's instead.
81 bool disallowLocalVariables = false;
82
83 /// If true, verification statements like `assert`, `assume`, and `cover` will
84 /// always be emitted with a label. If the statement has no label in the IR, a
85 /// generic one will be created. Some EDA tools require verification
86 /// statements to be labeled.
87 bool enforceVerifLabels = false;
88
89 /// This is the maximum number of terms in an expression before that
90 /// expression spills a wire.
91 enum { DEFAULT_TERM_LIMIT = 256 };
92 unsigned maximumNumberOfTermsPerExpression = DEFAULT_TERM_LIMIT;
93
94 /// This is the target width of lines in an emitted Verilog source file in
95 /// columns. Zero indicates no wrapping. Must be < 2^14.
96 enum { DEFAULT_LINE_LENGTH = 90, MAX_LINE_LENGTH = (1 << 14) - 1 };
97 unsigned emittedLineLength = DEFAULT_LINE_LENGTH;
98
99 /// Return std::nullopt if not wrapping, else the target emission line length.
100 std::optional<unsigned> getEmittedLineLength() const {
101 if (emittedLineLength == 0)
102 return std::nullopt;
103 // This should be rejected in parsing, but in case set directly, saturate.
104 return std::min<unsigned>(MAX_LINE_LENGTH, emittedLineLength);
105 }
106
107 /// Add an explicit bitcast for avoiding bitwidth mismatch LINT errors.
108 bool explicitBitcast = false;
109
110 /// If true, replicated ops are emitted to a header file.
111 bool emitReplicatedOpsToHeader = false;
112
113 /// This option controls emitted location information style.
114 enum LocationInfoStyle {
115 Plain, // Default.
116 WrapInAtSquareBracket, // Wrap location info in @[..].
117 None, // No location info comment.
118 } locationInfoStyle = Plain;
119
120 /// If true, every port is declared separately
121 /// (each includes direction and type (e.g., `input [3:0]`)).
122 /// When false (default), ports share declarations when possible.
123 bool disallowPortDeclSharing = false;
124
125 /// Print debug info.
126 bool printDebugInfo = false;
127
128 /// If true, every mux expression is spilled to a wire.
129 bool disallowMuxInlining = false;
130
131 /// This controls extra wire spilling performed in PrepareForEmission to
132 /// improve readablitiy and debuggability.
133 enum WireSpillingHeuristic : unsigned {
134 SpillLargeTermsWithNamehints = 1, // Spill wires for expressions with
135 // namehints if the term size is greater
136 // than `wireSpillingNamehintTermLimit`.
137 };
138
139 unsigned wireSpillingHeuristicSet = 0;
140
141 bool isWireSpillingHeuristicEnabled(WireSpillingHeuristic heurisic) const {
142 return static_cast<bool>(wireSpillingHeuristicSet & heurisic);
143 }
144
145 enum { DEFAULT_NAMEHINT_TERM_LIMIT = 3 };
146 unsigned wireSpillingNamehintTermLimit = DEFAULT_NAMEHINT_TERM_LIMIT;
147
148 /// If true, every expression passed to an instance port is driven by a wire.
149 /// Some lint tools dislike expressions being inlined into input ports so this
150 /// option avoids such warnings.
151 bool disallowExpressionInliningInPorts = false;
152
153 /// If true, every expression used as an array index is driven by a wire, and
154 /// the wire is marked as `(* keep = "true" *)`. Certain versions of Vivado
155 /// produce incorrect synthesis results for certain arithmetic ops inlined
156 /// into the array index.
157 bool mitigateVivadoArrayIndexConstPropBug = false;
158
159 /// If true, emit `wire` in port lists rather than nothing. Used in cases
160 /// where `default_nettype is not set to wire.
161 bool emitWireInPorts = false;
162
163 /// If true, emit a comment wherever an instance wasn't printed, because
164 /// it's emitted elsewhere as a bind.
165 bool emitBindComments = false;
166
167 /// If true, do not emit a version comment at the top of each verilog file.
168 bool omitVersionComment = false;
169
170 /// If true, then unique names that collide with keywords case insensitively.
171 /// This is used to avoid stricter lint warnings which, e.g., treat "REG" as a
172 /// Verilog keyword.
173 bool caseInsensitiveKeywords = false;
174
175 /// If true, then update the the mlir to include output verilog locations.
176 bool emitVerilogLocations = false;
177
178 /// If true, add a dummy wire to empty modules to prevent tools from regarding
179 /// the module as blackbox.
180 bool fixUpEmptyModules = false;
181
182 // If true, wrap concurrent assertions with explicit clocking events into
183 // procedural form, by hoisting the clock event into an enclosing always
184 // block. The same transformation is applied to `assume property` and `cover
185 // property`.
186 bool disallowClockedAssertions = false;
187
188 // If true, always emit assignments (both continuous and blocking) separately
189 // rather than inlining them in net and variable declarations
190 bool disallowDeclAssignments = false;
191
192 /// If true, always emit begin/end blocks around if/else/always bodies, even
193 /// when there is only a single statement.
194 bool alwaysEmitBeginEnd = false;
195};
196} // namespace circt

198#endif // CIRCT_SUPPORT_LOWERINGOPTIONS_H
