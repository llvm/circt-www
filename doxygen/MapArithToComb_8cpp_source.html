31// A type converter which legalizes integer types, thus ensuring that vector
32// types are illegal.
33class MapArithTypeConverter : public mlir::TypeConverter {
34public:
35 MapArithTypeConverter() {
36 addConversion([](Type type) {
37 if (hw::isHWValueType(type))
38 return type;
39
40 return Type();
41 });
42 }
43};

45template <typename TFrom, typename TTo, bool cloneAttrs = false>
46class OneToOnePattern : public OpConversionPattern<TFrom> {
47public:
48 using OpConversionPattern<TFrom>::OpConversionPattern;
49 using OpAdaptor = typename TFrom::Adaptor;
50
51 LogicalResult
52 matchAndRewrite(TFrom op, OpAdaptor adaptor,
53 ConversionPatternRewriter &rewriter) const override {
54 rewriter.replaceOpWithNewOp<TTo>(
55 op, adaptor.getOperands(),
56 cloneAttrs ? op->getAttrs() : ArrayRef<::mlir::NamedAttribute>());
57 return success();
58 }
59};

62public:

67 matchAndRewrite(arith::ExtSIOp op, OpAdaptor adaptor,

77class ExtZConversionPattern : public OpConversionPattern<arith::ExtUIOp> {
78public:
79 using OpConversionPattern<arith::ExtUIOp>::OpConversionPattern;
80 using OpAdaptor = typename arith::ExtUIOp::Adaptor;
81
82 LogicalResult
83 matchAndRewrite(arith::ExtUIOp op, OpAdaptor adaptor,
84 ConversionPatternRewriter &rewriter) const override {
85 auto loc = op.getLoc();
86 size_t outWidth = op.getOut().getType().getIntOrFloatBitWidth();
87 size_t inWidth = adaptor.getIn().getType().getIntOrFloatBitWidth();
88
89 rewriter.replaceOp(op, comb::ConcatOp::create(
90 rewriter, loc,
91 hw::ConstantOp::create(
92 rewriter, loc, APInt(outWidth - inWidth, 0)),
93 adaptor.getIn()));
94 return success();
95 }
96};

113class CompConversionPattern : public OpConversionPattern<arith::CmpIOp> {
114public:
115 using OpConversionPattern<arith::CmpIOp>::OpConversionPattern;
116 using OpAdaptor = typename arith::CmpIOp::Adaptor;
117
118 static comb::ICmpPredicate
119 arithToCombPredicate(arith::CmpIPredicate predicate) {
120 switch (predicate) {
121 case arith::CmpIPredicate::eq:
122 return comb::ICmpPredicate::eq;
123 case arith::CmpIPredicate::ne:
124 return comb::ICmpPredicate::ne;
125 case arith::CmpIPredicate::slt:
126 return comb::ICmpPredicate::slt;
127 case arith::CmpIPredicate::ult:
128 return comb::ICmpPredicate::ult;
129 case arith::CmpIPredicate::sle:
130 return comb::ICmpPredicate::sle;
131 case arith::CmpIPredicate::ule:
132 return comb::ICmpPredicate::ule;
133 case arith::CmpIPredicate::sgt:
134 return comb::ICmpPredicate::sgt;
135 case arith::CmpIPredicate::ugt:
136 return comb::ICmpPredicate::ugt;
137 case arith::CmpIPredicate::sge:
138 return comb::ICmpPredicate::sge;
139 case arith::CmpIPredicate::uge:
140 return comb::ICmpPredicate::uge;
141 }
142 llvm_unreachable("Unknown predicate");
143 }
144
145 LogicalResult
146 matchAndRewrite(arith::CmpIOp op, OpAdaptor adaptor,
147 ConversionPatternRewriter &rewriter) const override {
148 rewriter.replaceOpWithNewOp<comb::ICmpOp>(
149 op, arithToCombPredicate(op.getPredicate()), adaptor.getLhs(),
150 adaptor.getRhs());
151 return success();
152 }
153};

155struct ConstantConversionPattern
156 : public OpConversionPattern<arith::ConstantOp> {
157 using OpConversionPattern::OpConversionPattern;
158
159 LogicalResult
160 matchAndRewrite(arith::ConstantOp op, OpAdaptor adaptor,
161 ConversionPatternRewriter &rewriter) const override {
162 // `hw.constant` only supports integers.
163 if (!isa<IntegerType>(op.getType()))
164 return failure();
165
166 rewriter.replaceOpWithNewOp<hw::ConstantOp>(
167 op, cast<IntegerAttr>(adaptor.getValue()));
168 return success();
169 }
170};

172// Pattern for arith min/max operations.
173// These lower to: cmp + mux
174template <typename ArithOp, comb::ICmpPredicate pred>
175class MinMaxConversionPattern : public OpConversionPattern<ArithOp> {
176public:
177 using OpConversionPattern<ArithOp>::OpConversionPattern;
178 using OpAdaptor = typename ArithOp::Adaptor;
179
180 LogicalResult
181 matchAndRewrite(ArithOp op, OpAdaptor adaptor,
182 ConversionPatternRewriter &rewriter) const override {
183 // max(a, b) = mux(cmp(a, b), a, b)
184 // min(a, b) = mux(cmp(a, b), a, b)
185 auto cmp = comb::ICmpOp::create(rewriter, op.getLoc(), pred,
186 adaptor.getLhs(), adaptor.getRhs());
187 rewriter.replaceOpWithNewOp<comb::MuxOp>(op, cmp, adaptor.getLhs(),
188 adaptor.getRhs());
189 return success();
190 }
191};

193struct MapArithToCombPass
194 : public circt::impl::MapArithToCombPassBase<MapArithToCombPass> {
195public:
196 MapArithToCombPass(bool enableBestEffortLowering) {
197 this->enableBestEffortLowering = enableBestEffortLowering;
198 }
199
200 void runOnOperation() override {
201 auto *ctx = &getContext();
202
203 ConversionTarget target(*ctx);
204 target.addLegalDialect<comb::CombDialect, hw::HWDialect>();
205 if (!enableBestEffortLowering) {
206 target.addIllegalDialect<arith::ArithDialect>();
207 } else {
208 // We make all arith operations with a potential lowering here (as
209 // specified in circt::populateArithToCombPatterns) illegal
210 target.addIllegalOp<arith::AddIOp>();
211 target.addIllegalOp<arith::SubIOp>();
212 target.addIllegalOp<arith::MulIOp>();
213 target.addIllegalOp<arith::DivSIOp>();
214 target.addIllegalOp<arith::DivUIOp>();
215 target.addIllegalOp<arith::RemSIOp>();
216 target.addIllegalOp<arith::RemUIOp>();
217 target.addIllegalOp<arith::AndIOp>();
218 target.addIllegalOp<arith::OrIOp>();
219 target.addIllegalOp<arith::XOrIOp>();
220 target.addIllegalOp<arith::ShLIOp>();
221 target.addIllegalOp<arith::ShRSIOp>();
222 target.addIllegalOp<arith::ShRUIOp>();
223 target.addIllegalOp<arith::ExtSIOp>();
224 target.addIllegalOp<arith::ExtUIOp>();
225 target.addIllegalOp<arith::TruncIOp>();
226 target.addIllegalOp<arith::CmpIOp>();
227 target.addIllegalOp<arith::MaxSIOp>();
228 target.addIllegalOp<arith::MaxUIOp>();
229 target.addIllegalOp<arith::MinSIOp>();
230 target.addIllegalOp<arith::MinUIOp>();
231
232 // Force integer constants to be mapped to `hw.constant`.
233 target.addDynamicallyLegalOp<arith::ConstantOp>([](Operation *op) {
234 return !isa<IntegerType>(op->getResult(0).getType());
235 });
236
237 target.addDynamicallyLegalOp<arith::SelectOp>(
238 [](arith::SelectOp op) { return !hw::isHWValueType(op.getType()); });
239 }
240 MapArithTypeConverter typeConverter;
241 RewritePatternSet patterns(ctx);
242 populateArithToCombPatterns(patterns, typeConverter);
243
244 if (failed(applyPartialConversion(getOperation(), target,
245 std::move(patterns))))
246 signalPassFailure();
247 }
248};

252void circt::populateArithToCombPatterns(mlir::RewritePatternSet &patterns,
253 TypeConverter &typeConverter) {
254 patterns.insert<
255 OneToOnePattern<arith::AddIOp, comb::AddOp>,
256 OneToOnePattern<arith::SubIOp, comb::SubOp>,
257 OneToOnePattern<arith::MulIOp, comb::MulOp>,
258 OneToOnePattern<arith::DivSIOp, comb::DivSOp>,
259 OneToOnePattern<arith::DivUIOp, comb::DivUOp>,
260 OneToOnePattern<arith::RemSIOp, comb::ModSOp>,
261 OneToOnePattern<arith::RemUIOp, comb::ModUOp>,
262 OneToOnePattern<arith::AndIOp, comb::AndOp>,
263 OneToOnePattern<arith::OrIOp, comb::OrOp>,
264 OneToOnePattern<arith::XOrIOp, comb::XorOp>,
265 OneToOnePattern<arith::ShLIOp, comb::ShlOp>,
266 OneToOnePattern<arith::ShRSIOp, comb::ShrSOp>,
267 OneToOnePattern<arith::ShRUIOp, comb::ShrUOp>,
268 OneToOnePattern<arith::SelectOp, comb::MuxOp>,
269 MinMaxConversionPattern<arith::MaxSIOp, comb::ICmpPredicate::sge>,
270 MinMaxConversionPattern<arith::MaxUIOp, comb::ICmpPredicate::uge>,
271 MinMaxConversionPattern<arith::MinSIOp, comb::ICmpPredicate::sle>,
272 MinMaxConversionPattern<arith::MinUIOp, comb::ICmpPredicate::ule>,
273 ExtSConversionPattern, ExtZConversionPattern, TruncateConversionPattern,
274 CompConversionPattern, ConstantConversionPattern>(typeConverter,
275 patterns.getContext());
276}
