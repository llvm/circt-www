24LogicalResult circt::arc::collectAndExportModelInfo(ModuleOp module,
25 llvm::raw_ostream &os) {
26 SmallVector<ModelInfo> models;
27 if (failed(collectModels(module, models)))
28 return failure();
29 serializeModelInfoToJson(os, models);
