25/// This table tracks nlas and what modules participate in them.
26///
27/// To use this class, retrieve a cached copy from the analysis manager:
28/// auto &nlaTable = getAnalysis<NLATable>(getOperation());
29class NLATable {
30
31public:
32 /// Create a new NLA table of a circuit. This must be called on a FIRRTL
33 /// CircuitOp or MLIR ModuleOp. To ensure that the analysis does not return
34 /// stale data while a pass is running, it should be kept up-to-date when
35 /// modules are added or renamed and NLAs are updated.
36 explicit NLATable(Operation *operation);
37
38 /// Lookup all NLAs an operation participates in. This returns a reference to
39 /// the internal record, so make a copy before making any update to the
40 /// NLATable.
41 ArrayRef<hw::HierPathOp> lookup(Operation *op);
42
43 /// Lookup all NLAs an operation participates in. This returns a reference to
44 /// the internal record, so make a copy before making any update to the
45 /// NLATable.
46 ArrayRef<hw::HierPathOp> lookup(StringAttr name);
47
48 /// Resolve a symbol to an NLA.
49 hw::HierPathOp getNLA(StringAttr name);
50
51 /// Resolve a symbol to a Module.
52 FModuleLike getModule(StringAttr name);
53
54 /// Compute the NLAs that are common between the two modules, `mod1` and
55 /// `mod2` and insert them into the set `common`.
56 /// The set of NLAs that an instance op participates in is the set of common
57 /// NLAs between the parent module and the instance target. This can be used
58 /// to get the set of NLAs that an InstanceOp participates in, instead of
59 /// recording them on the op in the IR.
60 void commonNLAs(StringAttr mod1, StringAttr mod2,
61 DenseSet<hw::HierPathOp> &common) {
62 auto mod1NLAs = lookup(mod1);
63 auto mod2NLAs = lookup(mod2);
64 common.insert(mod1NLAs.begin(), mod1NLAs.end());
65 DenseSet<hw::HierPathOp> set2(mod2NLAs.begin(), mod2NLAs.end());
66 llvm::set_intersect(common, set2);
67 }
68
69 /// Get the NLAs that the InstanceOp participates in, insert it to the
70 /// DenseSet `nlas`.
71 void getInstanceNLAs(InstanceOp inst, DenseSet<hw::HierPathOp> &nlas) {
72 auto instSym = getInnerSymName(inst);
73 // If there is no inner sym on the InstanceOp, then it does not participate
74 // in any NLA.
75 if (!instSym)
76 return;
77 auto mod = inst->getParentOfType<FModuleOp>().getNameAttr();
78 // Get the NLAs that are common between the parent module and the target
79 // module. This should contain the NLAs that this InstanceOp participates
80 // in.
81 commonNLAs(inst->getParentOfType<FModuleOp>().getNameAttr(),
82 inst.getModuleNameAttr().getAttr(), nlas);
83 // Handle the case when there are more than one Instances for the same
84 // target module. Getting the `commonNLA`, in that case is not enough,
85 // remove the NLAs that donot have the InstanceOp as the innerSym.
86 nlas.remove_if([&](auto nla) { return !nla.hasInnerSym(mod, instSym); });
87 }
88
89 /// Get the NLAs that the module `modName` particiaptes in, and insert them
90 /// into the DenseSet `nlas`.
91 void getNLAsInModule(StringAttr modName, DenseSet<hw::HierPathOp> &nlas) {
92 for (auto nla : lookup(modName))
93 nlas.insert(nla);
94 }
95
96 //===-------------------------------------------------------------------------
97 // Methods to keep an NLATable up to date.
98 //
99 // These methods are not thread safe. Make sure that modifications are
100 // properly synchronized or performed in a serial context. When the
101 // NLATable is used as an analysis, this is only safe when the pass is
102 // on a CircuitOp.
103
104 /// Insert a new NLA. This updates two internal records,
105 /// 1. Update the map for the `nlaOp` name to the Operation.
106 /// 2. For each module in the NLA namepath, insert the NLA into the list of
107 /// hwHierPathOps that participate in the corresponding module. This does
108 /// not update the module name to module op map, if any potentially new module
109 /// in the namepath does not already exist in the record.
110 void addNLA(hw::HierPathOp nla);
111
112 /// Remove the NLA from the analysis. This updates two internal records,
113 /// 1. Remove the NLA name to the operation map entry.
114 /// 2. For each module in the namepath of the NLA, remove the entry from the
115 /// list of NLAs that the module participates in.
116 /// Note that this invalidates any reference to the NLA list returned by
117 /// 'lookup'.
118 void erase(hw::HierPathOp nlaOp, SymbolTable *symbolTable = nullptr);
119
120 /// Record a new FModuleLike operation. This updates the Module name to Module
121 /// operation map.
122 void addModule(FModuleLike mod) { symToOp[mod.getModuleNameAttr()] = mod; }
123
124 /// Stop tracking a module. Remove the module from two internal records,
125 /// 1. Module name to Module op map.
126 /// 2. Module name to list of NLAs that the module participates in.
127 void eraseModule(StringAttr name) {
128 symToOp.erase(name);
129 nodeMap.erase(name);
130 }
131
132 /// Replace the module `oldModule` with `newModule` in the namepath of the nla
133 /// `nlaName`. This moves the nla from the list of `oldModule` to `newModule`.
134 /// Move `nlaName` from the list of NLAs that `oldModule` participates in to
135 /// `newModule`. This can delete and invalidate any reference returned by
136 /// `lookup`.
137 void updateModuleInNLA(StringAttr nlaName, StringAttr oldModule,
138 StringAttr newModule);
139
140 /// Replace the module `oldModule` with `newModule` in the namepath of the nla
141 /// `nlaOp`. This moves the nla from the list of `oldModule` to `newModule`.
142 /// Move `nlaOp` from the list of NLAs that `oldModule` participates in to
143 /// `newModule`. This can delete and invalidate any reference returned by
144 /// `lookup`.
145 void updateModuleInNLA(hw::HierPathOp nlaOp, StringAttr oldModule,
146 StringAttr newModule);
147
148 /// Rename a module, this updates the name to module tracking and the name to
149 /// NLA tracking. This moves all the NLAs that `oldModName` is participating
150 /// in to the `newModName`. The `oldModName` must exist in the name to module
151 /// record. This also removes all the entries for `oldModName`.
152 void renameModule(StringAttr oldModName, StringAttr newModName);
153
154 /// Replace the module `oldModName` with `newModName` in the namepath of any
155 /// NLA. Since the module is being updated, the symbols inside the module
156 /// should also be renamed. Use the rename map `innerSymRenameMap` to update
157 /// the inner_sym names in the namepath.
158 void renameModuleAndInnerRef(
159 StringAttr newModName, StringAttr oldModName,
160 const DenseMap<StringAttr, StringAttr> &innerSymRenameMap);
161
162 /// Remove the NLA from the Module. This updates the module name to NLA
163 /// tracking.
164 void removeNLAfromModule(hw::HierPathOp nla, StringAttr mod) {
165 llvm::erase(nodeMap[mod], nla);
166 }
167
168 /// Remove all the nlas in the set `nlas` from the module. This updates the
169 /// module name to NLA tracking.
170 void removeNLAsfromModule(const DenseSet<hw::HierPathOp> &nlas,
171 StringAttr mod) {
172 llvm::erase_if(nodeMap[mod],
173 [&nlas](const auto &nla) { return nlas.count(nla); });
174 }
175
176 /// Add the nla to the module. This ensures that the list of NLAs that the
177 /// module participates in is updated. This will be required if `mod` is added
178 /// to the namepath of `nla`.
179 void addNLAtoModule(hw::HierPathOp nla, StringAttr mod) {
180 nodeMap[mod].push_back(nla);
181 }
182
183private:
184 NLATable(const NLATable &) = delete;
185
186 /// Map modules to the NLA's that target them.
187 llvm::DenseMap<StringAttr, SmallVector<hw::HierPathOp, 4>> nodeMap;
188
189 /// Map symbol names to module and NLA operations.
190 llvm::DenseMap<StringAttr, Operation *> symToOp;
191};
