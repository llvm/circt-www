71void ReadChannelPort::disconnect() {
72 {
73 std::unique_lock<std::mutex> lock(callbackMutex);
74 // Revoke the callback under the same mutex used by invokeCallback(). New
75 // deliveries need to observe the disconnected state and the callback
76 // nullification at the same time.
77 callback = nullptr;
78 mode = Mode::Disconnected;
79 // Wait for any callback which already snapped the old function to finish
80 // before clearing per-connection state below.
81 callbackCv.wait(lock, [this]() { return activeCallbacks == 0; });
82 }
83
84 pollingState.reset();
85 resetTranslationState();
86}

88bool ReadChannelPort::invokeCallback(
89 std::unique_ptr<SegmentedMessageData> &msg) {
90 ReadCallback activeCallback;
91 {
92 std::lock_guard<std::mutex> lock(callbackMutex);
93 // Check the disconnected state and snapshot the callback while holding the
94 // same mutex as disconnect(). That prevents disconnect() from clearing the
95 // callback after we decide to invoke it but before we've retained our own
96 // safe local copy.
97 if (mode == Mode::Disconnected)
98 return false;
99 assert(callback && "Callback should be set in non-disconnected mode");
100 activeCallback = callback;
101 // Count only callbacks which successfully captured a callable target.
102 // disconnect() waits on this count so it can safely tear down state which
103 // the callback path may still reference.
104 ++activeCallbacks;
105 }
106
107 // Release the in-flight slot on every exit path, including exceptions from
108 // the callback itself.
109 auto finish = [this]() {
110 std::lock_guard<std::mutex> lock(callbackMutex);
111 assert(activeCallbacks > 0 && "Callback count underflow");
112 --activeCallbacks;
113 if (activeCallbacks == 0)
114 callbackCv.notify_all();
115 };
116
117 try {
118 // For zero-width port types (e.g. `VoidType`), transports which require
119 // every message to be at least one byte send a single placeholder byte;
120 // strip it. Empty messages are also accepted, both from transports which
121 // support them and from backends re-delivering a message the callback
122 // previously rejected (and which was already stripped). Anything else is
123 // malformed transport data. The matching pad happens in
124 // `WriteChannelPort::maybePadEmptyMessage`.
125 if (msg && type && type->getBitWidth() == 0) {
126 size_t size = msg->totalSize();
127 if (size == 1)
128 msg = std::make_unique<MessageData>();
129 else if (size != 0)
130 throw std::runtime_error("zero-width message must be empty or use a "
131 "single-byte placeholder payload");
132 }
133
134 bool consumed = activeCallback(msg);
135 finish();
136 return consumed;
137 } catch (...) {
138 finish();
139 throw;
140 }
141}

143void ReadChannelPort::connect(ReadCallback callback,
144 const ConnectOptions &options) {
145 if (mode != Mode::Disconnected)
146 throw std::runtime_error("Channel already connected");
147
148 resetTranslationState();
149
150 if (translationInfo)
151 translationInfo->precomputeFrameInfo();
152
153 if (options.translateMessage && translationInfo) {
154 translationInfo->requireTranslationSupported();
155 this->callback = [this, cb = std::move(callback)](
156 std::unique_ptr<SegmentedMessageData> &data) {
157 if (!translatedMessage) {
158 MessageData flat = data->toMessageData();
159 if (!translateIncoming(flat))
160 return true;
161 translatedMessage =
162 std::make_unique<MessageData>(std::move(translationBuffer));
163 }
164
165 if (!cb(translatedMessage))
166 return false;
167
168 translatedMessage.reset();
169 return true;
170 };
171 } else {
172 this->callback = callback;
173 }
174 connectImpl(options);
175 mode = Mode::Callback;
176}

186void ReadChannelPort::connect(const ConnectOptions &options) {
187 if (mode != Mode::Disconnected)
188 throw std::runtime_error("Channel already connected");
189
190 if (options.bufferSize.has_value())
191 maxDataQueueMsgs = options.bufferSize.value();
192 pollingState.emplace(maxDataQueueMsgs);
193
194 resetTranslationState();
195
196 if (translationInfo)
197 translationInfo->precomputeFrameInfo();
198
199 bool translate = options.translateMessage && translationInfo;
200 if (translate)
201 translationInfo->requireTranslationSupported();
202 this->callback = [this,
203 translate](std::unique_ptr<SegmentedMessageData> &msg) {
204 MessageData data;
205 if (translate) {
206 if (!translatedMessage) {
207 MessageData flat = msg->toMessageData();
208 if (!translateIncoming(flat))
209 return true;
210 translatedMessage =
211 std::make_unique<MessageData>(std::move(translationBuffer));
212 }
213 data = translatedMessage->toMessageData();
214 } else {
215 data = msg->toMessageData();
216 }
217
218 if (!pollingState->enqueue(data))
219 return false;
220
221 translatedMessage.reset();
222 return true;
223 };
224 connectImpl(options);
225 mode = Mode::Polling;
226}

228std::future<MessageData> ReadChannelPort::readAsync() {
229 if (mode == Mode::Callback)
230 throw std::runtime_error(
231 "Cannot read from a callback channel. Call `connect()` without a "
232 "callback specified to use polling mode.");
233 if (mode == Mode::Disconnected)
234 throw std::runtime_error(
235 "Cannot read from a disconnected channel. `connect()` the channel "
236 "without a callback before calling `readAsync()`.");
237
238 assert(mode == Mode::Polling && "Channel must be in polling mode to read");
239 assert(pollingState && "Polling state should be initialized in polling mode");
240
241 return pollingState->readAsync();
242}

248void ChannelPort::TranslationInfo::requireTranslationSupported() const {
249 // Reject serial-encoded windows: any field with bulkCountWidth > 0 indicates
250 // serial (bulk) encoding, which the translator does not yet support. Surface
251 // this clearly at connect()-time rather than silently buffering forever in
252 // translateIncoming().
253 for (const auto &frame : windowType->getFrames()) {
254 for (const auto &field : frame.fields) {
255 if (field.bulkCountWidth > 0)
256 throw std::runtime_error(
257 "Window translation for serial (bulk) list encoding is not "
258 "implemented (window '" +
259 windowType->getName() + "', frame '" + frame.name + "', field '" +
260 field.name +
261 "'). Connect the port with ConnectOptions{translateMessage=false} "
262 "and decode the frames manually.");
263 }
264 }
265}

267void ChannelPort::TranslationInfo::precomputeFrameInfo() {
268 const Type *intoType = windowType->getIntoType();
269
270 const StructType *intoStruct = dynamic_cast<const StructType *>(intoType);
271 if (!intoStruct)
272 throw std::runtime_error(
273 "Window intoType must be a struct for translation");
274
275 const auto &intoFields = intoStruct->getFields();
276
277 // Build a map from field name to (offset, type, isListField).
278 // For list fields, the offset is where the list_length field will be stored,
279 // and we also track the element type.
280 struct FieldInfo {
281 size_t offset;
282 const Type *type;
283 bool isList;
284 const Type *listElementType; // Only valid if isList is true
285 size_t listElementSize; // Only valid if isList is true
286 };
287 std::map<std::string, FieldInfo> fieldMap;
288 size_t currentOffset = 0;
289 hasListField = false;
290
291 auto processField = [&](const std::string &name, const Type *fieldType) {
292 auto *listType = dynamic_cast<const ListType *>(fieldType);
293 if (listType) {
294 hasListField = true;
295 // For list fields in the intoType:
296 // - We store a list_length (size_t / 8 bytes) followed by the list data
297 // - The offset here is where the list_length goes
298 const Type *elemType = listType->getElementType();
299 std::ptrdiff_t elemBits = elemType->getBitWidth();
300 if (elemBits < 0)
301 throw std::runtime_error(
302 "Cannot translate list with dynamically-sized element type: " +
303 name);
304 if (elemBits == 0)
305 throw std::runtime_error(
306 "Cannot translate list with zero-width element type: " + name);
307 if (elemBits % 8 != 0)
308 throw std::runtime_error(
309 "Cannot translate list element with non-byte-aligned size: " +
310 name);
311 size_t elemBytes = (static_cast<size_t>(elemBits) + 7) / 8;
312 fieldMap[name] = {currentOffset, fieldType, true, elemType, elemBytes};
313 // Reserve space for list_length. We use size_t (8 bytes on 64-bit
314 // platforms) for consistency with standard C/C++ container sizes.
315 // Note: This means the translated message format is platform-dependent.
316 currentOffset += sizeof(size_t);
317 // List data will be appended dynamically after the fixed header
318 } else {
319 std::ptrdiff_t fieldBits = fieldType->getBitWidth();
320 if (fieldBits < 0)
321 throw std::runtime_error("Cannot translate field with dynamic size: " +
322 name);
323 if (fieldBits % 8 != 0)
324 throw std::runtime_error(
325 "Cannot translate field with non-byte-aligned size: " + name);
326 size_t fieldBytes = (static_cast<size_t>(fieldBits) + 7) / 8;
327 fieldMap[name] = {currentOffset, fieldType, false, nullptr, 0};
328 currentOffset += fieldBytes;
329 }
330 };
331
332 if (intoStruct->isReverse()) {
333 for (auto it = intoFields.rbegin(); it != intoFields.rend(); ++it)
334 processField(it->first, it->second);
335 } else {
336 for (const auto &[name, fieldType] : intoFields)
337 processField(name, fieldType);
338 }
339
340 // intoTypeBytes is now the size of the fixed header portion
341 // (for types with lists, this excludes the variable-length list data)
342 intoTypeBytes = currentOffset;
343
344 const auto &windowFrames = windowType->getFrames();
345 frames.clear();
346 frames.reserve(windowFrames.size());
347
348 for (const auto &frame : windowFrames) {
349 FrameInfo frameInfo;
350 size_t frameOffset = 0;
351
352 // Calculate frame layout in SV memory order.
353 // SV structs are laid out with the last declared field at the lowest
354 // address. So when we iterate fields in reverse order (rbegin to rend),
355 // we get the memory layout order.
356 //
357 // For list fields, the lowered struct in CIRCT adds fields in this order:
358 // 1. list_data[numItems] - array of list elements
359 // 2. list_size - (if numItems > 1) count of valid items
360 // 3. last - indicates end of list
361 //
362 // In SV memory layout (reversed), this becomes:
363 // offset 0: last (1 byte)
364 // offset 1: list_size (if present)
365 // offset after size: list_data[numItems]
366 // offset after list: header fields...
367 struct FrameFieldLayout {
368 std::string name;
369 size_t frameOffset;
370 size_t size;
371 bool isList;
372 size_t numItems; // From window spec
373 size_t listElementSize; // Size of each list element
374 size_t bufferOffset; // Offset in translation buffer
375 const Type *listElementType; // Element type for list fields
376 };
377 std::vector<FrameFieldLayout> fieldLayouts;
378
379 for (auto fieldIt = frame.fields.rbegin(); fieldIt != frame.fields.rend();
380 ++fieldIt) {
381 const WindowType::Field &field = *fieldIt;
382 auto it = fieldMap.find(field.name);
383 if (it == fieldMap.end())
384 throw std::runtime_error("Frame field '" + field.name +
385 "' not found in intoType");
386
387 const FieldInfo &fieldInfo = it->second;
388 FrameFieldLayout layout;
389 layout.name = field.name;
390 layout.bufferOffset = fieldInfo.offset;
391 layout.isList = fieldInfo.isList;
392 layout.numItems = field.numItems;
393 layout.listElementType = fieldInfo.listElementType;
394 layout.listElementSize = fieldInfo.listElementSize;
395
396 if (fieldInfo.isList) {
397 // For list fields, the frame layout is (in memory order):
398 // 1. 'last' field (1 byte)
399 // 2. list_data (one element per frame)
400 // Note: numItems > 1 is not yet supported.
401 size_t numItems = field.numItems > 0 ? field.numItems : 1;
402 if (numItems != 1)
403 throw std::runtime_error(
404 "List translation with numItems > 1 is not yet supported. "
405 "Field '" +
406 field.name + "' has numItems=" + std::to_string(numItems));
407
408 // 'last' field comes first in memory
409 size_t lastOffset = frameOffset;
410 frameOffset += 1;
411
412 // List data comes after 'last'
413 layout.frameOffset = frameOffset;
414 layout.size = fieldInfo.listElementSize;
415 fieldLayouts.push_back(layout);
416 frameOffset += layout.size;
417
418 // Create ListFieldInfo for this list field
419 ListFieldInfo listInfo;
420 listInfo.fieldName = layout.name;
421 listInfo.dataOffset = layout.frameOffset;
422 listInfo.elementSize = fieldInfo.listElementSize;
423 listInfo.listLengthBufferOffset = layout.bufferOffset;
424 listInfo.listDataBufferOffset = intoTypeBytes;
425 listInfo.lastFieldOffset = lastOffset;
426 frameInfo.listField = listInfo;
427 } else {
428 // Regular (non-list) field
429 std::ptrdiff_t fieldBits = fieldInfo.type->getBitWidth();
430 layout.frameOffset = frameOffset;
431 layout.size = (static_cast<size_t>(fieldBits) + 7) / 8;
432 fieldLayouts.push_back(layout);
433 frameOffset += layout.size;
434 }
435 }
436
437 frameInfo.expectedSize = frameOffset;
438
439 // Second pass: build copy ops for non-list fields
440 for (const auto &layout : fieldLayouts) {
441 if (!layout.isList) {
442 // Regular field - add copy op
443 frameInfo.copyOps.push_back(
444 {layout.frameOffset, layout.bufferOffset, layout.size});
445 }
446 // List fields were already handled in the first pass
447 }
448
449 // Sort copy ops by frameOffset to ensure processing order matches frame
450 // layout.
451 std::sort(frameInfo.copyOps.begin(), frameInfo.copyOps.end(),
452 [](const CopyOp &a, const CopyOp &b) {
453 return a.frameOffset < b.frameOffset;
454 });
455
456 // Merge adjacent copy ops
457 if (!frameInfo.copyOps.empty()) {
458 std::vector<CopyOp> mergedOps;
459 mergedOps.push_back(frameInfo.copyOps[0]);
460
461 for (size_t i = 1; i < frameInfo.copyOps.size(); ++i) {
462 CopyOp &last = mergedOps.back();
463 CopyOp current = frameInfo.copyOps[i];
464
465 if (last.frameOffset + last.size == current.frameOffset &&
466 last.bufferOffset + last.size == current.bufferOffset) {
467 last.size += current.size;
468 } else {
469 mergedOps.push_back(current);
470 }
471 }
472
473 frameInfo.copyOps = std::move(mergedOps);
474 }
475
476 frames.push_back(std::move(frameInfo));
477 }
478
479 // Set frameBytes from the lowered type's bit width. Fall back to the
480 // maximum computed frame expectedSize if the type system can't report width
481 // (e.g. opaque union types).
482 std::ptrdiff_t loweredBits = windowType->getLoweredType()->getBitWidth();
483 if (loweredBits > 0) {
484 frameBytes = (loweredBits + 7) / 8;
485 } else {
486 frameBytes = 0;
487 for (const auto &f : frames)
488 frameBytes = std::max(frameBytes, f.expectedSize);
489 }
490}

492bool ReadChannelPort::translateIncoming(MessageData &data) {
493 assert(translationInfo &&
494 "Translation type must be set for window translation.");
495
496 // Get the frame data directly.
497 const uint8_t *frameData = data.getBytes();
498 size_t frameDataSize = data.size();
499
500 // Get the frame metadata for the current expected frame.
501 const auto &frameInfo = translationInfo->frames[nextFrameIndex];
502
503 // Check size
504 if (frameDataSize < frameInfo.expectedSize)
505 throw std::runtime_error("Frame data too small: expected at least " +
506 std::to_string(frameInfo.expectedSize) +
507 " bytes, got " + std::to_string(frameDataSize) +
508 " bytes");
509
510 // Check if this is the first frame of a new message.
511 // For list frames, we use accumulatingListData to track whether we're
512 // continuing to accumulate list items.
513 bool isFirstFrame = (nextFrameIndex == 0) && !accumulatingListData;
514
515 if (isFirstFrame) {
516 // Initialize the translation buffer to hold the fixed header portion.
517 translationBuffer.resize(translationInfo->intoTypeBytes, 0);
518 // Clear list data buffer for types with lists
519 if (translationInfo->hasListField)
520 listDataBuffer.clear();
521 }
522
523 // Execute copy ops for non-list fields
524 for (const auto &op : frameInfo.copyOps)
525 std::memcpy(translationBuffer.data() + op.bufferOffset,
526 frameData + op.frameOffset, op.size);
527
528 // Handle list field if present in this frame
529 if (frameInfo.listField.has_value()) {
530 const auto &listInfo = frameInfo.listField.value();
531
532 // With numItems == 1, each frame contains exactly one list element
533 size_t bytesToCopy = listInfo.elementSize;
534 // Additional check: dataOffset must not be beyond frameDataSize
535 if (listInfo.dataOffset > frameDataSize)
536 throw std::runtime_error("List data offset is beyond frame bounds");
537 // Bounds check to prevent buffer overflow from corrupted _size field
538 if (listInfo.dataOffset + bytesToCopy > frameDataSize)
539 throw std::runtime_error("List data extends beyond frame bounds");
540 size_t oldSize = listDataBuffer.size();
541 listDataBuffer.resize(oldSize + bytesToCopy);
542 std::memcpy(listDataBuffer.data() + oldSize,
543 frameData + listInfo.dataOffset, bytesToCopy);
544
545 // Check if this is the last frame of the list
546 uint8_t lastFlag = frameData[listInfo.lastFieldOffset];
547 if (lastFlag) {
548 // List is complete. Build the final message:
549 // [fixed header with list_length (size_t)][list data...]
550 // Write the actual length to the listLengthBufferOffset position.
551 size_t listLength = listDataBuffer.size() / listInfo.elementSize;
552 size_t *listLengthPtr = reinterpret_cast<size_t *>(
553 translationBuffer.data() + listInfo.listLengthBufferOffset);
554 *listLengthPtr = listLength;
555
556 // Append list data to translation buffer
557 size_t headerSize = translationBuffer.size();
558 translationBuffer.resize(headerSize + listDataBuffer.size());
559 std::memcpy(translationBuffer.data() + headerSize, listDataBuffer.data(),
560 listDataBuffer.size());
561
562 // Reset for next message
563 nextFrameIndex = 0;
564 listDataBuffer.clear();
565 accumulatingListData = false;
566 return true;
567 }
568
569 // Not the last frame - stay on this frame index for list accumulation
570 // (list frames repeat until last=true)
571 accumulatingListData = true;
572 return false;
573 }
574
575 // No list field in this frame - advance to next frame
576 nextFrameIndex++;
577 size_t numFrames = translationInfo->frames.size();
578
579 // Check if all frames have been received.
580 if (nextFrameIndex >= numFrames) {
581 // Reset for the next message.
582 nextFrameIndex = 0;
583 return true;
584 }
585
586 return false;
587}

589void WriteChannelPort::translateOutgoing(const MessageData &data) {
590 assert(translationInfo &&
591 "Translation type must be set for window translation.");
592
593 const uint8_t *srcData = data.getBytes();
594 size_t srcDataSize = data.size();
595
596 if (srcDataSize < translationInfo->intoTypeBytes)
597 throw std::runtime_error("Source data too small: expected at least " +
598 std::to_string(translationInfo->intoTypeBytes) +
599 " bytes, got " + std::to_string(srcDataSize) +
600 " bytes");
601
602 // Check if we have list fields
603 if (!translationInfo->hasListField) {
604 // No list fields - simple fixed-size translation
605 for (const auto &frameInfo : translationInfo->frames) {
606 std::vector<uint8_t> frameBuffer(frameInfo.expectedSize, 0);
607 for (const auto &op : frameInfo.copyOps)
608 std::memcpy(frameBuffer.data() + op.frameOffset,
609 srcData + op.bufferOffset, op.size);
610 translationBuffer.emplace_back(std::move(frameBuffer));
611 }
612 return;
613 }
614
615 // Handle list fields - need to split list data into multiple frames
616 for (const auto &frameInfo : translationInfo->frames) {
617 if (!frameInfo.listField.has_value()) {
618 // Non-list frame - copy as normal
619 std::vector<uint8_t> frameBuffer(frameInfo.expectedSize, 0);
620 for (const auto &op : frameInfo.copyOps)
621 std::memcpy(frameBuffer.data() + op.frameOffset,
622 srcData + op.bufferOffset, op.size);
623 translationBuffer.emplace_back(std::move(frameBuffer));
624 } else {
625 // List frame - need to generate multiple frames
626 const auto &listInfo = frameInfo.listField.value();
627
628 // Read the list length from the source data
629 size_t listLength = 0;
630 std::memcpy(&listLength, srcData + listInfo.listLengthBufferOffset,
631 sizeof(size_t));
632
633 // Check that the buffer is large enough for the list data
634 if (translationInfo->intoTypeBytes + (listLength * listInfo.elementSize) >
635 srcDataSize) {
636 throw std::runtime_error(
637 "Source buffer too small for list data: possible corrupted or "
638 "inconsistent list length field");
639 }
640 // Get pointer to list data (after the fixed header)
641 const uint8_t *listData = srcData + translationInfo->intoTypeBytes;
642
643 // Generate frames for the list
644 size_t itemsRemaining = listLength;
645 size_t listDataOffset = 0;
646
647 // Handle empty list case - still need to send one frame with last=true
648 if (listLength == 0)
649 throw std::runtime_error(
650 "Cannot send empty lists - parallel ESI list encoding requires at "
651 "least one frame to be sent, and each frame must contain at least "
652 "one element.");
653
654 while (itemsRemaining > 0) {
655 std::vector<uint8_t> frameBuffer(frameInfo.expectedSize, 0);
656
657 // Copy non-list fields (header data) to each frame
658 for (const auto &op : frameInfo.copyOps)
659 std::memcpy(frameBuffer.data() + op.frameOffset,
660 srcData + op.bufferOffset, op.size);
661
662 // With numItems == 1, each frame contains exactly one list element
663 size_t bytesInThisFrame = listInfo.elementSize;
664
665 // Copy list data
666 std::memcpy(frameBuffer.data() + listInfo.dataOffset,
667 listData + listDataOffset, bytesInThisFrame);
668
669 // Update remaining count
670 itemsRemaining -= 1;
671 listDataOffset += bytesInThisFrame;
672
673 // Set last field
674 frameBuffer[listInfo.lastFieldOffset] = (itemsRemaining == 0) ? 1 : 0;
675
676 translationBuffer.emplace_back(std::move(frameBuffer));
677 }
678 }
679 }
680}

682std::vector<MessageData>
683WriteChannelPort::getMessageFrames(const MessageData &data) {
684 size_t frameBytes = getFrameSizeBytes();
685 assert(frameBytes > 0 && "Frame size must be greater than 0");
686 if (data.getSize() % frameBytes != 0)
687 throw std::runtime_error(
688 "write message size (" + std::to_string(data.getSize()) +
689 ") is not a multiple of frame size (" + std::to_string(frameBytes) +
690 ") on type " + type->toString());
691 std::vector<MessageData> frames;
692 size_t numFrames = data.getSize() / frameBytes;
693 const uint8_t *ptr = data.getBytes();
694 for (size_t i = 0; i < numFrames; ++i) {
695 frames.emplace_back(ptr, frameBytes);
696 ptr += frameBytes;
697 }
698 return frames;
699}

710 /// Stop this connection: after this returns, it doesn't touch the port
711 /// again. Waits for another thread writing on it to stop.
712 void stop() {
713 // Set before waiting, so a writer on another thread drops what it fetches
714 // meanwhile.
715 stopped = true;
716 std::lock_guard<std::recursive_mutex> lock(mutex);
717 pendingMsg.reset();
718 }
719
720 /// Mark that the DataFetchCallback may have data, then poll().
721 void notify() {
722 dataAvailable = true;
723 poll();
724 }
725
726 /// Write fetched data (fetching more as needed) until the DataFetchCallback
727 /// has none, the backend backpressures, or the connection is stopped. Does
728 /// nothing if another thread is doing so, or if called from a callback.
729 void poll();
730
731private:
732 enum class DrainResult {
733 /// The DataFetchCallback has no data.
734 Idle,
735 /// The backend backpressured.
736 Blocked,
737 /// Wrote the maximum number of messages for one call.
738 Yielded,
739 /// The connection was stopped.
740 Stopped,
741 };
742 /// The body of poll(). Requires `mutex`.
743 DrainResult drain();
744
745 WriteChannelPort &port;
746 const DataFetchCallback fetch;
747 const WriteErrorCallback onError;
748
749 /// Held while writing, including while calling the callbacks. Recursive so
750 /// that the callbacks can disconnect the port or call
751 /// notifyAsyncDataAvailable().
752 std::recursive_mutex mutex;
753 /// Set by the thread holding `mutex` while in drain(), so callbacks don't
754 /// start another one. We need this instead of relying on a normal mutex
755 /// because a normal mutex when 'try_to_lock' is called by the same thread the
756 /// behavior is undefined.
757 bool draining = false;
758 /// Set when the port is disconnected (or reconnected) or destroyed. After
759 /// that, this connection doesn't touch the port.
760 std::atomic<bool> stopped = false;
761 /// False once `fetch` returns nullptr (or throws), until notify() is called.
762 /// Atomic since notify() sets it without `mutex`.
763 std::atomic<bool> dataAvailable = true;
764 /// A fetched message which the port hasn't accepted yet. Guarded by
765 /// `mutex`.
766 std::unique_ptr<SegmentedMessageData> pendingMsg;
767};
768
769void WriteChannelPort::DefaultAsyncWriter::poll() {
770 DrainResult result = DrainResult::Idle;
771 do {
772 std::unique_lock<std::recursive_mutex> lock(mutex, std::try_to_lock);
773 if (!lock.owns_lock() || draining)
774 break;
775 draining = true;
776 try {
777 result = drain();
778 } catch (...) {
779 // Thrown by the error callback.
780 draining = false;
781 throw;
782 }
783 draining = false;
784 // Repeat for a notify() which arrived after drain() last checked
785 // `dataAvailable` but before the lock was released, since it couldn't
786 // poll() itself. Otherwise, the next poll() continues.
787 } while (result == DrainResult::Idle && dataAvailable);
788}

790WriteChannelPort::DefaultAsyncWriter::DrainResult
791WriteChannelPort::DefaultAsyncWriter::drain() {
792 size_t written = 0;
793 while (!stopped) {
794 if (!pendingMsg) {
795 // Leave the rest for the next call.
796 if (written == kMaxAsyncWritesPerCall)
797 return DrainResult::Yielded;
798 // Cleared before fetching, so a notification which arrives while
799 // fetching (and which would be missed if this returns nullptr) isn't
800 // lost.
801 if (!dataAvailable.exchange(false)) {
802 // tryWrite() only sends the rest of a windowed message's frames
803 // before the next message, so with none, send them here.
804 try {
805 return port.flushInternal() ? DrainResult::Idle
806 : DrainResult::Blocked;
807 } catch (...) {
808 port.discardPendingFrames();
809 onError(std::current_exception());
810 continue;
811 }
812 }
813 std::exception_ptr fetchError;
814 try {
815 pendingMsg = fetch();
816 } catch (...) {
817 fetchError = std::current_exception();
818 }
819 // If the callback disconnected the port, drop the message (or error).
820 if (stopped) {
821 pendingMsg.reset();
822 break;
823 }
824 if (fetchError) {
825 onError(fetchError);
826 continue;
827 }
828 if (!pendingMsg)
829 continue;
830 // The callback had data, so it may have more.
831 dataAvailable = true;
832 }
833
834 try {
835 if (!port.tryWriteSegmentedImpl(pendingMsg))
836 return DrainResult::Blocked;
837 ++written;
838 } catch (...) {
839 pendingMsg.reset();
840 port.discardPendingFrames();
841 onError(std::current_exception());
842 }
843 }
844 return DrainResult::Stopped;
845}

948void WriteChannelPort::writeInternal(const MessageData &data) {
949 if (translateMessages) {
950 assert(translationBuffer.empty() &&
951 "Cannot call write() with pending translated messages");
952 translateOutgoing(data);
953 for (auto &msg : translationBuffer)
954 writeImpl(maybePadEmptyMessage(msg));
955 translationBuffer.clear();
956 } else {
957 writeImpl(maybePadEmptyMessage(data));
958 }
959}

961bool WriteChannelPort::tryWriteInternal(const MessageData &data) {
962 if (translateMessages) {
963 // Do not accept a new message if there are pending messages to flush.
964 if (!flushInternal())
965 return false;
966 assert(translationBuffer.empty() &&
967 "Translation buffer should be empty after successful flush");
968 translateOutgoing(data);
969 // If the first frame isn't accepted, don't keep the message: nothing
970 // would flush it unless the caller writes again.
971 if (!translationBuffer.empty() &&
972 !tryWriteImpl(maybePadEmptyMessage(translationBuffer.front()))) {
973 translationBuffer.clear();
974 return false;
975 }
976 translationBufferIdx = 1;
977 flushInternal();
978 return true;
979 }
980 return tryWriteImpl(maybePadEmptyMessage(data));
981}
982
983bool WriteChannelPort::flushInternal() {
984 while (translationBufferIdx < translationBuffer.size()) {
985 if (!tryWriteImpl(
986 maybePadEmptyMessage(translationBuffer[translationBufferIdx])))
987 return false;
988 ++translationBufferIdx;
989 }
990 discardPendingFrames();
991 return true;
992}
993
994void WriteChannelPort::discardPendingFrames() {
995 translationBuffer.clear();
996 translationBufferIdx = 0;
