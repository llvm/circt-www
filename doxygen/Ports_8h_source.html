41/// Shared queue/promise helper for polling-style read APIs.
42///
43/// Producers either fulfill the oldest waiting reader immediately or enqueue
44/// the value for a later `readAsync()`. Consumers either receive buffered data
45/// immediately or install a promise which is fulfilled by the next enqueue.
46template <typename BufferedT>
47class PollingBuffer {
48public:
49 explicit PollingBuffer(uint64_t maxQueued) : maxQueued(maxQueued) {}
50
51 /// Return the current bounded queue size. `0` means unbounded.
52 uint64_t getMaxQueued() {
53 std::scoped_lock<std::mutex> lock(mutex);
54 return maxQueued;
55 }
56
57 /// Update the bounded queue size. `0` means unbounded.
58 void setMaxQueued(uint64_t maxMsgs) {
59 std::scoped_lock<std::mutex> lock(mutex);
60 maxQueued = maxMsgs;
61 }
62
63 /// Try to deliver or enqueue a produced value.
64 ///
65 /// Returns `false` only when the bounded queue is full.
66 bool enqueue(BufferedT &value) {
67 std::scoped_lock<std::mutex> lock(mutex);
68 assert(!(!promiseQueue.empty() && !dataQueue.empty()) &&
69 "Both queues are in use.");
70
71 if (!promiseQueue.empty()) {
72 std::promise<BufferedT> promise = std::move(promiseQueue.front());
73 promiseQueue.pop();
74 promise.set_value(std::move(value));
75 } else {
76 if (dataQueue.size() >= maxQueued && maxQueued != 0)
77 return false;
78 dataQueue.push(std::move(value));
79 }
80 return true;
81 }
82
83 /// Read the next value asynchronously.
84 ///
85 /// If a value is already buffered, the returned future is ready
86 /// immediately. Otherwise the future is backed by an internal promise which
87 /// is fulfilled by a later `enqueue()`.
88 std::future<BufferedT> readAsync() {
89 std::scoped_lock<std::mutex> lock(mutex);
90 assert(!(!promiseQueue.empty() && !dataQueue.empty()) &&
91 "Both queues are in use.");
92
93 if (!dataQueue.empty()) {
94 std::promise<BufferedT> promise;
95 std::future<BufferedT> future = promise.get_future();
96 promise.set_value(std::move(dataQueue.front()));
97 dataQueue.pop();
98 return future;
99 }
100
101 promiseQueue.emplace();
102 return promiseQueue.back().get_future();
103 }
104
105private:
106 std::mutex mutex;
107 std::queue<BufferedT> dataQueue;
108 uint64_t maxQueued;
109 std::queue<std::promise<BufferedT>> promiseQueue;
110};

114/// Unidirectional channels are the basic communication primitive between the
115/// host and accelerator. A 'ChannelPort' is the host side of a channel. It can
116/// be either read or write but not both. At this level, channels are untyped --
117/// just streams of bytes. They are not intended to be used directly by users
118/// but used by higher level APIs which add types.
119class ChannelPort {
120public:
121 ChannelPort(const Type *type);
122 virtual ~ChannelPort() {}

124 struct ConnectOptions {
125 /// The buffer size is optional and should be considered merely a hint.
126 /// Individual implementations use it however they like. The unit is number
127 /// of messages of the port type.
128 std::optional<unsigned> bufferSize = std::nullopt;
129
130 /// If the type of this port is a window, translate the incoming/outgoing
131 /// data into its underlying ('into') type. For 'into' types without lists,
132 /// just re-arranges the data fields from the lowered type to the 'into'
133 /// type.
134 ///
135 /// If this option is false, no translation is done and the data is
136 /// passed through as-is. Same is true for non-windowed types.
137 ///
138 /// For messages with lists, only two types are supported:
139 /// 1) Parallel encoding includes any 'header' data with each frame. Said
140 /// header data is the same across all frames, so this encoding is
141 /// inefficient but is commonly used for on-chip streaming interfaces.
142 /// Each frame contains a 'last' field to indicate the end of the list.
143 /// In cases where 'numItems' is greater than 1, a field named
144 /// '<listField>_size' indicates the number of valid items in that
145 /// frame.
146 /// 2) Serial (bulk transfer) encoding, where a 'header' frame precedes
147 /// the list data frame. Said header frame contains a 'count' field
148 /// indicating the number of items in the list. Importantly, the
149 /// header frame is always re-transmitted after the specified number of
150 /// list items have been sent. If the 'count' field is zero, the end of
151 /// the list has been reached. If it is non-zero, the message has not
152 /// been completely transmitted and reading should continue until a
153 /// 'count' of zero is received.
154 ///
155 /// In both cases, the host-side MessageData contains the complete header
156 /// followed by the list data. In other words, header data is not duplicated
157 /// in the returned message. So for a windowed type with header fields and
158 /// a list of (x,y) coordinates, the host memory layout would be:
159 /// ```
160 /// struct ExampleList {
161 /// uint32_t headerField2; // SystemVerilog ordering
162 /// uint32_t headerField1;
163 /// size_t list_length; // Number list items
164 /// struct { uint16_t y, x; } list_data[];
165 /// }
166 /// ```
167 ///
168 /// In a parallel encoding, each frame's wire format (from hardware) is:
169 /// ```
170 /// struct ExampleListFrame {
171 /// uint8_t list_last; // Non-zero indicates last item in list
172 /// struct { uint16_t y, x; } list_data[numItems]; // SV field ordering
173 /// uint32_t headerField2; // SV struct ordering (reversed)
174 /// uint32_t headerField1;
175 /// }
176 /// ```
177 ///
178 /// Important note: for consistency, preserves SystemVerilog struct field
179 /// ordering! So it's the opposite of C struct ordering.
180 ///
181 /// Implementation status:
182 /// - Only parallel list encoding is supported.
183 /// - Fields must be byte-aligned.
184 ///
185 /// See the CIRCT documentation (or td files) for more details on windowed
186 /// messages.
187 bool translateMessage = true;
188
189 ConnectOptions(std::optional<unsigned> bufferSize = std::nullopt,
190 bool translateMessage = true)
191 : bufferSize(bufferSize), translateMessage(translateMessage) {}
192 };

194 /// Set up a connection to the accelerator.
195 virtual void connect(const ConnectOptions &options = ConnectOptions()) = 0;
196 virtual void disconnect() = 0;
197 virtual bool isConnected() const = 0;

199 /// Poll for incoming data. Returns true if data was read or written into a
200 /// buffer as a result of the poll, though it may return false regardless
201 /// (e.g. a write port in callback mode doesn't report its writes). Calling
202 /// the call back could (will) also happen in that case. Some backends need
203 /// this to be called periodically. In the usual case, this will be called by
204 /// a background thread, but the ESI runtime does not want to assume that the
205 /// host processes use standard threads. If the user wants to provide their
206 /// own threads, they need to call this on each port occasionally. This is
207 /// also called from the 'master' poll method in the Accelerator class.
208 bool poll() {
209 if (isConnected())
210 return pollImpl();
211 return false;
212 }
213
214 /// Get the size of each frame in bytes. For windowed types, this is the
215 /// lowered type's width; otherwise, the port type's width. Transports
216 /// (DMA engines, cosim, etc.) require every message to be at least one
217 /// byte, so void / zero-width port types are reported as 1 byte here; the
218 /// transport pads or strips that placeholder byte transparently.
219 size_t getFrameSizeBytes() const {
220 size_t bytes = translationInfo ? translationInfo->frameBytes
221 : utils::bitsToBytes(type->getBitWidth());
222 return std::max<size_t>(1, bytes);
223 }
224 const Type *getType() const { return type; }

226 /// If this port carries a windowed type, return the original WindowType
227 /// (whose `intoType` is what `getType()` returns). Returns nullptr for
228 /// non-windowed ports.
229 const WindowType *getWindowType() const {
230 return translationInfo ? translationInfo->windowType : nullptr;
231 }

233protected:
234 const Type *type;

241 /// Precompute and optimize the copy operations for translating frames.

244 /// Throw if this window cannot be translated by the runtime translator

246 void requireTranslationSupported() const;

248 /// The window type being translated.
249 const WindowType *windowType;

253 struct CopyOp {
254 /// Offset in the incoming/outgoing frame data.
255 size_t frameOffset;
256 /// Offset in the translation buffer.
257 size_t bufferOffset;

262 /// Information about a list field within a frame (for parallel encoding).
263 /// Note: Currently only numItems == 1 is supported (one list element per
264 /// frame).
265 struct ListFieldInfo {
266 /// Name of the list field.
267 std::string fieldName;
268 /// Offset of the list data array in the frame.
269 size_t dataOffset;
270 /// Size of each list element in bytes.
271 size_t elementSize;
272 /// Offset of the 'last' field in the frame.
273 size_t lastFieldOffset;

275 size_t listLengthBufferOffset;
276 /// Offset in the translation buffer where list data starts.
277 size_t listDataBufferOffset;
278 };

280 /// Information about each frame in the windowed type.
281 struct FrameInfo {
282 /// The total size of a frame in bytes.
283 size_t expectedSize;
284 /// Precomputed copy operations for translating this frame (non-list
285 /// fields).
286 std::vector<CopyOp> copyOps;
287 /// Information about list fields in this frame (parallel encoding).
288 /// Currently only one list field per frame is supported.
289 std::optional<ListFieldInfo> listField;
290 };
291 /// Precomputed information about each frame.
292 std::vector<FrameInfo> frames;
293 /// Size of the 'into' type in bytes (for fixed-size types).
294 /// For types with lists, this is the size of the fixed header portion.
295 size_t intoTypeBytes = 0;
296 /// Number of bytes per wire frame (from the lowered type's bit width).
297 size_t frameBytes = 0;
298 /// True if the window contains a list field (variable-size message).
299 bool hasListField = false;
300 };
301 std::unique_ptr<TranslationInfo> translationInfo;

303 /// Method called by poll() to actually poll the channel if the channel is
304 /// connected.
305 virtual bool pollImpl() { return false; }

308 /// connections.
309 virtual void connectImpl(const ConnectOptions &options) {}

312/// A ChannelPort which sends data to the accelerator. It has two modes: Sync
313/// and Callback, which cannot be used at the same time. The mode is set at
314/// connect() time. To change the mode, disconnect() and then connect() again.
315class WriteChannelPort : public ChannelPort {
316public:
317 using ChannelPort::ChannelPort;
318 /// Stops the default callback mode implementation, if in use, dropping a
319 /// fetched message which hasn't been completely written.
320 ~WriteChannelPort() override;
321
322 /// Callback mode data source. Called whenever the port can accept a message:
323 /// returns the next message to write (ownership is transferred), or nullptr
324 /// if none is available yet. After returning nullptr (or throwing), it isn't
325 /// called again until notifyAsyncDataAvailable() is. Must return
326 /// immediately.
327 using DataFetchCallback =
328 std::function<std::unique_ptr<SegmentedMessageData>()>;
329
330 /// Callback mode error handler. Called with the exception thrown while
331 /// writing a message returned by the DataFetchCallback (which is then
332 /// dropped), or by the DataFetchCallback itself.
333 using WriteErrorCallback = std::function<void(std::exception_ptr)>;
334
335 /// Disconnect the port. In callback mode, drops a fetched message which
336 /// hasn't been completely written. With the default callback mode
337 /// implementation, blocks until another thread writing fetched data
338 /// (including the callbacks it invokes) has stopped. Must not be called
339 /// concurrently with connect() or disconnect() on another thread. Subclasses
340 /// which override this must call it.
341 virtual void disconnect() override;
342 virtual bool isConnected() const override {
343 return mode != Mode::Disconnected;
344 }
345
346 //===--------------------------------------------------------------------===//
347 // Sync mode: connect without a callback to use write(), tryWrite(), and
348 // flush(). These throw in callback mode.
349 //===--------------------------------------------------------------------===//
350
351 /// Connect in sync mode. Throws if connected in callback mode.
352 virtual void connect(const ConnectOptions &options = {}) override;
353
354 /// A very basic blocking write API. Will likely change for performance
355 /// reasons.
356 void write(const MessageData &data) {
357 requireNotCallbackMode("write");
358 writeInternal(data);
359 }
360
361 /// Write a multi-segment message. Takes ownership so the backend can hold
362 /// the message across partial writes / async completions. See
363 /// writeSegmentedImpl().
364 void write(std::unique_ptr<SegmentedMessageData> msg) {
365 requireNotCallbackMode("write");
366 if (!msg)
367 throw std::runtime_error(
368 "WriteChannelPort::write: null SegmentedMessageData");
369 writeSegmentedImpl(std::move(msg));
370 }
371
372 /// A basic non-blocking write API. Returns true if any of the data was
373 /// sent; false means none of it was, and the write should be retried. If
374 /// the data type is a window a 'true' return does not indicate that the
375 /// message has been completely written. The 'flush' method can be used to
376 /// check that the entire buffer has been written. It is invalid for
377 /// backends to always return false (i.e. backends must eventually ensure
378 /// that writes may succeed).
379 bool tryWrite(const MessageData &data) {
380 requireNotCallbackMode("tryWrite");
381 return tryWriteInternal(data);
382 }
383
384 /// Non-blocking write of a multi-segment message. On success, takes
385 /// ownership of `msg` (leaving it null) and returns true. On failure, returns
386 /// false and leaves `msg` untouched so the caller can retry with the same
387 /// message. Same completion semantics as `tryWrite(const MessageData &)`.
388 /// See tryWriteSegmentedImpl().
389 bool tryWrite(std::unique_ptr<SegmentedMessageData> &msg) {
390 requireNotCallbackMode("tryWrite");
391 if (!msg)
392 throw std::runtime_error(
393 "WriteChannelPort::tryWrite: null SegmentedMessageData");
394 return tryWriteSegmentedImpl(msg);
395 }
396
397 /// Flush any buffered data. Returns true if all data was flushed.
398 ///
399 /// If `translateMessages` is false, calling `flush()` will immediately return
400 /// true and perform no action, as there is no buffered data to flush.
401 bool flush() {
402 requireNotCallbackMode("flush");
403 return flushInternal();
404 }
405
406 //===--------------------------------------------------------------------===//
407 // Callback mode: connect with a data fetch callback, which the port calls to
408 // get messages to write whenever it can accept them. connect() and
409 // notifyAsyncDataAvailable() provide a default implementation, which
410 // backends may override (e.g. to fetch messages as DMA buffers free up).
411 //===--------------------------------------------------------------------===//
412
413 /// Connect in callback mode. Throws if already connected, if either callback
414 /// is null, or if frames of a windowed message written in sync mode haven't
415 /// all been sent yet (reconnect in sync mode and flush() them first). Must
416 /// not be called concurrently with connect() or disconnect() on another
417 /// thread. Callbacks may be invoked from any thread, but never
418 /// concurrently with each other, and must return quickly. Exceptions thrown
419 /// by `fetch` are reported to `onError`; those thrown by `onError`
420 /// propagate out of whichever call invoked it (e.g.
421 /// notifyAsyncDataAvailable() or poll(); the accelerator's service thread
422 /// logs them). They may call notifyAsyncDataAvailable() and disconnect() (and
423 /// then connect()), but must not wait on another thread which uses the port.
424 ///
425 /// Default implementation: the port calls `fetch` and writes what it returns
426 /// with tryWriteSegmentedImpl() (so backends which override that for
427 /// scatter-gather support receive messages as-is), until `fetch` returns
428 /// nullptr or the backend backpressures. It does so from
429 /// notifyAsyncDataAvailable() and from poll() (e.g. called by the
430 /// accelerator's service thread via `AcceleratorServiceThread::addPoll()`),
431 /// which also retries the one fetched message the backend didn't accept.
432 /// Each call writes at most `kMaxAsyncWritesPerCall` messages, so that it
433 /// returns even if `fetch` always has data; poll() writes the rest.
434 /// `fetch` is first called by the first of either after connecting. Its
435 /// state is only allocated while it is in use.
436 ///
437 /// Backends which override this must also override
438 /// notifyAsyncDataAvailable(), uphold the callback contracts, and call
439 /// connectCallbackMode().
440 virtual void connect(DataFetchCallback fetch, WriteErrorCallback onError,
441 const ConnectOptions &options = {});
442
443 /// The most messages the default callback mode implementation writes per
444 /// call to notifyAsyncDataAvailable() or poll().
445 static constexpr size_t kMaxAsyncWritesPerCall = 64;
446
447 /// Notify the port that the DataFetchCallback has data again. Throws if the
448 /// port isn't connected in callback mode.
449 ///
450 /// Default implementation: unless another thread is already writing fetched
451 /// data (which then picks the new data up), writes it immediately on the
452 /// calling thread.
453 virtual void notifyAsyncDataAvailable();
454
455protected:
456 /// If the default callback mode implementation is in use, write fetched data
457 /// (see connect()). Then, if the port is still connected (a callback may
458 /// disconnect it), call pollWriteImpl() and return its result.
459 bool pollImpl() final;
460
461 /// Backend-specific polling. Write port backends override this rather than
462 /// pollImpl(). Returns true if any progress was made.
463 virtual bool pollWriteImpl() { return false; }
464
465 /// Indicates the current mode of the channel.
466 enum Mode { Disconnected, Sync, Callback };
467 std::atomic<Mode> mode = Mode::Disconnected;
468
469 /// For backends which override connect(DataFetchCallback, ...): check that
470 /// the port is disconnected, set up message translation, call connectImpl(),
471 /// and enter callback mode, in which the sync write APIs throw.
472 void connectCallbackMode(const ConnectOptions &options);
473
474 /// Pad an empty payload with a single zero byte if the port type has a
475 /// zero-width logical width (e.g. `VoidType`). Transports (DMA engines,
476 /// cosim, etc.) require every message to carry at least one byte.
477 const MessageData &maybePadEmptyMessage(const MessageData &data) {
478 if (!data.getSize() && type && type->getBitWidth() == 0)
479 return transportPad0Bytes;
480 return data;
481 }
482
483 /// Backing storage for `maybePadEmptyMessage` return so that the returned
484 /// reference outlives the call.
485 inline static const MessageData transportPad0Bytes = MessageData({0});
486
487 /// Implementation for write(). Subclasses must implement this.
488 virtual void writeImpl(const MessageData &) = 0;
489
490 /// Implementation for tryWrite(). Subclasses must implement this.
491 virtual bool tryWriteImpl(const MessageData &data) = 0;
492
493 /// Implementation for the segmented write(). `msg` is non-null. The default
494 /// flattens the message (without copying if it is already a MessageData)
495 /// and writes it like write(const MessageData &). Backends override for
496 /// scatter-gather / chunked-DMA support.
497 virtual void writeSegmentedImpl(std::unique_ptr<SegmentedMessageData> msg) {
498 MessageData scratch;
499 writeInternal(msg->asMessageData(scratch));
500 }
501
502 /// Implementation for the segmented tryWrite(), also used by the default
503 /// callback mode implementation. `msg` is non-null. On success, takes
504 /// ownership of it (leaving it null) and returns true; otherwise leaves it
505 /// untouched. The default flattens the message (without copying if it is
506 /// already a MessageData) and writes it like tryWrite(const MessageData &).
507 /// Backends override for scatter-gather / chunked-DMA support.
508 virtual bool
509 tryWriteSegmentedImpl(std::unique_ptr<SegmentedMessageData> &msg) {
510 MessageData scratch;
511 if (!tryWriteInternal(msg->asMessageData(scratch)))
512 return false;
513 msg.reset();
514 return true;
515 }
516
517 /// Break a message into its frames.
518 std::vector<MessageData> getMessageFrames(const MessageData &data);
519
520 /// Whether to translate outgoing data if the port type is a window type. Set
521 /// by the connect() method.
522 bool translateMessages = false;
523 /// If tryWrite cannot write all the messages of a windowed type at once, it
524 /// stores them here and writes them out one by one on subsequent calls.
525 std::vector<MessageData> translationBuffer;
526 /// Index of the next message to write in translationBuffer.
527 size_t translationBufferIdx = 0;
528 /// Translate outgoing data if the port type is a window type. Append the new
529 /// message 'chunks' to translationBuffer.
530 void translateOutgoing(const MessageData &data);
531
532private:
533 /// Throw if the port is connected in callback mode.
534 void requireNotCallbackMode(const char *method) const;
535 /// Set up translation and call connectImpl(). Shared by both modes.
536 void connectCommon(const ConnectOptions &options);
537 /// write(), tryWrite() and flush() without the mode check.
538 void writeInternal(const MessageData &data);
539 bool tryWriteInternal(const MessageData &data);
540 bool flushInternal();
541 /// Whether a windowed message's frames are partially written.
542 bool hasPendingFrames() const {
543 return translationBufferIdx < translationBuffer.size();
544 }
545 /// Drop the frames of a partially written (windowed) message.
546 void discardPendingFrames();
547
548 //===--------------------------------------------------------------------===//
549 // Default callback mode implementation.
550 //===--------------------------------------------------------------------===//
551
552 /// One connection using the default implementation: its state and the logic
553 /// which writes fetched data. Defined in Ports.cpp.
554 struct DefaultAsyncWriter;
555 /// Return the default implementation's current connection, if any.
556 std::shared_ptr<DefaultAsyncWriter> getAsyncWriter();
557 /// Detach and stop the default implementation's current connection, if any.
558 void stopAsyncWriter();
559
560 /// Protects `asyncWriter`.
561 std::mutex asyncWriterMutex;
562 /// The default implementation's current connection, if in use.
563 std::shared_ptr<DefaultAsyncWriter> asyncWriter;
564};

591/// A ChannelPort which reads data from the accelerator. It has two modes:
592/// Callback and Polling which cannot be used at the same time. The mode is set
593/// at connect() time. To change the mode, disconnect() and then connect()
594/// again. When the port is disconnected, it will backpressure hardware.
595class ReadChannelPort : public ChannelPort {
596
597public:
598 /// Primary callback API for raw reads.
599 ///
600 /// The message is passed by owning reference so the callee can consume it,
601 /// move storage out of it, or leave it intact when returning `false` to
602 /// request a retry with the same message object.
603 using ReadCallback =
604 std::function<bool(std::unique_ptr<SegmentedMessageData> &)>;
605
606 /// Compatibility callback API for callers which want flattened message
607 /// bytes instead of the owning segmented message object.
608 using FlatReadCallback = std::function<bool(MessageData)>;
609
610 ReadChannelPort(const Type *type)
611 : ChannelPort(type), mode(Mode::Disconnected) {}
612
613 /// Disconnect the channel. Warning: this method may block until all callbacks
614 /// have returned. Do not call it anywhere which could be in a callback
615 /// control path otherwise deadlock will occur.
616 virtual void disconnect() override;
617 virtual bool isConnected() const override {
618 return mode != Mode::Disconnected;
619 }
620
621 //===--------------------------------------------------------------------===//
622 // Callback mode: To use a callback, connect with a callback function which
623 // will get called with incoming data. This function can be called from any
624 // thread. It shall return true to indicate that the data was consumed. False
625 // if it could not accept the data and should be tried again at some point in
626 // the future. Callback is not allowed to block and needs to execute quickly.
627 //
628 // TODO: Have the callback return something upon which the caller can check,
629 // wait, and be notified.
630 //===--------------------------------------------------------------------===//
631
632 virtual void connect(ReadCallback callback,
633 const ConnectOptions &options = {});
634
635 /// Connect a compatibility callback which receives flattened `MessageData`
636 /// objects. This adapts the primary segmented callback path.
637 virtual void connect(FlatReadCallback callback,
638 const ConnectOptions &options = {});
639
640 //===--------------------------------------------------------------------===//
641 // Polling mode methods: To use futures or blocking reads, connect without any
642 // arguments. You will then be able to use readAsync() or read().
643 //===--------------------------------------------------------------------===//
644
645 /// Default max data queue size set at connect time.
646 static constexpr uint64_t DefaultMaxDataQueueMsgs = 32;
647
648 /// Connect to the channel in polling mode.
649 virtual void connect(const ConnectOptions &options = {}) override;
650
651 /// Asynchronous polling read.
652 ///
653 /// Throws if the port is disconnected or currently connected in callback
654 /// mode.
655 virtual std::future<MessageData> readAsync();
656
657 /// Specify a buffer to read into. Blocking. Basic API, will likely change
658 /// for performance and functionality reasons.
659 virtual void read(MessageData &outData) {
660 std::future<MessageData> f = readAsync();
661 f.wait();
662 outData = std::move(f.get());
663 }
664
665 /// Set maximum number of messages to store in the dataQueue. 0 means no
666 /// limit. This is only used in polling mode and is set to default of 32 upon

668 /// may be (and are) backends which have a very small amount of memory which
