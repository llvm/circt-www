338}
339
340MlirAttribute rtgMapAttrGet(MlirContext ctx, MlirType mapType,
341 intptr_t numEntries, MlirAttribute const *keys,
342 MlirAttribute const *values) {
343 DenseMap<TypedAttr, TypedAttr> entries;
344 for (unsigned i = 0; i < numEntries; ++i) {
345 entries.insert(
346 {cast<TypedAttr>(unwrap(keys[i])), cast<TypedAttr>(unwrap(values[i]))});
347 }
348 return wrap(rtg::MapAttr::get(cast<rtg::MapType>(unwrap(mapType)), &entries));
349}
350
351MlirAttribute rtgMapAttrLookup(MlirAttribute attr, MlirAttribute key) {
352 auto mapAttr = cast<rtg::MapAttr>(unwrap(attr));
353 auto keyAttr = cast<TypedAttr>(unwrap(key));
354
355 auto it = mapAttr.getEntries()->find(keyAttr);
356 if (it == mapAttr.getEntries()->end())
357 return wrap(Attribute()); // Return null attribute if key not found
358
359 return wrap(it->second);
360}
