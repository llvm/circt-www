31/// Analysis that computes resource usage for Synth dialect operations.
32/// This analysis walks module hierarchies and counts resources such as:
33/// - And-inverter graph (AIG)
34/// - Truth tables (LUTs)
35/// - Sequential elements (DFFs)
36class ResourceUsageAnalysis {
37public:
38 ResourceUsageAnalysis(mlir::Operation *moduleOp, mlir::AnalysisManager &am);
39
40 /// Resource usage counts for a set of operations.
41 struct ResourceUsage {
42 ResourceUsage(llvm::StringMap<uint64_t> counts)
43 : counts(std::move(counts)) {}
44 ResourceUsage() = default;
45
46 /// Accumulate resource counts from another ResourceUsage.
47 ResourceUsage &operator+=(const ResourceUsage &other) {
48 for (const auto &count : other.counts)
49 counts[count.getKey()] += count.second;
50 return *this;
51 }
52
53 const auto &getCounts() const { return counts; }
54
55 private:
56 llvm::StringMap<uint64_t> counts;
57 };
58
59 /// Resource usage for a single module, including local and total counts.
60 struct ModuleResourceUsage {
61 ModuleResourceUsage(StringAttr moduleName, ResourceUsage local,
62 ResourceUsage total)
63 : moduleName(moduleName), local(std::move(local)),
64 total(std::move(total)) {}
65
66 StringAttr moduleName;
67 ResourceUsage local; ///< Resources used directly in this module.
68 ResourceUsage total; ///< Resources including all child instances.
69
70 /// Information about a child module instance.
71 struct InstanceResource {
72 StringAttr moduleName, instanceName;
73 ModuleResourceUsage *usage;
74 InstanceResource(StringAttr moduleName, StringAttr instanceName,
75 ModuleResourceUsage *usage)
76 : moduleName(moduleName), instanceName(instanceName), usage(usage) {}
77 };
78
79 SmallVector<InstanceResource> instances;
80 const ResourceUsage &getTotal() const { return total; }
81 const ResourceUsage &getLocal() const { return local; }
82 void emitJSON(raw_ostream &os) const;
83 };
84
85 /// Get resource usage for a module.
86 ModuleResourceUsage *getResourceUsage(igraph::ModuleOpInterface module);
87 ModuleResourceUsage *getResourceUsage(StringAttr moduleName);
88
89 /// Get the instance graph used by this analysis.
90 igraph::InstanceGraph *getInstanceGraph() const { return instanceGraph; }
91
92private:
93 /// Cache of computed resource usage per module.
94 DenseMap<StringAttr, std::unique_ptr<ModuleResourceUsage>> designUsageCache;
95
96 /// Instance graph for module hierarchy traversal.
97 igraph::InstanceGraph *instanceGraph;
98};
