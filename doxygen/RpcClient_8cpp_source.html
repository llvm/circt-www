119 // Per to-server channel credit state. `outstanding` counts frames sent but
120 // not yet reported consumed by the server. Writers block on `cv` while
121 // `outstanding >= maxInFlight` (unless it is 0, i.e. unbounded). The map
122 // is populated once in the constructor and never mutated afterwards, so
123 // lookups need no lock.
124 struct WriteCredit {
125 std::mutex mutex;
126 std::condition_variable cv;
127 uint64_t outstanding = 0;
128 uint64_t maxInFlight = RpcClient::DefaultToServerMaxInFlight;
129 };
130 std::unordered_map<uint64_t, std::unique_ptr<WriteCredit>> writeCredits;
131 std::atomic<bool> connectionLost{false};
132 // Set once the constructor has populated the channel tables. Events from
133 // the server which arrive earlier (e.g. stale credits for a previous
134 // client's frames) are ignored.
135 std::atomic<bool> channelTablesReady{false};
136
137 // Pending control-plane requests, keyed by id.
138 std::mutex pendingMutex;
139 std::atomic<uint64_t> nextRequestId{1};
140 std::unordered_map<uint64_t, std::shared_ptr<std::promise<json>>> pending;
141
142 // Per-channel read state. Connection-time `subscribe` registers; the
143 // ReadChannelConnection destructor unregisters and sends `unsubscribe`.
144 //
145 // Each entry owns its own non-blocking TSQueue: `handleBinaryFrame` only
146 // pushes into the queue (on IX's single per-connection network thread),
147 // and a dedicated transport thread is what actually invokes the
148 // user-supplied callback. That decoupling is what makes the client
149 // cross-channel-deadlock-safe regardless of what the user's callback
150 // does: a callback that returns `false` (rejects the message) only
151 // delays its own channel -- the IX thread is free to keep delivering on
152 // other channels.
153 //
154 // The queue's `notifier` rings the server-wide doorbell
155 // (`Impl::markReady`) so the transport thread wakes up only when there's
156 // work for a specific channel.
157 //
158 // Held by `shared_ptr` so the transport thread can copy the entry under
159 // the lock and outlive a concurrent `unregisterCallback` that erases the
160 // map entry: the cancel flag flips, the transport thread sees it on its
161 // next iteration, and drops anything still in the queue.
162 struct ReadCallbackEntry {
163 RpcClient::ReadCallback callback;
164 std::atomic<bool> canceled{false};
165 uint64_t channelId;
166 utils::TSQueue<MessageData> queue;
167
168 ReadCallbackEntry(uint64_t channelId, std::function<void()> notifier)
169 : channelId(channelId), queue(std::move(notifier)) {}
170 };
171 std::mutex readCallbacksMutex;
172 std::unordered_map<uint64_t, std::shared_ptr<ReadCallbackEntry>>
173 readCallbacks;
174 // Orders sending acknowledgements against canceling a subscription:
175 // `consumed` events are sent under it and leave out canceled entries, and
176 // entries are canceled under it before `unsubscribe` is sent. So an
177 // acknowledgement never follows its subscription's `unsubscribe` (and
178 // perhaps a new `subscribe`) and gets counted against the next one.
179 std::mutex ackMutex;
180
181 // Dirty-set doorbell for the transport thread. Each entry's TSQueue
182 // notifier inserts the channel id into `readyIds` and wakes the consumer.
183 utils::ReadyIdSet<uint64_t> readyIds;
184 std::thread transportThread;
185
186 void transportLoop();
187 /// Report `acks` (subscription -> count) of client-bound frames as
188 /// consumed, leaving out canceled subscriptions.
189 void sendConsumedEvent(
190 const std::unordered_map<std::shared_ptr<ReadCallbackEntry>, uint64_t>
191 &acks);
192
193 // Connection-open synchronization. The WS callback fires on IX's background
194 // thread, so the constructor blocks here until either Open or Error fires.
195 std::mutex openMutex;
196 std::condition_variable openCV;
197 enum class OpenState { Pending, Open, Failed } openState{OpenState::Pending};
198 std::string openError;
199
200 // Most recent unsolicited server-initiated error (`type="error"`), if any.
201 // The server sends one of these before closing the WS to explain why (e.g.
202 // `server_busy`). Recorded so a subsequent Close/Error path can surface it
203 // instead of the much-less-actionable raw WebSocket close reason.
204 std::mutex lastServerErrorMutex;
205 std::string lastServerError;
206
207 std::atomic<bool> disconnecting{false};
208
209 // Cross-thread fault propagation out of the IX network thread. See
210 // FaultStash docs in RpcWire.h.
211 ::esi::cosim::FaultStash faultStash;
212
213 // Declared last so it's destroyed first: its destructor stops IX's threads,
214 // whose callbacks use the members above. That matters when the constructor
215 // throws after starting it, since then `~Impl()` doesn't run.
216 ix::WebSocket ws;
217};

220// ReadChannelConnectionImpl
221//
222// Held by the consumer of `connectClientReceiver`. Destructor unregisters the
223// callback and sends `unsubscribe`. The actual incoming bytes are delivered
224// by the shared websocket callback in RpcClient::Impl.
225//===----------------------------------------------------------------------===//

227namespace {
228class ReadChannelConnectionImpl : public RpcClient::ReadChannelConnection {
229public:
230 ReadChannelConnectionImpl(RpcClient::Impl *impl, uint64_t channelId)
231 : impl(impl), channelId(channelId) {}
232 ~ReadChannelConnectionImpl() override { disconnect(); }
233
234 void disconnect() override {
235 if (disconnected.exchange(true))
236 return;
237 impl->unregisterCallback(channelId);
238 try {
239 impl->unsubscribe(channelId);
240 } catch (const std::exception &e) {
241 // We're tearing down a per-channel reader and can't propagate the
242 // exception (this runs in a destructor). Log it so the failure isn't
243 // silently lost.
244 impl->getLogger().warning(
245 "cosim",
246 std::string("unsubscribe during disconnect failed: ") + e.what());
247 }
248 }
249
250private:
251 RpcClient::Impl *impl;
252 uint64_t channelId;
253 std::atomic<bool> disconnected{false};
254};

261RpcClient::Impl::Impl(Logger &logger, const std::string &hostname,
262 uint16_t port)
263 : host(hostname), logger(logger) {
264 // On Windows, `ix::initNetSystem()` calls `WSAStartup`; bail out cleanly if
265 // it fails rather than letting the WS start path fail with a less
266 // actionable `WSANOTINITIALISED`. No-op on other platforms.
267 if (!ix::initNetSystem())
268 throw std::runtime_error(
269 "RpcClient: ix::initNetSystem() failed (WSAStartup)");
270
271 ws.setUrl("ws://" + host + ":" + std::to_string(port) + "/esi/cosim/v" +
272 std::to_string(::esi::cosim::kProtocolVersion));
273 ws.disablePerMessageDeflate();
274 ws.disableAutomaticReconnection();
275
276 ws.setOnMessageCallback(
277 [this](const ix::WebSocketMessagePtr &msg) { onMessage(msg); });
278
279 ws.start();
280
281 // Wait for the WS to reach Open (or fail).
282 {
283 std::unique_lock<std::mutex> lock(openMutex);
284 openCV.wait(lock, [&] { return openState != OpenState::Pending; });
285 if (openState == OpenState::Failed)
286 throw std::runtime_error("RpcClient: failed to connect to " +
287 ws.getUrl() + ": " + openError);
288 }
289
290 // Send `hello`; the server blocks the response until its manifest is ready.
291 json result = call(
292 "hello", {{"client_protocol_version", ::esi::cosim::kProtocolVersion}});
293 auto verIt = result.find("protocol_version");
294 if (verIt == result.end() || !verIt->is_number_unsigned() ||
295 verIt->get<uint64_t>() != ::esi::cosim::kProtocolVersion)
296 throw std::runtime_error(
297 "RpcClient: server speaks an unsupported cosim protocol version (" +
298 (verIt == result.end() ? std::string("none") : verIt->dump()) +
299 "); expected " + std::to_string(::esi::cosim::kProtocolVersion));
300
301 // Cache manifest blob.
302 if (auto it = result.find("esi_version"); it != result.end())
303 esiVersion = it->get<uint32_t>();
304 if (auto it = result.find("compressed_manifest_b64"); it != result.end()) {
305 // macaron::Base64::Decode returns an empty error string on success and
306 // writes raw bytes into the out string; reinterpret as a byte vector.
307 std::string raw;
308 std::string err = macaron::Base64::Decode(it->get<std::string>(), raw);
309 if (!err.empty())
310 throw std::runtime_error("RpcClient: invalid base64 in manifest blob: " +
311 err);
312 manifest.assign(raw.begin(), raw.end());
313 }
314
315 // Cache channel table.
316 auto channelsIt = result.find("channels");
317 if (channelsIt == result.end() || !channelsIt->is_array())
318 throw std::runtime_error("RpcClient: hello response missing channels");
319 for (const json &c : *channelsIt) {
320 ChannelMeta meta;
321 meta.id = c.at("channel_id").get<uint64_t>();
322 meta.name = c.at("name").get<std::string>();
323 meta.type = c.at("type").get<std::string>();
324 std::string dir = c.at("direction").get<std::string>();
325 meta.direction = dir == "to_server" ? RpcClient::ChannelDirection::ToServer
326 : RpcClient::ChannelDirection::ToClient;
327 if (meta.direction == RpcClient::ChannelDirection::ToServer)
328 writeCredits.emplace(meta.id, std::make_unique<WriteCredit>());
329 channelsByName.emplace(meta.name, meta);
330 channelsById.emplace(meta.id, std::move(meta));
331 }
332 // Publish the channel tables to the IX network thread (see handleEvent).
333 channelTablesReady.store(true, std::memory_order_release);
334
335 // Start the transport thread now that initialisation is complete. It
336 // sleeps on `readyIds`'s internal CV until a per-channel TSQueue push
337 // wakes it.
338 transportThread = std::thread([this] { transportLoop(); });
339}
340
341RpcClient::Impl::~Impl() {
342 disconnecting = true;
343 failAllPending("client shutdown");
344 markConnectionLost();
345 // Mark every per-channel entry canceled so the transport loop drops any
346 // remaining queued frames promptly instead of spinning on the user
347 // callback.
348 {
349 std::lock_guard<std::mutex> lock(readCallbacksMutex);
350 for (auto &[id, entry] : readCallbacks)
351 entry->canceled.store(true);
352 }
353 // Stop IX *first* so no more `handleBinaryFrame` calls can fire and push
354 // into the per-channel queues. ws.stop() joins IX's internal threads
355 // synchronously.
356 ws.stop();
357 // Now retire the transport thread; nothing else is producing for it.
358 if (transportThread.joinable()) {
359 readyIds.requestShutdown();
360 transportThread.join();
361 }
362}

368void RpcClient::Impl::onMessage(const ix::WebSocketMessagePtr &msg) {
369 switch (msg->type) {
370 case ix::WebSocketMessageType::Open: {
371 std::lock_guard<std::mutex> lock(openMutex);
372 openState = OpenState::Open;
373 openCV.notify_all();
374 return;
375 }
376 case ix::WebSocketMessageType::Error: {
377 std::string reason = msg->errorInfo.reason;
378 {
379 std::lock_guard<std::mutex> lock(lastServerErrorMutex);
380 if (!lastServerError.empty())
381 reason = lastServerError;
382 }
383 {
384 std::lock_guard<std::mutex> lock(openMutex);
385 if (openState == OpenState::Pending) {
386 openState = OpenState::Failed;
387 openError = reason;
388 openCV.notify_all();
389 }
390 }
391 failAllPending("websocket error: " + reason);
392 markConnectionLost();
393 return;
394 }
395 case ix::WebSocketMessageType::Close: {
396 std::string reason = msg->closeInfo.reason;
397 {
398 std::lock_guard<std::mutex> lock(lastServerErrorMutex);
399 if (!lastServerError.empty())
400 reason = lastServerError;
401 }
402 // If the server closed us before `Open` settled (e.g. immediate
403 // `server_busy` reject), surface that as an Open failure so the
404 // constructor's wait unblocks with a useful error.
405 {
406 std::lock_guard<std::mutex> lock(openMutex);
407 if (openState == OpenState::Pending) {
408 openState = OpenState::Failed;
409 openError = reason;
410 openCV.notify_all();
411 }
412 }
413 failAllPending("websocket closed: " + reason);
414 markConnectionLost();
415 return;
416 }
417 case ix::WebSocketMessageType::Message:
418 // An exception escaping this callback would kill IX's network thread.
419 // Stash the first one so the next public RpcClient method rethrows it
420 // on the user's thread.
421 try {
422 if (msg->binary)
423 handleBinaryFrame(msg->str);
424 else
425 handleControlFrame(msg->str);
426 } catch (...) {
427 faultStash.record(std::current_exception());
428 }
429 return;
430 default:
431 return;
432 }
433}

435void RpcClient::Impl::handleControlFrame(const std::string &text) {
436 json resp;
437 try {
438 resp = json::parse(text);
439 } catch (const std::exception &e) {
440 // Unparseable text frame: nothing useful we can do.
441 return;
442 }
443 auto typeIt = resp.find("type");
444 if (typeIt == resp.end() || !typeIt->is_string())
445 return;
446 std::string type = typeIt->get<std::string>();
447
448 // Unsolicited server-initiated error. The server sends one of these before
449 // closing the WS (e.g. `server_busy`). Stash the message so the imminent
450 // Close event surfaces this human-readable reason instead of just the raw
451 // WebSocket close reason.
452 if (type == "error") {
453 if (auto errIt = resp.find("error"); errIt != resp.end()) {
454 std::string code = errIt->value("code", std::string("internal"));
455 std::string message = errIt->value("message", std::string());
456 std::lock_guard<std::mutex> lock(lastServerErrorMutex);
457 lastServerError = "server error [" + code + "]: " + message;
458 }
459 return;
460 }
461
462 if (type == "event") {
463 handleEvent(resp);
464 return;
465 }
466
467 if (type != "response") {
468 // Unknown control-frame type. Per the protocol, receivers MUST ignore
469 // unrecognised fields/types so the wire format can evolve additively
470 // under the same `protocol_version`, but warn so genuine misbehavior
471 // doesn't go silently dropped.
472 logger.debug("cosim",
473 "Ignoring control frame with unknown type \"" + type + "\"");
474 return;
475 }
476 auto idIt = resp.find("request_id");
477 if (idIt == resp.end())
478 return;
479 uint64_t requestId = idIt->get<uint64_t>();
480
481 std::shared_ptr<std::promise<json>> promise;
482 {
483 std::lock_guard<std::mutex> lock(pendingMutex);
484 auto it = pending.find(requestId);
485 if (it == pending.end())
486 return;
487 promise = it->second;
488 // Erase under the lock so a concurrent `failAllPending` won't also
489 // try to fulfill the same promise.
490 pending.erase(it);
491 }
492
493 try {
494 if (auto errIt = resp.find("error"); errIt != resp.end()) {
495 std::string code = errIt->value("code", std::string("internal"));
496 std::string message = errIt->value("message", std::string());
497 promise->set_exception(std::make_exception_ptr(
498 std::runtime_error("Server error [" + code + "]: " + message)));
499 } else if (auto resultIt = resp.find("result"); resultIt != resp.end()) {
500 promise->set_value(*resultIt);
501 } else {
502 // Per the wire spec, a `type=response` frame MUST carry either
503 // `result` (success) or `error` (failure). Surface the protocol
504 // violation to the caller instead of pretending it succeeded.
505 promise->set_exception(std::make_exception_ptr(std::runtime_error(
506 "Server response missing both \"result\" and \"error\" fields "
507 "(protocol violation) for request_id=" +
508 std::to_string(requestId))));
509 }
510 } catch (const std::exception &e) {
511 // Promise might be already satisfied if we raced with a shutdown that
512 // already fulfilled it via failAllPending. Benign, but log so unexpected
513 // duplicate responses from the server don't go entirely unnoticed.
514 logger.debug("cosim",
515 std::string("failed to fulfill promise for request_id=") +
516 std::to_string(requestId) + ": " + e.what());
517 }
518}

520void RpcClient::Impl::handleEvent(const json &frame) {
521 std::string event = frame.value("event", std::string());
522 if (event != "consumed") {
523 logger.debug("cosim", "Ignoring unknown server event \"" + event + "\"");
524 return;
525 }
526 if (!channelTablesReady.load(std::memory_order_acquire))
527 return;
528 auto paramsIt = frame.find("params");
529 if (paramsIt == frame.end() || !paramsIt->is_object())
530 return;
531 auto chIt = paramsIt->find("channels");
532 if (chIt == paramsIt->end() || !chIt->is_array())
533 return;
534 for (const json &c : *chIt) {
535 auto idIt = c.find("channel_id");
536 auto countIt = c.find("count");
537 if (idIt == c.end() || !idIt->is_number_unsigned() || countIt == c.end() ||
538 !countIt->is_number_unsigned()) {
539 logger.warning("cosim", "Malformed consumed event entry: " + c.dump());
540 continue;
541 }
542 auto credIt = writeCredits.find(idIt->get<uint64_t>());
543 if (credIt == writeCredits.end())
544 continue;
545 WriteCredit &cred = *credIt->second;
546 uint64_t count = countIt->get<uint64_t>();
547 {
548 std::lock_guard<std::mutex> lock(cred.mutex);
549 // Clamp: credits for frames sent by a previous client are possible.
550 cred.outstanding -= (std::min)(cred.outstanding, count);
551 }
552 cred.cv.notify_all();
553 }
554}
555
556void RpcClient::Impl::markConnectionLost() {
557 connectionLost.store(true);
558 if (!channelTablesReady.load(std::memory_order_acquire))
559 return;
560 for (auto &[id, cred] : writeCredits) {
561 // Take the lock so a writer can't miss the wakeup between checking its
562 // predicate and blocking.
563 { std::lock_guard<std::mutex> lock(cred->mutex); }
564 cred->cv.notify_all();
565 }
566}

568void RpcClient::Impl::handleBinaryFrame(const std::string &data) {
569 uint64_t channelId;
570 const uint8_t *payloadBytes;
571 size_t payloadSize;
572 if (!::esi::cosim::parseDataFrame(data, channelId, payloadBytes, payloadSize))
573 return;
574
575 std::shared_ptr<ReadCallbackEntry> entry;
576 {
577 std::lock_guard<std::mutex> lock(readCallbacksMutex);
578 auto it = readCallbacks.find(channelId);
579 if (it == readCallbacks.end()) {
580 // Per the protocol, the server MUST stop sending frames for a channel
581 // after the unsubscribe-ack. A frame for an unknown channel is
582 // therefore either a server bug or a misrouted message: warn rather
583 // than silently dropping so the issue is visible.
584 logger.warning("cosim", "Dropping inbound frame for channel id " +
585 std::to_string(channelId) +
586 ": no active subscriber");
587 return;
588 }
589 entry = it->second;
590 }
591
592 // Push the payload into the per-channel queue and return immediately;
593 // never block IX's network thread on the user-supplied callback. The
594 // queue's notifier rings the transport doorbell, and the dedicated
595 // transport thread is what actually invokes the callback.
596 entry->queue.push(MessageData(payloadBytes, payloadSize));
597}
598
599void RpcClient::Impl::failAllPending(const std::string &reason) {
600 std::lock_guard<std::mutex> lock(pendingMutex);
601 for (auto &[id, pr] : pending) {
602 try {
603 pr->set_exception(std::make_exception_ptr(std::runtime_error(reason)));
604 } catch (const std::exception &e) {
605 // Already-satisfied promise; benign but log it in case a real bug is
606 // racing with shutdown.
607 logger.debug("cosim", std::string("failAllPending: request_id=") +
608 std::to_string(id) +
609 " already satisfied: " + e.what());
610 }
611 }
612 pending.clear();
613}

619json RpcClient::Impl::call(const std::string &method, json params) {
620 faultStash.check();
621 uint64_t requestId = nextRequestId.fetch_add(1);
622 std::future<json> future;
623 {
624 std::lock_guard<std::mutex> lock(pendingMutex);
625 auto promise = std::make_shared<std::promise<json>>();
626 future = promise->get_future();
627 pending.emplace(requestId, std::move(promise));
628 }
629
630 json req;
631 req["type"] = "request";
632 req["request_id"] = requestId;
633 req["method"] = method;
634 req["params"] = std::move(params);
635 std::string text = req.dump();
636
637 // No send-side mutex needed: IXWebSocket's `sendUtf8Text` is internally
638 // serialized with `sendBinary`, and every cosim frame is a single send
639 // call (no header+body splits).
640 auto info = ws.sendUtf8Text(text);
641 if (!info.success) {
642 std::lock_guard<std::mutex> plock(pendingMutex);
643 pending.erase(requestId);
644 throw std::runtime_error("RpcClient: failed to send " + method);
645 }
646
647 // Bounded wait so callers cannot hang forever on a missing/lost response.
648 // On WS close or error, `failAllPending()` fires and the future becomes
649 // ready immediately. The timeout only matters if the server stays
650 // connected but never replies (server bug, lost control frame).
651 constexpr auto kCallTimeout = std::chrono::seconds(30);
652 if (future.wait_for(kCallTimeout) != std::future_status::ready) {
653 // Drop the pending entry so a late response doesn't fulfill a
654 // destroyed promise (with shared_ptr it's still safe, but it would
655 // otherwise leak the slot forever).
656 std::lock_guard<std::mutex> plock(pendingMutex);
657 pending.erase(requestId);
658 throw std::runtime_error("RpcClient: timed out waiting for response to " +
659 method);
660 }
661 return future.get();
662}

686void RpcClient::Impl::writeToServer(const std::string &channelName,
687 const MessageData &data) {
688 faultStash.check();
689 const ChannelMeta &meta = getToServerChannel(channelName);
690 WriteCredit &cred = *writeCredits.at(meta.id);
691 {
692 std::unique_lock<std::mutex> lock(cred.mutex);
693 cred.cv.wait(lock, [&] {
694 return connectionLost.load() || cred.maxInFlight == 0 ||
695 cred.outstanding < cred.maxInFlight;
696 });
697 if (connectionLost.load())
698 throw std::runtime_error("Connection to cosim server lost while "
699 "writing to channel '" +
700 channelName + "'");
701 ++cred.outstanding;
702 }
703 sendFrame(meta, data);
704}

706bool RpcClient::Impl::tryWriteToServer(const std::string &channelName,
707 const std::vector<MessageData> &frames,
708 bool force) {
709 faultStash.check();
710 const ChannelMeta &meta = getToServerChannel(channelName);
711 WriteCredit &cred = *writeCredits.at(meta.id);
712 {
713 std::lock_guard<std::mutex> lock(cred.mutex);
714 if (connectionLost.load())
715 throw std::runtime_error("Connection to cosim server lost while "
716 "writing to channel '" +
717 channelName + "'");
718 if (!force && cred.maxInFlight != 0 && cred.outstanding >= cred.maxInFlight)
719 return false;
720 cred.outstanding += frames.size();
721 }
722 for (const MessageData &frame : frames)
723 sendFrame(meta, frame);
724 return true;
725}

766//===----------------------------------------------------------------------===//
767// RpcClient::Impl - subscribe / unsubscribe
768//===----------------------------------------------------------------------===//
769
770std::unique_ptr<RpcClient::ReadChannelConnection>
771RpcClient::Impl::connectClientReceiver(const std::string &channelName,
772 RpcClient::ReadCallback callback,
773 unsigned maxInFlight) {
774 faultStash.check();
775 auto it = channelsByName.find(channelName);
776 if (it == channelsByName.end())
777 throw std::runtime_error("Unknown channel '" + channelName + "'");
778 if (it->second.direction != RpcClient::ChannelDirection::ToClient)
779 throw std::runtime_error("Channel '" + channelName +
780 "' is not a to-client channel");
781 uint64_t channelId = it->second.id;
782
783 // Register the callback first so any racing inbound binary frame after the
784 // server's subscribe-ack has somewhere to land. Refuse a second receiver
785 // here rather than letting it replace the first one's callback.
786 auto entry = std::make_shared<ReadCallbackEntry>(
787 channelId, [this, channelId] { readyIds.markReady(channelId); });
788 entry->callback = std::move(callback);
789 {
790 std::lock_guard<std::mutex> lock(readCallbacksMutex);
791 if (!readCallbacks.emplace(channelId, entry).second)
792 throw std::runtime_error("Channel '" + channelName +
793 "' already has a receiver");
794 }
795
796 try {
797 call("subscribe",
798 {{"channel_id", channelId}, {"max_in_flight", maxInFlight}});
799 } catch (...) {
800 std::lock_guard<std::mutex> lock(readCallbacksMutex);
801 auto it = readCallbacks.find(channelId);
802 if (it != readCallbacks.end() && it->second == entry)
803 readCallbacks.erase(it);
804 throw;
805 }
806
807 return std::make_unique<ReadChannelConnectionImpl>(this, channelId);
808}
809
810void RpcClient::Impl::unsubscribe(uint64_t channelId) {
811 if (disconnecting)
812 return;
813 // Skip the RPC if the WS is already closed (server tore down first, network
814 // drop, etc.). The unsubscribe is moot anyway, and otherwise every
815 // per-channel cleanup during teardown would generate a noisy "failed to
816 // send unsubscribe" warning from ReadChannelConnectionImpl::disconnect.
817 if (ws.getReadyState() != ix::ReadyState::Open)
818 return;
819 call("unsubscribe", {{"channel_id", channelId}});
820}
821
822void RpcClient::Impl::unregisterCallback(uint64_t channelId) {
823 std::shared_ptr<ReadCallbackEntry> entry;
824 {
825 std::lock_guard<std::mutex> lock(readCallbacksMutex);
826 auto it = readCallbacks.find(channelId);
827 if (it == readCallbacks.end())
828 return;
829 entry = it->second;
830 readCallbacks.erase(it);
831 }
832 // Flip the cancel flag on the entry that any in-flight transport-loop
833 // iteration already copied out, so it bails out promptly and drops any
834 // remaining queued frames (and their acknowledgements; see `ackMutex`).
835 {
836 std::lock_guard<std::mutex> lock(ackMutex);
837 entry->canceled.store(true);
838 }
839 // Nudge the transport thread so it observes the cancelation promptly and
840 // drains+drops whatever's still queued.
841 readyIds.markReady(channelId);
842}

844//===----------------------------------------------------------------------===//
845// RpcClient::Impl - transport thread (per-channel deadlock isolation)
846//===----------------------------------------------------------------------===//
847
848void RpcClient::Impl::transportLoop() {
849 // Channels whose last delivery attempt left a message at the head of the
850 // queue (user callback returned `false`). We retry them after a short
851 // backoff so other channels still get a chance every wake, and so a
852 // permanently-busy channel doesn't pin the CPU.
853 std::unordered_set<uint64_t> retry;
854
855 while (true) {
856 std::unordered_set<uint64_t> ids;
857 if (!readyIds.waitDrain(
858 ids, retry.empty() ? std::optional<std::chrono::milliseconds>{}
859 : std::chrono::milliseconds(1)))
860 return;
861 ids.insert(retry.begin(), retry.end());
862 retry.clear();
863
864 // Frames consumed this pass, per subscription. Reported to the server in
865 // one batched event at the end of the pass to return their credits.
866 std::unordered_map<std::shared_ptr<ReadCallbackEntry>, uint64_t> acks;
867 for (uint64_t id : ids) {
868 std::shared_ptr<ReadCallbackEntry> entry;
869 {
870 std::lock_guard<std::mutex> lock(readCallbacksMutex);
871 auto it = readCallbacks.find(id);
872 if (it == readCallbacks.end())
873 continue;
874 entry = it->second;
875 }
876
877 if (entry->canceled.load() || disconnecting)
878 continue;
879
880 // Best-effort drain. `TSQueue::pop(callback)` peeks at the front and
881 // only pops if our callback returns true; on `false` we leave the
882 // message at the head and mark the channel for retry.
883 bool stuck = false;
884 while (!stuck && !entry->canceled.load() && !disconnecting) {
885 bool gotOne = false;
886 entry->queue.pop([&](const MessageData &data) {
887 gotOne = true;
888 std::unique_ptr<SegmentedMessageData> msg =
889 std::make_unique<MessageData>(data);
890 if (entry->callback(msg)) {
891 ++acks[entry];
892 return true;
893 }
894 stuck = true;
895 return false;
896 });
897 if (!gotOne)
898 break; // queue empty
899 if (stuck)
900 retry.insert(id);
901 }
902 }
903 if (!acks.empty())
904 sendConsumedEvent(acks);
905 }
906}
907
908void RpcClient::Impl::sendConsumedEvent(
909 const std::unordered_map<std::shared_ptr<ReadCallbackEntry>, uint64_t>
910 &acks) {
911 // Held while sending; see `ackMutex`.
912 std::lock_guard<std::mutex> lock(ackMutex);
913 json channels = json::array();
914 for (const auto &[entry, count] : acks)
915 if (!entry->canceled.load())
916 channels.push_back({{"channel_id", entry->channelId}, {"count", count}});
917 if (channels.empty())
918 return;
919 json event;
920 event["type"] = "event";
921 event["event"] = "consumed";
922 event["params"] = {{"channels", std::move(channels)}};
923 // A failed send means the connection is going away, and with it the
924 // server's flow-control state; nothing to recover.
925 ws.sendUtf8Text(event.dump());
926}
