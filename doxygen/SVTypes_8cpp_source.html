25/// Return the element type of an ArrayType or UnpackedArrayType, or null if the
26/// operand isn't an array.
27Type circt::sv::getAnyHWArrayElementType(Type type) {
28 if (!type)
29 return {};
30 if (auto array = hw::type_dyn_cast<hw::ArrayType>(type))
31 return array.getElementType();
32 if (auto array = hw::type_dyn_cast<hw::UnpackedArrayType>(type))
33 return array.getElementType();
34
35 return {};
36}

38/// Return true if the specified type has a packed layout, i.e. it
39/// represents a contiguous, bit-sliceable vector: basic bit-vectors, or
40/// packed aggregates (hw.array, hw.struct, hw.union) built up entirely out of
41/// packed element types.
42static bool isPackedType(Type type) {
43
44 if (isa<hw::IntType, IntegerType, hw::EnumType>(type))
45 return true;
46
47 if (auto array = dyn_cast<hw::ArrayType>(type))
48 return isPackedType(array.getElementType());
49
50 if (auto t = dyn_cast<hw::StructType>(type))
51 return llvm::all_of(t.getElements(),
52 [](auto f) { return isPackedType(f.type); });
53
54 if (auto t = dyn_cast<hw::UnionType>(type))
55 return llvm::all_of(t.getElements(),
56 [](auto m) { return isPackedType(m.type); });
57
58 return false;
59}

85/// Return the innermost non-unpacked element type.
86static Type stripUnpackedDimensions(Type type) {
87 type = hw::getCanonicalType(type);
88 while (auto uarray = dyn_cast<hw::UnpackedArrayType>(type))
89 type = uarray.getElementType();
90 return type;
91}

93/// Return whether a type is valid as the element type of a NetType.
94bool circt::sv::isValidNetElementType(Type type) {
95 return isPackedType(stripUnpackedDimensions(type));
96}

98LogicalResult NetType::verify(function_ref<InFlightDiagnostic()> emitError,
99 Type elementType) {
100 if (isa_and_present<NetType, VarType>(elementType))
101 return emitError() << "sv.net element type may not be itself an sv.net or "
102 "sv.var handle";
103
104 Type packedElementType = stripUnpackedDimensions(elementType);
105 if (!isPackedType(packedElementType))
106 return emitError()
107 << "sv.net element type must have a packed base type, but got "
108 << packedElementType;
109
110 return success();
111}

125static bool isValidVarElementTypeImpl(Type type) {
126 if (isPackedType(type) || isa<hw::StringType>(type))
127 return true;
128
129 if (auto uarray = dyn_cast<hw::UnpackedArrayType>(type))
130 return isValidVarElementTypeImpl(uarray.getElementType());
131
132 return false;
133}

135/// Return true if the type can be stored in a variable, including
136/// simulation-only and unpacked data types that are not valid net elements.
137bool circt::sv::isValidVarElementType(Type type) {
138 return isValidVarElementTypeImpl(hw::getCanonicalType(type));
139}

141LogicalResult VarType::verify(function_ref<InFlightDiagnostic()> emitError,
142 Type elementType) {

148 "but got "
