37// Allocate 10MB for the uncompressed manifest. This should be plenty.
38constexpr uint32_t MAX_MANIFEST_SIZE = 10 << 20;
39/// Get the compressed manifest, uncompress, and return it.
40std::string SysInfo::getJsonManifest() const {
41 std::vector<uint8_t> compressed = getCompressedManifest();
42 std::vector<Bytef> dst(MAX_MANIFEST_SIZE);
43 uLongf dstSize = MAX_MANIFEST_SIZE;
44 int rc =
45 uncompress(dst.data(), &dstSize, compressed.data(), compressed.size());
46 if (rc != Z_OK)
47 throw std::runtime_error("zlib uncompress failed with rc=" +
48 std::to_string(rc));
49 return std::string(reinterpret_cast<char *>(dst.data()), dstSize);
50}

56MMIO::MMIO(AcceleratorConnection &conn, const AppIDPath &idPath,
57 const HWClientDetails &clients)
58 : Service(conn) {
59 AppIDPath idParent = idPath.parent();
60 for (const HWClientDetail &client : clients) {
61 auto offsetIter = client.implOptions.find("offset");
62 if (offsetIter == client.implOptions.end())
63 throw std::runtime_error("MMIO client missing 'offset' option");
64 const Constant *offset = std::any_cast<Constant>(&offsetIter->second);
65 if (!offset)
66 throw std::runtime_error(
67 "MMIO client 'offset' option must be a constant");
68 const uint64_t *offsetVal = std::any_cast<uint64_t>(&offset->value);
69 if (!offsetVal)
70 throw std::runtime_error(
71 "MMIO client 'offset' option must be an integer");
72 if (*offsetVal >= 1ull << 32)
73 throw std::runtime_error("MMIO client offset mustn't exceed 32 bits");
74
75 auto sizeIter = client.implOptions.find("size");
76 if (sizeIter == client.implOptions.end())
77 throw std::runtime_error("MMIO client missing 'size' option");
78 const Constant *size = std::any_cast<Constant>(&sizeIter->second);
79 if (!size)
80 throw std::runtime_error("MMIO client 'size' option must be a constant");
81 const uint64_t *sizeVal = std::any_cast<uint64_t>(&size->value);
82 if (!sizeVal)
83 throw std::runtime_error("MMIO client 'size' option must be an integer");
84 if (*sizeVal >= 1ull << 32)
85 throw std::runtime_error("MMIO client size mustn't exceed 32 bits");
86 regions[client.relPath] = RegionDescriptor{
87 static_cast<uint32_t>(*offsetVal), static_cast<uint32_t>(*sizeVal)};
88 }
89}

164std::vector<uint8_t> MMIOSysInfo::getCompressedManifest() const {
165 uint64_t version = getEsiVersion();
166 if (version != 0)
167 throw std::runtime_error("Unsupported ESI header version: " +
168 std::to_string(version));
169 uint64_t manifestPtr = mmio->read(MetadataOffset + ManifestPtrOffset);
170 uint64_t size = mmio->read(manifestPtr);
171 uint64_t numWords = (size + 7) / 8;
172 std::vector<uint64_t> manifestWords(numWords);
173 for (size_t i = 0; i < numWords; ++i)
174 manifestWords[i] = mmio->read(manifestPtr + 8 + (i * 8));
175
176 std::vector<uint8_t> manifest;
177 for (size_t i = 0; i < size; ++i) {
178 uint64_t word = manifestWords[i / 8];
179 manifest.push_back(word >> (8 * (i % 8)));
180 }
181 return manifest;
182}

343void CallService::Callback::connect(
344 std::function<MessageData(const MessageData &)> callback, bool quick,
345 const ChannelPort::ConnectOptions &options) {
346 if (channels.size() != 2)
347 throw std::runtime_error("CallService must have exactly two channels");
348 result = &getRawWrite("result");
349 result->connect(options);
350 arg = &getRawRead("arg");
351 if (quick) {
352 // If it's quick, we can just call the callback directly.
353 arg->connect([this, callback](MessageData argMsg) -> bool {
354 MessageData resultMsg = callback(std::move(argMsg));
355 this->result->write(std::move(resultMsg));
356 return true;
357 });
358 } else {
359 // If it's not quick, we need to use the service thread.
360 arg->connect(options);
361 acc.getServiceThread()->addListener(
362 {arg}, [this, callback](ReadChannelPort *, MessageData argMsg) -> void {
363 MessageData resultMsg = callback(std::move(argMsg));
364 this->result->write(std::move(resultMsg));
365 });
366 }
367}

369TelemetryService::TelemetryService(AppIDPath idPath,
370 AcceleratorConnection &conn,
371 ServiceImplDetails details,
372 HWClientDetails clients)
373 : Service(conn), id(idPath), mmio(nullptr) {
374 // Compute our parents idPath path.
375 AppIDPath prefix = std::move(idPath);
376 if (prefix.size() > 0)
377 prefix.pop_back();
378 for (const HWClientDetail &client : clients) {
379 if (client.implOptions.contains("type") &&
380 std::any_cast<std::string>(client.implOptions.at("type")) != "mmio")
381 continue; // Not an MMIO assignment.
382 AppIDPath fullClientPath = prefix + client.relPath;
383 auto offsetIter = client.implOptions.find("offset");
384 if (offsetIter == client.implOptions.end()) {
385 conn.getLogger().warning("Telemetry",
386 "mmio client " + fullClientPath.toStr() +
387 " missing 'offset' option, skipping");
388 continue;
389 }
390 const Constant *offset = std::any_cast<Constant>(&offsetIter->second);
391 if (offset == nullptr) {
392 conn.getLogger().warning(
393 "Telemetry", "mmio client " + fullClientPath.toStr() +
394 " 'offset' option must be a constant, skipping");
395 continue;
396 }
397 const uint64_t *offsetVal = std::any_cast<uint64_t>(&offset->value);
398 if (offsetVal == nullptr) {
399 conn.getLogger().warning(
400 "Telemetry", "mmio client " + fullClientPath.toStr() +
401 " 'offset' option must be an integer, skipping");
402 continue;
403 }
404 portAddressAssignments.emplace(fullClientPath, *offsetVal);
405 }
406}

412MMIO::MMIORegion *TelemetryService::getMMIORegion() const {
413 if (!mmio) {
414 AppIDPath lastPath;
415 AppIDPath mmioPath = id;
416 mmioPath.pop_back();
417 mmioPath.push_back(AppID("__telemetry_mmio"));
418 auto port = conn.getAccelerator().resolvePort(mmioPath, lastPath);
419 if (!port)
420 throw std::runtime_error("TelemetryService: could not resolve port " +
421 id.toStr() + ". Got as far as " +
422 lastPath.toStr());
423 mmio = dynamic_cast<MMIO::MMIORegion *>(port);
424 if (!mmio)
425 throw std::runtime_error("TelemetryService: port " + id.toStr() +
426 " is not a MMIO region");
427 }
428 return mmio;
429}
