38/// Helper function to populate additional legal ops for partial legalization.
39template <typename... AllowedOpTy>
40static void addOpName(SmallVectorImpl<std::string> &ops) {
41 (ops.push_back(AllowedOpTy::getOperationName().str()), ...);
42}
43template <typename... OpToLowerTy>
44static std::unique_ptr<Pass>
45createLowerVariadicPass(bool timingAware, bool reuseSubsets = false) {
46 LowerVariadicOptions options;
47 addOpName<OpToLowerTy...>(options.opNames);
48 options.timingAware = timingAware;
49 options.reuseSubsets = reuseSubsets;
50 return createLowerVariadic(options);
51}
52void circt::synth::buildCombLoweringPipeline(
53 OpPassManager &pm, const CombLoweringPipelineOptions &options) {
54 {
55 if (!options.disableDatapath) {
56 // Lower variadic Mul into a binary op to enable datapath lowering.
57 pm.addPass(createLowerVariadicPass<comb::MulOp>(options.timingAware));
58 pm.addPass(createConvertCombToDatapath());
59 pm.addPass(createSimpleCanonicalizerPass());
60 pm.addPass(createCSEPass());
61 if (options.synthesisStrategy == OptimizationStrategyTiming)
62 pm.addPass(datapath::createDatapathReduceDelay());
63 circt::ConvertDatapathToCombOptions datapathOptions;
64 datapathOptions.timingAware = options.timingAware;
65 datapathOptions.lowerCompress = false;
66 // Incrementally lower - first lower partial product operations
67 pm.addPass(createConvertDatapathToComb(datapathOptions));
68 pm.addPass(createSimpleCanonicalizerPass());
69 datapathOptions.lowerCompress = true;
70 // Then lower compress operations after canononicalization to reduce the
71 // number of compress operations.
72 pm.addPass(createConvertDatapathToComb(datapathOptions));
73 }
74 pm.addPass(createCSEPass());
75 pm.addPass(createSimpleCanonicalizerPass());
76 // Partially legalize Comb, then run CSE and canonicalization.
77 circt::ConvertCombToSynthOptions convOptions;
78 addOpName<comb::AndOp, comb::OrOp, comb::XorOp, comb::MuxOp, comb::ICmpOp,
79 hw::ArrayGetOp, hw::ArraySliceOp, hw::ArrayCreateOp,
80 hw::ArrayConcatOp, hw::AggregateConstantOp>(
81 convOptions.additionalLegalOps);
82 pm.addPass(circt::createConvertCombToSynth(convOptions));
83 }
84 pm.addPass(createCSEPass());
85 pm.addPass(createSimpleCanonicalizerPass());
86 // Balance mux chains. For area oriented flow, we want to keep the mux chains
87 // unless they are very deep.
88 comb::BalanceMuxOptions balanceOptions{OptimizationStrategyTiming ? 16 : 64};
89 pm.addPass(comb::createBalanceMux(balanceOptions));
90
91 // Lower variadic ops before running full lowering to target IR.
92 // For AIG, lower variadic XoR since AIG cannot keep variadic
93 // representation.
94 pm.addPass(createLowerVariadicPass<comb::XorOp>(
95 options.timingAware,
96 options.synthesisStrategy == OptimizationStrategyArea));
97
98 pm.addPass(circt::hw::createHWAggregateToComb());
99 pm.addPass(circt::createConvertCombToSynth());
100 pm.addPass(createCSEPass());
101 pm.addPass(createSimpleCanonicalizerPass());
102 pm.addPass(createCSEPass());
103}

105void circt::synth::buildSynthOptimizationPipeline(
106 OpPassManager &pm, const SynthOptimizationPipelineOptions &options) {
107 // LowerWordToBits may not be scalable for large designs so conditionally
108 // disable it. It's also worth considering keeping word-level representation
109 // for faster synthesis.
110 if (!options.disableWordToBits)
111 pm.addPass(synth::createLowerWordToBits());
112 pm.addPass(createCSEPass());
113 // Run after LowerWordToBits for more precise timing-info & scalability.
114 pm.addPass(createLowerVariadicPass(options.timingAware));
115 pm.addPass(createStructuralHash());
116 pm.addPass(createSimpleCanonicalizerPass());
117 pm.addPass(synth::createMaximumAndCover());
118 pm.addPass(createLowerVariadicPass(options.timingAware));
119 pm.addPass(createStructuralHash());
120
121 // SOP balancing.
122 if (!options.disableSOPBalancing) {
123 SOPBalancingOptions sopOptions;
124 // FIXME: The following is very small compared to the default value of ABC
125 // (6/8) and mockturtle(4/25) due to inefficient implementation of
126 // CutRewriter.
127 sopOptions.maxCutInputSize = 4;
128 sopOptions.maxCutsPerRoot = 4;
129 pm.addPass(synth::createSOPBalancing(sopOptions));
130 pm.addPass(createStructuralHash());
131 }
132
133 if (!options.disableFunctionalReduction && hasIncrementalSATSolverBackend()) {
134 FunctionalReductionOptions functionalReductionOptions;
135 functionalReductionOptions.conflictLimit =
136 options.functionalReductionConflictLimit;
137 pm.addPass(createFunctionalReduction(functionalReductionOptions));
138 }
139
140 if (!options.abcCommands.empty()) {
141 synth::ABCRunnerOptions abcOptions;
142 abcOptions.abcPath = options.abcPath;
143 abcOptions.abcCommands.assign(options.abcCommands.begin(),
144 options.abcCommands.end());
145 abcOptions.continueOnFailure = options.ignoreAbcFailures;
146 pm.addPass(synth::createABCRunner(abcOptions));
147 }
148 // TODO: Add more balancing and rewriting passes.
149}
