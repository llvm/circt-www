29TraceEncoder::TraceEncoder(const ArcRuntimeModelInfo *modelInfo,
30 ArcState *state, unsigned numBuffers,
31 bool debug = false)
32 : numBuffers(numBuffers), modelInfo(modelInfo), debug(debug),
33 activeBuffer(modelInfo->traceInfo->traceBufferCapacity) {
34 assert(numBuffers > 0);
35 assert(!!modelInfo->traceInfo && modelInfo->traceInfo->numTraceTaps > 0);
36 timeStep = 0;
37 simTime = 0;
38 isFinished = false;
39 worker = {};
40 // Put the first buffer in place
41 state->traceBuffer = activeBuffer.getData();
42 // Prime the buffer return stack with additional buffers
43 for (unsigned i = 1; i < numBuffers; ++i)
44 availableBuffers.emplace_back(modelInfo->traceInfo->traceBufferCapacity);
45}

47// Enqueue a trace buffer for processing by the worker thread.
48void TraceEncoder::enqueueBuffer(TraceBuffer &&buffer) {
49 assert(worker.has_value());
50 std::unique_lock<std::mutex> lock(bufferQueueMutex);
51 bufferQueue.emplace(std::move(buffer));
52 lock.unlock();
53 bufferQueueCv.notify_one();
54}

56// Retrieve a buffer from the stack of processed buffers.
57TraceBuffer TraceEncoder::getBuffer() {
58 assert(worker.has_value());
59 std::unique_lock<std::mutex> lock(availableBuffersMutex);
60 availableBuffersCv.wait(lock, [this]() { return !availableBuffers.empty(); });
61 auto buffer = std::move(availableBuffers.back());
62 availableBuffers.pop_back();
63 return buffer;
64}

66void TraceEncoder::run(ArcState *state) {
67 if (worker.has_value() || !initialize(state))
68 return;
69 // Flush the trace buffer which may contain data from the init function
70 state->traceBufferSize = 0;
71 timeStep = 0;
72 activeBuffer.firstStep = 0;
73 activeBuffer.firstSimTime = 0;
74 assert(activeBuffer.stepMarkers.empty());
75 assert(bufferQueue.empty());
76 // Start the worker thread
77 if (debug)
78 std::cout << "[ArcRuntime] Starting trace worker thread." << std::endl;
79 std::thread newWorker([this]() { this->workLoop(); });
80 worker = std::move(newWorker);
81}

83void TraceEncoder::workLoop() {
84 startUpWorker();
85 std::optional<TraceBuffer> work = {};
86 while (true) {
87 {
88 // Take buffer
89 std::unique_lock<std::mutex> lock(bufferQueueMutex);
90 bufferQueueCv.wait(
91 lock, [this]() { return isFinished || !bufferQueue.empty(); });
92
93 // Still work to do?
94 if (bufferQueue.empty() && isFinished)
95 break; // Release bufferQueueMutex
96
97 assert(!bufferQueue.empty());
98 work = std::move(bufferQueue.front());
99 bufferQueue.pop();
100 } // Release bufferQueueMutex
101
102 // Process the taken buffer
103 assert(work->size > 0);
104 assert(work->size <= work->capacity);
105 encode(*work);
106 work->clear();
107
108 {
109 // Return buffer
110 std::lock_guard<std::mutex> lock(availableBuffersMutex);
111 availableBuffers.emplace_back(std::move(*work));
112 } // Release availableBuffersMutex
113 work.reset();
114 availableBuffersCv.notify_one();
115 }
116 windDownWorker();
117}

119void TraceEncoder::step(const ArcState *state) {
120 activeBuffer.assertSentinel();
121 ++timeStep;
122 if (!worker)
123 return;
124 assert(activeBuffer.getData() == state->traceBuffer);
125 simTime = readTimeFromState(state);
126 if (state->traceBufferSize > 0) {
127 // Mark the beginning of the new step in the active trace buffer
128 auto &offsets = activeBuffer.stepMarkers;
129 if (!offsets.empty() && offsets.back().offset == state->traceBufferSize) {
130 // No new data produced since last time: Bump the existing last step.
131 offsets.back().numSteps++;
132 offsets.back().simTime = simTime;
133 } else {
134 // Store the current offset
135 offsets.emplace_back(state->traceBufferSize, simTime);
136 }
137 } else {
138 // Untouched buffer: Adjust the first step value and simulation time.
139 assert(activeBuffer.stepMarkers.empty());
140 activeBuffer.firstStep = timeStep;
141 activeBuffer.firstSimTime = simTime;
142 }
143}

145void TraceEncoder::finish(const ArcState *state) {
146 activeBuffer.assertSentinel();
147 if (worker.has_value()) {
148 // Dispatch the final buffer and request the worker to finish.
149 if (state->traceBufferSize > 0)
150 dispatch(state->traceBufferSize);
151 {
152 std::lock_guard<std::mutex> lock(bufferQueueMutex);
153 isFinished = true;
154 } // Release bufferQueueMutex
155 bufferQueueCv.notify_one();
156 // Wait for the worker to finish.
157 worker->join();
158 worker.reset();
159 if (debug)
160 std::cout << "[ArcRuntime] Trace worker thread finished." << std::endl;
161 }
162 finalize(state);
163}

165uint64_t *TraceEncoder::dispatch(uint32_t oldBufferSize) {
166 activeBuffer.assertSentinel();
167 if (oldBufferSize == 0)
168 impl::fatalError("Trace dispatch called on an empty buffer");
169 if (oldBufferSize > activeBuffer.capacity)
170 impl::fatalError("Trace buffer size exceeds capacity");
171 if (worker.has_value()) {
172 activeBuffer.size = oldBufferSize;
173 enqueueBuffer(std::move(activeBuffer));
174 activeBuffer = getBuffer();
175 activeBuffer.firstStep = timeStep;
176 activeBuffer.firstSimTime = simTime;
177 }
178 return activeBuffer.getData();
179}
