77 class HeaderMMIO(Module):
78 """Construct the ESI header MMIO adhering to the MMIO layout specified in
79 the ChannelMMIO service implementation."""
80
81 clk = Clock()
82 rst = Reset()
83 read = Input(esi.MMIO.read_write.type)
84 # Asserted for one cycle when the host writes the reset magic number to
85 # header slot 7. Propagates up to the BSP which performs the actual reset.
86 reset_request = Output(Bits(1))
87
88 @generator
89 def build(ports):
90 clk = ports.clk
91 rst = ports.rst
92 data_chan_wire = Wire(Channel(esi.MMIODataType))
93 input_bundles = ports.read.unpack(data=data_chan_wire)
94 cmd_chan = input_bundles['cmd']
95
96 # Two-stage half-throughput pipeline: stage 1 captures the incoming
97 # command, stage 2 holds the looked-up response. Each stage carries its
98 # own occupancy bit.
99 cmd_ready = Wire(Bits(1))
100 s1_to_s2_xact = Wire(Bits(1))
101 cmd_raw, cmd_valid = cmd_chan.unwrap(cmd_ready)
102
103 # Stage 1: command capture register and occupancy bit.
104 s1_load = cmd_valid & cmd_ready
105 cmd = cmd_raw.reg(clk, rst, ce=s1_load, name="cmd")
106 s1_valid = ControlReg(clk,
107 rst,
108 asserts=[s1_load],
109 resets=[s1_to_s2_xact],
110 name="s1_valid")
111 # Accept a new command when stage 1 is empty.
112 cmd_ready.assign(~s1_valid)
113
114 address_words = cmd.offset.as_bits()[3:] # Lop off the lower three bits.
115 slot = address_words[:3]
116
117 cycles = Counter(64)(clk=ports.clk,
118 rst=ports.rst,
119 clear=Bits(1)(0),
120 increment=Bits(1)(1),
121 instance_name="cycle_counter")
122
123 # Layout the header as an array.
124 core_freq = System.current().core_freq
125 if core_freq is None:
126 core_freq = 0
127 header = Array(Bits(64), 8)([
128 0, # Generally a good idea to not use address 0.
129 MagicNumber, # ESI magic number.
130 VersionNumber, # ESI version number.
131 manifest_loc, # Absolute address of the manifest ROM.
132 0, # Reserved for future use.
133 cycles.out.as_bits(), # Cycle counter.
134 core_freq, # Core frequency, if known.
135 0, # Slot 7: write the reset magic number here to request a reset.
136 ])
137 header.name = "header"
138
139 # Stage 2: registered response value and its occupancy bit.
140 s2_valid = Wire(Bits(1))
141 data_chan_ready = Wire(Bits(1))
142 s2_xact = s2_valid & data_chan_ready
143 # Stage 1 advances into stage 2 only when stage 2 is empty.
144 s1_to_s2_xact.assign(s1_valid & ~s2_valid)
145
146 header_out = header[slot].reg(clk=clk,
147 rst=rst,
148 ce=s1_to_s2_xact,
149 name="header_out")
150 s2_valid.assign(
151 ControlReg(clk,
152 rst,
153 asserts=[s1_to_s2_xact],
154 resets=[s2_xact],
155 name="header_out_valid"))
156 # Wrap the response.
157 data_chan, data_chan_ready_sig = Channel(esi.MMIODataType).wrap(
158 header_out, s2_valid)
159 data_chan_wire.assign(data_chan)
160 data_chan_ready.assign(data_chan_ready_sig)
161
162 # Detect a write of the reset magic number to slot 7. Register the request
163 # so it is a clean one-cycle pulse, asserted as the command advances into
164 # the response stage. 'DesignResetController' latches it, so a single-cycle
165 # pulse is sufficient to trigger the reset.
166 reset_detect = (cmd.write & (slot == Bits(3)(7)) &
167 (cmd.data == Bits(64)(ResetMagicNumber)))
168 ports.reset_request = reset_detect & s1_to_s2_xact
169
170 return HeaderMMIO

173@modparams
174def ChannelDemuxN_HalfStage_ReadyBlocking(
175 data_type: Type, num_outs: int,
176 next_sel_width: int) -> type["ChannelDemuxNImpl"]:
177 """N-way channel demultiplexer for valid/ready signaling. Contains
178 valid/ready registers on the output channels. The selection signal is now
179 embedded in the input channel payload as a struct {sel, data}. Input
180 signals ready when the selected output register is empty."""
181
182 assert num_outs >= 1, "num_outs must be at least 1."
183
184 class ChannelDemuxNImpl(Module):
185 clk = Clock()
186 rst = Reset()
187
188 # Input channel now carries selection along with data.
189 InPayloadType = StructType([
190 ("sel", Bits(clog2(num_outs))),
191 ("next_sel", Bits(next_sel_width)),
192 ("data", data_type),
193 ])
194 inp = Input(Channel(InPayloadType))
195 OutPayloadType = StructType([
196 ("next_sel", Bits(next_sel_width)),
197 ("data", data_type),
198 ])
199 # Outputs are channels of OutPayloadType, which includes both 'next_sel' and 'data' fields.
200 for i in range(num_outs):
201 locals()[f"output_{i}"] = Output(Channel(OutPayloadType))
202
203 @generator
204 def generate(ports) -> None:
205 # Half-stage demux: one register per output channel. Input is ready
206 # when the currently selected output register is empty (not valid).
207 clk = ports.clk
208 rst = ports.rst
209 sel_width = clog2(num_outs)
210
211 # Unwrap input with backpressure from selected output register.
212 input_ready = Wire(Bits(1), name="input_ready")
213 in_payload, in_valid = ports.inp.unwrap(input_ready)
214 in_sel = in_payload.sel
215 in_next_sel = in_payload.next_sel
216 in_data = in_payload.data
217
218 # Track per-output valid regs and build a purely combinational
219 # expression 'selected_valid_expr' = OR_i((sel==i)&valid_i). Avoid
220 # assigning to a Wire multiple times.
221 valid_regs: List[BitsSignal] = []
222 selected_valid_expr = Bits(1)(0)
223
224 for i in range(num_outs):
225 # Write when input transaction targets this output and output not holding data yet.
226 will_write = Wire(Bits(1), name=f"will_write_{i}")
227 write_cond = (in_valid & input_ready & (in_sel == Bits(sel_width)(i)))
228 will_write.assign(write_cond)
229
230 # Data and next_sel registers.
231 out_msg_reg = ChannelDemuxNImpl.OutPayloadType({
232 "next_sel": in_next_sel,
233 "data": in_data
234 }).reg(clk=clk, rst=rst, ce=will_write, name=f"out{i}_msg_reg")
235
236 # Valid register cleared on successful downstream consume.
237 consume = Wire(Bits(1), name=f"consume_{i}")
238 valid_reg = ControlReg(
239 clk=clk,
240 rst=rst,
241 asserts=[will_write],
242 resets=[consume],
243 name=f"out{i}_valid_reg",
244 )
245 valid_regs.append(valid_reg)
246
247 # Channel wrapper.
248 ch_sig, ch_ready = Channel(ChannelDemuxNImpl.OutPayloadType).wrap(
249 out_msg_reg, valid_reg)
250 setattr(ports, f"output_{i}", ch_sig)
251 consume.assign(valid_reg & ch_ready)
252
253 # Accumulate selected_valid expression.
254 selected_valid_expr = selected_valid_expr | (
255 (in_sel == Bits(sel_width)(i)) & valid_reg)
256
257 # Input ready only when selected output has no valid data latched.
258 input_ready.assign(selected_valid_expr ^ Bits(1)(1))
259
260 def get_out(self, index: int) -> ChannelSignal:
261 return getattr(self, f"output_{index}")
262
263 return ChannelDemuxNImpl

266@modparams
267def ChannelDemuxTree_HalfStage_ReadyBlocking(
268 data_type: Type, num_outs: int,
269 branching_factor_log2: int) -> type["ChannelDemuxTree"]:
270 """Pipelined N-way channel demultiplexer for valid/ready signaling. This
271 implementation uses a tree structure of
272 ChannelDemuxN_HalfStage_ReadyBlocking modules to reduce fanout pressure.
273 Supports maximum half-throughput to save complexity and area.
274 """
275
276 root_sel_width = clog2(num_outs)
277 # Simplify algorithm by making sure num_outs is a power of two.
278 num_outs = 2**root_sel_width
279 sel_width = branching_factor_log2
280 fanout = 2**sel_width
281
282 class ChannelDemuxTree(Module):
283 clk = Clock()
284 rst = Reset()
285 # Input now embeds selection bits alongside data.
286 InPayloadType = StructType([
287 ("sel", Bits(clog2(num_outs))),
288 ("data", data_type),
289 ])
290 inp = Input(Channel(InPayloadType))
291
292 # Outputs (data only).
293 for i in range(num_outs):
294 locals()[f"output_{i}"] = Output(Channel(data_type))
295
296 @generator
297 def build(ports) -> None:
298 assert branching_factor_log2 > 0
299 if num_outs == 1:
300 # Strip selection bits and return single channel.
301 setattr(ports, "output_0", ports.inp.transform(lambda p: p.data))
302 return
303
304 def payload_type(sel_width: int, next_sel_width: int) -> Type:
305 return StructType([
306 ("sel", Bits(sel_width)),
307 ("next_sel", Bits(next_sel_width)),
308 ("data", data_type),
309 ])
310
311 def next_sel_width_calc(curr_sel_width) -> int:
312 return max(curr_sel_width - sel_width, 0)
313
314 def payload_next(curr_msg: StructSignal) -> StructSignal:
315 """Given current level payload, produce next level payload by
316 stripping off the top selection bits."""
317
318 next_sel_width = next_sel_width_calc(curr_msg.next_sel.type.width)
319 curr_sel_width = curr_msg.next_sel.type.width
320 new_sel_width = min(curr_sel_width, sel_width)
321 return payload_type(
322 new_sel_width,
323 next_sel_width,
324 )({
325 # Use the MSB bits of next_sel as the next level selection.
326 "sel": (curr_msg.next_sel[next_sel_width:]
327 if curr_sel_width > 0 else Bits(0)(0)),
328 "next_sel": (curr_msg.next_sel[:next_sel_width]
329 if next_sel_width > 0 else Bits(0)(0)),
330 "data": curr_msg.data,
331 })
332
333 current_channels: List[ChannelSignal] = [
334 ports.inp.transform(lambda m: payload_type(0, root_sel_width)({
335 "sel": Bits(0)(0),
336 "next_sel": m.sel,
337 "data": m.data,
338 }))
339 ]
340
341 curr_sel_width = root_sel_width
342 level = 0
343 while len(current_channels) < num_outs:
344 next_level: List[ChannelSignal] = []
345 level_num_outs = min(2**curr_sel_width, fanout)
346 for i, c in enumerate(current_channels):
347 dmux = ChannelDemuxN_HalfStage_ReadyBlocking(
348 data_type,
349 num_outs=level_num_outs,
350 next_sel_width=next_sel_width_calc(curr_sel_width),
351 )(
352 clk=ports.clk,
353 rst=ports.rst,
354 inp=c.transform(payload_next),
355 instance_name=f"demux_l{level}_i{i}",
356 )
357 for j in range(level_num_outs):
358 next_level.append(dmux.get_out(j))
359 current_channels = next_level
360 curr_sel_width -= sel_width
361 level += 1
362
363 for i in range(num_outs):
364 # Strip off next_sel bits for final output.
365 setattr(
366 ports,
367 f"output_{i}",
368 current_channels[i].transform(lambda p: p.data),
369 )
370
371 def get_out(self, index: int) -> ChannelSignal:
372 return getattr(self, f"output_{index}")
373
374 return ChannelDemuxTree

377def MMIOPrefixRouter(
378 regions: Tuple[Tuple[int, Optional[int]],
379 ...]) -> type["MMIOPrefixRouterImpl"]:
380 """Build a pipelined address-prefix tree which routes MMIO commands.
381
382 Each ``(base, size)`` must describe a disjoint, power-of-two-aligned block.
383 The last region may pass ``size=None`` to claim every address at or above its
384 base; the manifest uses this since its size isn't known during generation.
385 Internal nodes test one address bit and register both outgoing paths. Leaves
386 replace the global address with the block-local low bits, so no subtractor is
387 needed (the open-ended region is the one exception).
388
389 For four 0x100-byte regions at 0x0/0x100/0x200/0x300 plus an open-ended
390 region at 0x400: the open-ended region is peeled off first, then each level
391 splits the remaining candidates in half, so depth is ``O(log N)`` rather than
392 one level per region::
393
394 cmd ──▶ addr[31:10]≠0 ──1──▶ @0x400 (open ended, offset -= 0x400)
395 │
396 0
397 ▼
398 addr[9]
399 ╱ ╲
400 0 1
401 ▼ ▼
402 addr[8] addr[8]
403 ╱ ╲ ╱ ╲
404 0 1 0 1
405 ▼ ▼ ▼ ▼
406 @0x0 @0x100 @0x200 @0x300
407
408 The four leaves all take their client-local offset as ``addr[7:0]``.
409 """
410
411 assert len(regions) > 1, "MMIO routing requires at least two regions"
412
413 # An open-ended final region matches anything with a bit set above the sized
414 # regions, so it needs no size and can never overflow its window.
415 open_ended_base: Optional[int] = None
416 sized_regions = regions
417 if regions[-1][1] is None:
418 open_ended_base = regions[-1][0]
419 assert open_ended_base > 0 and \
420 open_ended_base & (open_ended_base - 1) == 0, \
421 "an open-ended MMIO region must start at a power-of-two address"
422 sized_regions = regions[:-1]
423
424 for base, size in sized_regions:
425 assert size > 0 and size & (size - 1) == 0, \
426 "MMIO region sizes must be powers of two"
427 assert base % size == 0, "MMIO regions must be aligned to their size"
428 assert all(
429 a_base + a_size <= b_base
430 for (a_base, a_size), (b_base, _) in zip(sized_regions, regions[1:])), \
431 "MMIO regions must be sorted and disjoint"
432
433 entries = [(idx, base, size) for idx, (base, size) in enumerate(sized_regions)
434 ]
435
436 def build_tree(node_entries):
437 if len(node_entries) == 1:
438 return node_entries[0][0]
439
440 lowest_fixed_bit = max(size.bit_length() - 1 for _, _, size in node_entries)
441 candidates = []
442 for bit in range(31, lowest_fixed_bit - 1, -1):
443 zeros = [entry for entry in node_entries if not (entry[1] >> bit) & 1]
444 if zeros and len(zeros) != len(node_entries):
445 candidates.append(
446 (abs(len(node_entries) - 2 * len(zeros)), -bit, bit, zeros))
447 assert candidates, "unable to distinguish disjoint MMIO regions"
448 _, _, bit, zeros = min(candidates)
449 zero_indices = {entry[0] for entry in zeros}
450 ones = [entry for entry in node_entries if entry[0] not in zero_indices]
451 return bit, build_tree(zeros), build_tree(ones)
452
453 tree = build_tree(entries)
454
455 class MMIOPrefixRouterImpl(Module):
456 clk = Clock()
457 rst = Reset()
458 inp = Input(Channel(esi.MMIOReadWriteCmdType))
459 for idx in range(len(regions)):
460 locals()[f"output_{idx}"] = Output(Channel(esi.MMIOReadWriteCmdType))
461
462 @generator
463 def build(ports):
464 """Route through one registered address-bit branch per tree level."""
465
466 def new_demux(command_channel: ChannelSignal, select, name: str):
467 Demux = ChannelDemuxN_HalfStage_ReadyBlocking(esi.MMIOReadWriteCmdType,
468 num_outs=2,
469 next_sel_width=0)
470 demux_input = command_channel.transform(
471 lambda cmd, _sel=select, _type=Demux.InPayloadType: _type({
472 "sel": _sel(cmd),
473 "next_sel": Bits(0)(0),
474 "data": cmd
475 }))
476 return Demux(clk=ports.clk,
477 rst=ports.rst,
478 inp=demux_input,
479 instance_name=name)
480
481 def route(node, command_channel: ChannelSignal, path: str):
482 if isinstance(node, int):
483 _, size = regions[node]
484 local_width = size.bit_length() - 1
485
486 def localize(cmd):
487 local_offset = cmd.offset.as_bits()[:local_width].pad_or_truncate(
488 32).as_uint()
489 return esi.MMIOReadWriteCmdType({
490 "write": cmd.write,
491 "offset": local_offset,
492 "data": cmd.data
493 })
494
495 setattr(ports, f"output_{node}", command_channel.transform(localize))
496 return
497
498 bit, zero_node, one_node = node
499 demux = new_demux(command_channel,
500 lambda cmd, _bit=bit: cmd.offset.as_bits()[_bit],
501 f"prefix_{path}_bit{bit}")
502 route(zero_node,
503 demux.get_out(0).transform(lambda p: p.data), path + "0")
504 route(one_node,
505 demux.get_out(1).transform(lambda p: p.data), path + "1")
506
507 client_channel = ports.inp
508 if open_ended_base is not None:
509 # Peel off the open-ended region first: any address bit above the sized
510 # regions selects it. Its base isn't a bit slice away from the local
511 # offset (the region is unbounded), so subtract it explicitly.
512 split_bit = open_ended_base.bit_length() - 1
513 split = new_demux(
514 ports.inp, lambda cmd: cmd.offset.as_bits()[split_bit:].or_reduce(),
515 f"prefix_above_bit{split_bit}")
516
517 def localize_open_ended(payload):
518 cmd = payload.data
519 local_offset = (cmd.offset - UInt(32)(open_ended_base)).as_uint(32)
520 return esi.MMIOReadWriteCmdType({
521 "write": cmd.write,
522 "offset": local_offset,
523 "data": cmd.data
524 })
525
526 setattr(ports, f"output_{len(regions) - 1}",
527 split.get_out(1).transform(localize_open_ended))
528 client_channel = split.get_out(0).transform(lambda p: p.data)
529
530 route(tree, client_channel, "")
531
532 def get_out(self, index: int) -> ChannelSignal:
533 return getattr(self, f"output_{index}")
534
535 return MMIOPrefixRouterImpl

538@modparams
539def DesignResetController(
540 delay_cycles: int) -> type["DesignResetControllerImpl"]:
541 """Counts `delay_cycles` clock cycles after a reset request is observed, then
542 asserts `design_reset` for one cycle. This module must be driven by the
543 *external* reset only (not the reset it generates) so that the countdown is
544 not disturbed by the reset it produces.
545
546 `reset_pending` is asserted from the moment a reset is requested until it
547 fires. It is intended to be used to quiesce the design (e.g. stop accepting
548 new transactions) so that nothing is in flight when the reset is asserted."""
549
550 if delay_cycles < 1:
551 raise ValueError("'delay_cycles' must be at least 1.")
552
553 counter_width = max(clog2(delay_cycles), 1)
554
555 class DesignResetControllerImpl(Module):
556 clk = Clock()
557 rst = Reset()
558 reset_request = Input(Bits(1))
559 design_reset = Output(Bits(1))
560 # High from the cycle a reset is requested until it fires. Use this to stop
561 # accepting new work so in-flight transactions can drain before the reset.
562 reset_pending = Output(Bits(1))
563
564 @generator
565 def build(ports):
566 fire = Wire(Bits(1))
567 # Latch that a reset has been requested until we fire the reset.
568 pending = ControlReg(clk=ports.clk,
569 rst=ports.rst,
570 asserts=[ports.reset_request],
571 resets=[fire],
572 name="reset_pending")
573 # Count cycles while a reset is pending.
574 count = Counter(counter_width)(clk=ports.clk,
575 rst=ports.rst,
576 clear=fire | ~pending,
577 increment=pending,
578 instance_name="reset_delay_counter")
579 fire.assign(pending &
580 (count.out == UInt(counter_width)(delay_cycles - 1)))
581 ports.design_reset = fire
582 ports.reset_pending = pending
583
584 return DesignResetControllerImpl

587class ChannelMMIO(esi.ServiceImplementation):
588 """MMIO service implementation with MMIO bundle interfaces. Should be
589 relatively easy to adapt to physical interfaces by wrapping the wires to
590 channels then bundles. Allows the implementation to be shared and (hopefully)
591 platform independent.
592
593 Whether or not to support unaligned accesses is up to the clients. The header
594 and manifest do not support unaligned accesses and throw away the lower three
595 bits.
596
597 Only allows one outstanding request at a time. This is enforced in hardware
598 by a `MaxOutstandingLimiter` on the command channel, which stalls incoming
599 commands until the previous response has been consumed. If a client fails to
600 return a response, the MMIO service will hang. TODO: add some kind of
601 timeout.
602
603 Implementation-defined MMIO layout:
604 - 0x0: 0 constant
605 - 0x8: Magic number (0x207D98E5_E5100E51)
606 - 0x10: ESI version number (1)
607 - 0x18: Location of the manifest ROM (absolute address)
608
609 - 0x100: Start of MMIO space for requests. Mapping is contained in the
610 manifest so can be dynamically queried.
611
612 - Directly above the last client allocation (rounded up to a power of two):
613 Start of the manifest ROM. It matches any address with a bit set above
614 the client space, so it is never truncated by its window.
615 - addr(Manifest ROM) + 0: Size of compressed manifest
616 - addr(Manifest ROM) + 8: Start of compressed manifest
617
618 This layout _should_ be pretty standard, but different BSPs may have various
619 different restrictions. Any BSP which uses this service implementation will
620 have this layout, possibly with an offset or address window.
621 """
622
623 clk = Clock()
624 rst = Input(Bits(1))
625
626 cmd = Input(esi.MMIO.read_write.type)
627
628 # Asserted for one cycle when the host requests a design reset via an MMIO
629 # write to the header. Propagates up to the BSP which performs the reset.
630 reset_request = Output(Bits(1))
631
632 # Amount of MMIO space a client gets when it doesn't request a size. Must be
633 # at least large enough for the header block, which uses eight 64-bit slots.
634 DefaultRegionSpace = 0x100
635
636 # Start at this address for assigning MMIO addresses to service requests.
637 initial_offset: int = DefaultRegionSpace
638
639 @generator
640 def generate(ports, bundles: esi._ServiceGeneratorBundles):
641 table, manifest_loc = ChannelMMIO.build_table(bundles)

645 @staticmethod
646 def build_table(
647 bundles) -> Tuple[Dict[int, Tuple[Optional[int], AssignableSignal]], int]:
648 """Build a table of read and write addresses to BundleSignals."""
649
650 def align(value: int, alignment: int) -> int:
651 return (value + alignment - 1) // alignment * alignment
652
653 offset = ChannelMMIO.initial_offset
654 table: Dict[int, Tuple[Optional[int], AssignableSignal]] = {}
655 for bundle in bundles.to_client_reqs:
656 requested_size = bundle.options.get("size",
657 ChannelMMIO.DefaultRegionSpace)
658 if isinstance(requested_size,
659 bool) or not isinstance(requested_size, int):
660 raise ValueError("MMIO request option 'size' must be an integer")
661 if requested_size <= 0:
662 raise ValueError("MMIO request option 'size' must be positive")
663 # Round the allocation up to a power of two and align the base to it so
664 # that the address decode is a pure prefix match and the client-local
665 # offset is a bit slice.
666 size = 1 << (max(requested_size, MMIOWordBytes) - 1).bit_length()
667 offset = align(offset, size)
668 next_offset = offset + size
669 if next_offset >= 1 << 32:
670 raise ValueError("MMIO address allocation exceeds the 32-bit space")
671
672 if bundle.port == 'read':
673 table[offset] = size, bundle
674 bundle.add_record(details={
675 "offset": offset,
676 "size": size,
677 "type": "ro"
678 })
679 elif bundle.port == 'read_write':
680 table[offset] = size, bundle
681 bundle.add_record(details={
682 "offset": offset,
683 "size": size,
684 "type": "rw"
685 })
686 else:
687 raise ValueError(f"Unrecognized MMIO port name: {bundle.port}")
688 offset = next_offset
689
690 # The manifest sits directly above the clients and claims every address
691 # with a bit set above them. Its size isn't known until CIRCT builds the
692 # manifest ROM, so leaving the region open-ended keeps it from overflowing
693 # while letting the BAR be only as large as the manifest actually needs.
694 manifest_loc = 1 << max(offset - 1, 0).bit_length()
695 if manifest_loc >= 1 << 32:
696 raise ValueError(
697 "MMIO address allocation leaves no room for the manifest")
698 return table, manifest_loc

766 # half matters, and note it is required by `ChannelMux` too -- at most one

777class TelemetryMMIO(esi.ServiceImplementation):
778 """An ESI service implementation which provides telemetry data through an MMIO
779 region. Each client request is assigned a register in the MMIO space. When a
780 read request is received for the assigned address, it gets routed to the
781 assigned client. When a write request is received, it is discarded. The
782 assignment table is stored in the manifest.
783
784 **REQUIREMENTS.** Both are needed to make the response merge safe:
785
786 1. The `MMIO` service implementation this connects to must not issue a read
787 command while a previous read's response is still outstanding.
788 2. Every telemetry client must assert its `data` channel's `valid` only in
789 response to a `get`. `Telemetry.report_signal` does this; a client which
790 holds `valid` high permanently -- legal ESI, and the natural way to
791 express an always-available counter -- does not.
792
793 Given both, each command is demuxed to exactly one client and at most one
794 client is offering a response at a time, so the responses can be merged with
795 `ChannelMergeOneValid` instead of arbitrated -- which keeps the response path
796 from building a combinational cone across every telemetry client.
797
798 Nothing here enforces either, and violating either loses responses. Note (2)
799 is not specific to this merge: `ChannelMux2` is fixed-priority, so under an
800 arbiter a permanently-valid client starves every client behind it."""
801
802 clk = Clock()
803 rst = Reset()
804

872class MMIOIndirection(Module):
873 """Some platforms do not support MMIO space greater than a certain size (e.g.
874 Vitis 2022's limit is 4k). This module implements a level of indirection to
875 provide access to a full 32-bit address space.
876
877 MMIO addresses:
878 - 0x0: 0 constant
879 - 0x8: 64 bit ESI magic number for Indirect MMIO (0x312bf0cc_E5100E51)
880 - 0x10: Version number for Indirect MMIO (0)
881 - 0x18: Location of read/write in the virtual MMIO space.
882 - 0x20: A read from this location will initiate a read in the virtual MMIO
883 space specified by the address stored in 0x18 and return the result.
884 A write to this location will initiate a write into the virtual MMIO
885 space to the virtual address specified in 0x18.
886 """
887 clk = Clock()
888 rst = Reset()
889
890 upstream = Input(esi.MMIO.read_write.type)
891 downstream = Output(esi.MMIO.read_write.type)
892
893 @generator
894 def build(ports):
895 # This implementation assumes there is only one outstanding upstream MMIO
896 # transaction in flight at once. TODO: enforce this or make it more robust.
897
898 reg_bits = 8
899 location_reg = UInt(reg_bits)(0x18)
900 indirect_mmio_reg = UInt(reg_bits)(0x20)
901 virt_address = Wire(UInt(32))
902
903 # Set up the upstream MMIO interface. Capture last upstream command in a
904 # mailbox which never empties to give access to the last command for all
905 # time.
906 upstream_resp_chan_wire = Wire(Channel(esi.MMIODataType))
907 upstream_cmd_chan = ports.upstream.unpack(
908 data=upstream_resp_chan_wire)["cmd"]
909 _, _, upstream_cmd_data = upstream_cmd_chan.snoop()
910

912 # locally with ones which should be transformed and fowarded downstream.
913 phys_loc = upstream_cmd_data.offset.as_uint(reg_bits)
914 fwd_upstream = NamedWire(phys_loc == indirect_mmio_reg, "fwd_upstream")
915 local_reg_cmd_chan, downstream_cmd_channel = esi.ChannelDemux(
916 upstream_cmd_chan, fwd_upstream, 2, "upstream_demux")

918 # Set up the downstream MMIO interface.
919 downstream_cmd_channel = downstream_cmd_channel.transform(
920 lambda cmd: esi.MMIOReadWriteCmdType({
921 "write": cmd.write,
922 "offset": virt_address,
923 "data": cmd.data
924 }))
925 ports.downstream, froms = esi.MMIO.read_write.type.pack(
926 cmd=downstream_cmd_channel)
927 downstream_data_chan = froms["data"]

929 # Process local regs.
930 (local_reg_cmd_valid, local_reg_cmd_ready,
931 local_reg_cmd) = local_reg_cmd_chan.snoop()

936 name="virt_address",

938 ce=write_virt_address,
939 ))

945 IndirectionVersionNumber, # 0x10
946 virt_address.as_bits(64), # 0x18

952 # create the upstream response.
953 upstream_resp = esi.ChannelMux([local_reg_resp_chan, downstream_data_chan])
954 upstream_resp_chan_wire.assign(upstream_resp)

1007@modparams
1008def ConcatReadGearbox(input_bitwidth: int,
1009 output_bitwidth: int) -> type["ConcatReadGearboxImpl"]:
1010 """Concatenate ``ceil(OUT/IN)`` consecutive engine words into one client
1011 element wider than the word (``OUT > IN``). Serves single-message reads (any
1012 ``OUT > IN``; the low ``OUT`` bits of the concatenation are the element) and
1013 contiguous list reads whose element is a whole number of output_bitwidth

1015 such lists have no partial words and a single element is one flit. Straddling
1016 lists use `ShiftReadGearbox`."""
1017
1018 if input_bitwidth <= 0 or input_bitwidth % 8 != 0:
1019 raise ValueError("engine word width must be a positive multiple of 8 bits")
1020 if output_bitwidth <= input_bitwidth:
1021 raise ValueError("ConcatReadGearbox requires output > input")
1022
1023 in_bytes = input_bitwidth // 8
1024 vb_width = max(1, clog2(in_bytes))
1025
1026 class ConcatReadGearboxImpl(Module):
1027 clk = Clock()
1028 rst = Reset()
1029 in_ = InputChannel(
1030 StructType([
1031 ("tag", esi.HostMem.TagType),
1032 ("data", Bits(input_bitwidth)),
1033 ("valid_bytes", UInt(vb_width)),
1034 ("last", Bits(1)),
1035 ]))
1036 out = OutputChannel(
1037 StructType([
1038 ("tag", esi.HostMem.TagType),
1039 ("data", Bits(output_bitwidth)),
1040 ("last", Bits(1)),
1041 ]))
1042
1043 @generator
1044 def build(ports):
1045 ready_for_upstream = Wire(Bits(1), name="ready_for_upstream")
1046 # Register the input for fmax; the ESI channel buffer keeps the handshake
1047 # elastic.
1048 in_reg = ports.in_.buffer(ports.clk, ports.rst, stages=1)
1049 up, upstream_valid = in_reg.unwrap(ready_for_upstream)
1050 upstream_data = up.data
1051 upstream_last = up.last
1052 upstream_xact = ready_for_upstream & upstream_valid
1053
1054 # Registers accumulate `chunks` upstream words into one client element;
1055 # the output is their concatenation. For a list, elements stream back to
1056 # back and 'last' rides the final word of the burst's final element.
1057 chunks = ceil(output_bitwidth / input_bitwidth)
1058 counter_width = clog2(chunks)
1059 reg_ces = [Wire(Bits(1)) for _ in range(chunks)]
1060 regs = [
1061 upstream_data.reg(ports.clk,
1062 ports.rst,
1063 ce=reg_ces[idx],
1064 name=f"chunk_reg_{idx}") for idx in range(chunks)
1065 ]

1069 # chunk_reg[counter]. 'Counter' clears in preference to incrementing, so
1070 # mask the clear with the accept -- a consume and an accept on the same
1071 # cycle means the accepted word is chunk 0 of the *next* element, so the
1072 # index must land on 1, not 0. 'chunks' need not be a power of two, so
1073 # wrap explicitly rather than relying on the counter's natural rollover.

1079 rst=ports.rst,

1081 (client_xact & ~upstream_xact),
1082 increment=upstream_xact,
1083 instance_name="chunk_counter").out)

1085 [set_client_valid & upstream_xact],
1086 [client_xact])

1089 # 'last' of the final engine word that completes this client flit.
1090 client_last = upstream_last.reg(ports.clk,
1091 ports.rst,
1092 ce=upstream_xact,
1093 name="last_reg")

1095 ports.rst,
1096 ce=upstream_xact,
1097 name="tag_reg")

1099 client_channel, client_ready = ConcatReadGearboxImpl.out.type.wrap(
1100 {
1101 "tag": tag_reg,
1102 "data": client_data_bits,
1103 "last": client_last,
1104 }, client_valid)
1105 client_xact.assign(client_valid & client_ready)
1106 ready_for_upstream.assign(~client_valid | client_ready)
1107 ports.out = client_channel

1109 return ConcatReadGearboxImpl

1116 (``OUT % 8 == 0`` and ``IN % OUT == 0``) from a contiguous list response. Each

1126 raise ValueError(
1127 "DepackReadGearbox requires a byte-aligned element that divides the "
1128 "engine word")

1130 in_bytes = input_bitwidth // 8
1131 # 'valid_bytes' is the real byte count minus 1; a word always has >= 1 byte.
1132 vb_width = max(1, clog2(in_bytes))
1133 count_width = clog2(in_bytes + 1)

1135 elem_bytes = output_bitwidth // 8

1139 rst = Reset()

1141 StructType([
1142 ("tag", esi.HostMem.TagType),
1143 ("data", Bits(input_bitwidth)),
1144 ("valid_bytes", UInt(vb_width)),
1145 ("last", Bits(1)),
1146 ]))

1148 StructType([
1149 ("tag", esi.HostMem.TagType),
1150 ("data", Bits(output_bitwidth)),
1151 ("last", Bits(1)),
1152 ]))

1154 @generator
1155 def build(ports):

1157 up_ready = Wire(Bits(1), name="up_ready")

1159 # elastic and decouples the ready path.
1160 in_reg = ports.in_.buffer(ports.clk, ports.rst, stages=1)
1161 up, up_valid = in_reg.unwrap(up_ready)
1162 client_xact = up_valid & client_ready

1164 if parts == 1:
1165 # One element per word; nothing to select.
1166 last_in_word = Bits(1)(1)
1167 client_data = up.data
1168 else:

1171 clk=ports.clk,
1172 rst=ports.rst,
1173 rst_value=0,

1175 name="idx")

1178 # the biased 'valid_bytes' to recover the real count.

1186 for k in range(parts)
1187 ])
1188 client_data = word_parts[idx]

1190 Mux(last_in_word, (idx + UInt(1)(1)).as_uint(idx_width),
1191 UInt(idx_width)(0)))

1213 Elements are packed at their natural byte stride ``stride = ceil(OUT/8)``
1214 bytes, so element k begins at wire bit ``k*stride*8`` and, in general,
1215 straddles engine-word boundaries at an arbitrary bit offset. A byte-addressed
1216 shift-register accumulator realigns each element across words. This is correct
1217 for every width relationship; `SliceReadGearbox`, `ConcatReadGearbox` and
1218 `DepackReadGearbox` are optimizations that avoid this barrel shifter for the
1219 regular (non-straddling) cases.
1220
1221 Each input word carries ``valid_bytes`` (how many of its bytes are real) and
1222 ``last``. Both are framed to one whole `read_list` request rather than to the
1223 transport: `HostMemReadReqSplitter` drops the per-chunk framing of the reads
1224 it issues and re-derives these from the request's total length, so only the
1225 request's final word is ever partial. That length is ``num_elements *
1226 stride``, so tracking real bytes lets the gearbox emit exactly the right
1227 elements and place the list-terminating ``last`` on the final one -- no
1228 padding element is ever emitted."""
1229
1230 if input_bitwidth % 8 != 0:
1231 raise ValueError("engine word width must be a multiple of 8 bits")
1232 if output_bitwidth <= 0:
1233 raise ValueError("client element width must be positive")
1234 in_bytes = input_bitwidth // 8
1235 stride_bytes = (output_bitwidth + 7) // 8
1236 stride_bits = stride_bytes * 8
1237 # Hold at most one partial element plus one freshly accepted word.
1238 buf_bytes = stride_bytes + in_bytes
1239 buf_bits = buf_bytes * 8
1240 # 'valid_bytes' is the real byte count minus 1; a word always has >= 1 byte.
1241 vb_width = max(1, clog2(in_bytes))
1242 cnt_width = clog2(buf_bytes + 1)
1243 # The append offset is only ever in [0, stride_bytes] (has_room), so the shift
1244 # index needs fewer bits than the full count -- see `build`.
1245 offset_width = clog2(stride_bytes + 1)
1246
1247 class ShiftReadGearboxImpl(Module):
1248 clk = Clock()
1249 rst = Reset()
1250 in_ = InputChannel(
1251 StructType([
1252 ("tag", esi.HostMem.TagType),
1253 ("data", Bits(input_bitwidth)),
1254 ("valid_bytes", UInt(vb_width)),
1255 ("last", Bits(1)),
1256 ]))
1257 out = OutputChannel(
1258 StructType([
1259 ("tag", esi.HostMem.TagType),
1260 ("data", Bits(output_bitwidth)),
1261 ("last", Bits(1)),
1262 ]))
1263
1264 @generator
1265 def build(ports):
1266 client_ready = Wire(Bits(1), name="client_ready")
1267 up_ready = Wire(Bits(1), name="up_ready")
1268 # Register the input for fmax; the ESI channel buffer keeps the handshake
1269 # elastic and decouples the ready path.
1270 in_reg = ports.in_.buffer(ports.clk, ports.rst, stages=1)
1271 up, up_valid = in_reg.unwrap(up_ready)
1272
1273 from pycde.circt.dialects import comb
1274
1275 # Byte-addressed accumulator: `buffer` holds `count` valid bytes packed
1276 # from bit 0 up; the element being emitted is buffer[0:OUT].
1277 buffer = Reg(Bits(buf_bits),
1278 clk=ports.clk,
1279 rst=ports.rst,
1280 rst_value=0,
1281 name="buffer")
1282 count = Reg(UInt(cnt_width),
1283 clk=ports.clk,
1284 rst=ports.rst,
1285 rst_value=0,
1286 name="count")
1287 saw_last = Wire(Bits(1), name="saw_last")
1288
1289 # Accept a whole engine word only when there's room, and never while
1290 # draining a finished burst -- otherwise the next burst's bytes would mix
1291 # into this one's buffer.
1292 has_room = count <= UInt(cnt_width)(buf_bytes - in_bytes)
1293 up_ready.assign(has_room & ~saw_last)
1294 up_xact = up_ready & up_valid
1295
1296 # Emit an element once a full stride slot is buffered.
1297 client_valid = count >= UInt(cnt_width)(stride_bytes)
1298 client_xact = client_valid & client_ready
1299
1300 # The burst's final word sets `saw_last`; the emit that drains the buffer
1301 # to empty terminates the list. These never coincide: emitting needs a
1302 # slot buffered by a prior cycle's accept, so `after_emit == 0` on an
1303 # accept cycle is impossible.
1304 added = Mux(up_xact,
1305 UInt(cnt_width)(0), (up.valid_bytes.as_uint(cnt_width) +
1306 UInt(1)(1)).as_uint(cnt_width))
1307 after_add = (count + added).as_uint(cnt_width)
1308 after_emit = (after_add -
1309 UInt(cnt_width)(stride_bytes)).as_uint(cnt_width)
1310 set_saw_last = (up_xact & up.last).as_bits()
1311 is_final_slot = (after_emit == UInt(cnt_width)(0))
1312 burst_ending = saw_last | set_saw_last
1313 client_last = client_valid & burst_ending & is_final_slot
1314 final_emit = client_xact & client_last
1315
1316 # Append the accepted word at bit offset count*8 (dynamic left shift);
1317 # then, if we emit this cycle, drop the consumed slot (constant right
1318 # shift by the stride). The append offset is <= stride_bytes when
1319 # accepting, so bound the shift index to that range: its high bits are
1320 # constant 0, which lets constant-propagation prune the upper barrel-
1321 # shifter stages (synthesis won't infer this bound from the count reg).
1322 append_off = count.as_bits()[0:offset_width]
1323 shamt = BitsSignal.concat([append_off,
1324 Bits(3)(0)]).pad_or_truncate(buf_bits)
1325 word_ext = up.data.pad_or_truncate(buf_bits)
1326 shifted_word = BitsSignal(
1327 comb.ShlOp(word_ext.value, shamt.value).result, Bits(buf_bits))
1328 appended = buffer | Mux(up_xact, Bits(buf_bits)(0), shifted_word)
1329 drained = appended[stride_bits:buf_bits].pad_or_truncate(buf_bits)
1330 buffer.assign(
1331 Mux(final_emit, Mux(client_xact, appended, drained),
1332 Bits(buf_bits)(0)))
1333
1334 # count += accepted real bytes (biased 'valid_bytes' + 1); -= stride on
1335 # emit.
1336 count.assign(Mux(client_xact, after_add, after_emit))
1337
1338 saw_last.assign(
1339 ControlReg(ports.clk, ports.rst, [set_saw_last], [final_emit]))
1340
1341 tag_reg = up.tag.reg(ports.clk, ports.rst, ce=up_xact, name="tag_reg")
1342 client_channel, client_ready_sig = ShiftReadGearboxImpl.out.type.wrap(
1343 {
1344 "tag": tag_reg,
1345 "data": buffer[0:output_bitwidth],
1346 "last": client_last,
1347 }, client_valid)
1348 client_ready.assign(client_ready_sig)
1349 ports.out = client_channel
1350
1351 return ShiftReadGearboxImpl

1354def select_read_gearbox(is_list: bool, input_bitwidth: int,
1355 output_bitwidth: int):
1356 """Pick the read-gearbox module for a client of the given kind and width
1357 relationship. Every gearbox shares the {tag, data, valid_bytes, last} input
1358 (from `HostMemReadReqSplitter`) and the {tag, data, last} output, so callers
1359 wire them identically. `ShiftReadGearbox` is the correct-for-everything
1360 fallback; the others avoid its barrel shifter for regular relationships."""
1361 if not is_list:
1362 # A single element starts at bit 0 and never straddles at a bit offset.
1363 if output_bitwidth <= input_bitwidth:
1364 return SliceReadGearbox(input_bitwidth, output_bitwidth)
1365 return ConcatReadGearbox(input_bitwidth, output_bitwidth)
1366 if output_bitwidth > input_bitwidth:
1367 # Super-word list: a whole-word-multiple element never straddles.
1368 if output_bitwidth % input_bitwidth == 0:
1369 return ConcatReadGearbox(input_bitwidth, output_bitwidth)
1370 return ShiftReadGearbox(input_bitwidth, output_bitwidth)
1371 # Sub-word list: a byte-aligned element that divides the word never straddles.
1372 if input_bitwidth % output_bitwidth == 0 and output_bitwidth % 8 == 0:
1373 return DepackReadGearbox(input_bitwidth, output_bitwidth)
1374 return ShiftReadGearbox(input_bitwidth, output_bitwidth)

1390@modparams
1391def HostMemReadReqSplitter(req_channel_type: Channel,
1392 resp_channel_type: Channel,
1393 max_chunk_bytes: int,
1394 max_outstanding: int = 1,
1395 round_read_lengths: bool = True):
1396 """Split oversized host memory read requests into request-sized chunks before
1397 arbitration and reassemble the per-chunk responses into a single logical
1398 burst per request.
1399
1400 A burst read (`read_list`) can request many more bytes than a single upstream
1401 read request can carry. This module breaks such a request into
1402 `max_chunk_bytes`-sized (word-aligned) chunks addressed sequentially from the
1403 base. Splitting here -- *before* the requests
1404 are arbitrated onto the shared upstream read channel -- lets each client's
1405 chunks interleave with other clients' requests, so one large burst does not
1406 monopolize host memory bandwidth.
1407
1408 On the response path the per-chunk end-of-list markers are dropped and a
1409 single burst-final `last` (plus the final word's `valid_bytes`) is re-derived
1410 from each logical request's length, so the gearbox and client see one
1411 contiguous response stream per request, identical to an unsplit read.
1412
1413 Up to `max_outstanding` logical requests may be in flight at once. On accept,
1414 each request's {word count, final-word valid bytes} is queued in a metadata
1415 FIFO (depth `max_outstanding`); the response path frames each burst from the
1416 head of that FIFO. This **requires the upstream to return response words in
1417 request order** (no interleaving of different requests' words) -- the
1418 response `tag` is passed through from `resp_in` and is not used for matching.
1419 A new request is accepted whenever the chunk emitter is free (or is issuing
1420 its final chunk this cycle) and fewer than `max_outstanding` requests are in
1421 flight; draining responses does not block accepting. Throughput: a
1422 single-chunk request can be accepted every cycle, chunks are issued one per
1423 cycle, and responses flow one word per cycle with no bubble between bursts.
1424 No latency is added over the single-request version. There is no
1425 combinational path from `resp_in` to `req_in`'s ready; with
1426 `max_outstanding` > 1, `req_out`'s ready feeds `req_in`'s ready.
1427
1428 Zero-length requests are not supported.
1429
1430 req_channel_type: channel of the upstream read request {address, length
1431 (bytes), tag}.
1432 resp_channel_type: channel of the upstream response {tag, data, last}.
1433 max_chunk_bytes: largest per-chunk byte count; must be > 0 and a multiple of
1434 the response word size.
1435 max_outstanding: maximum number of logical requests in flight (accepted
1436 but whose final response word has not yet been delivered). The default of 1
1437 serializes requests.
1438 round_read_lengths: round chunk lengths up to a whole response word for
1439 transports requiring word-granular reads. Disable for byte-granular
1440 transports such as cosim, which zero-pad a partial final response word.
1441 """
1442 assert max_chunk_bytes > 0
1443 assert max_outstanding >= 1
1444
1445 req_struct = req_channel_type.inner_type
1446 resp_struct = resp_channel_type.inner_type
1447 req_fields = dict(req_struct.fields)
1448 addr_width = req_fields["address"].bitwidth
1449 length_width = req_fields["length"].bitwidth
1450 tag_type = req_fields["tag"]
1451 word_bytes = dict(resp_struct.fields)["data"].bitwidth // 8
1452 word_shift = clog2(word_bytes)
1453 words_width = length_width - word_shift
1454 # The response is augmented with a per-word 'valid_bytes': the number of real
1455 # bytes in the (possibly partial) final word, biased by -1. A burst word
1456 # always has >= 1 real byte, so encoding count-1 fits in one fewer bit.
1457 vb_width = max(1, clog2(word_bytes))
1458 resp_fields = dict(resp_struct.fields)
1459 resp_out_struct = StructType([
1460 ("tag", resp_fields["tag"]),
1461 ("data", resp_fields["data"]),
1462 ("valid_bytes", UInt(vb_width)),
1463 ("last", Bits(1)),
1464 ])
1465 resp_out_channel_type = Channel(resp_out_struct)
1466
1467 # Per-request response metadata, computed on accept. 'words_m1' is the
1468 # request's word count minus one, 'final_vb' the valid bytes in its final
1469 # word (biased by -1) and 'single' flags a one-word request.
1470 meta_struct = StructType([
1471 ("words_m1", UInt(words_width)),
1472 ("final_vb", UInt(vb_width)),
1473 ("single", Bits(1)),
1474 ])
1475 inflight_width = clog2(max_outstanding + 1)
1476
1477 class HostMemReadReqSplitterImpl(Module):
1478 clk = Clock()
1479 rst = Reset()
1480 req_in = Input(req_channel_type)
1481 req_out = Output(req_channel_type)
1482 resp_in = Input(resp_channel_type)
1483 resp_out = Output(resp_out_channel_type)
1484
1485 @generator
1486 def build(ports):
1487 clk = ports.clk
1488 rst = ports.rst
1489
1490 # --- Request intake and splitting ---
1491 emit_busy = Wire(Bits(1), name="emit_busy") # issuing chunk requests
1492 cur_addr = Wire(UInt(addr_width), name="cur_addr")
1493 remaining = Wire(UInt(length_width), name="remaining") # req bytes left
1494 # Registered 'remaining <= max_chunk': the chunk being issued is the
1495 # request's last.
1496 last_chunk = Wire(Bits(1), name="last_chunk")
1497 tag_reg = Wire(tag_type, name="tag_reg")
1498 inflight_full = Wire(Bits(1), name="inflight_full")
1499
1500 max_chunk = UInt(length_width)(max_chunk_bytes)
1501 chunk_len = Mux(last_chunk, max_chunk, remaining)
1502
1503 # Round the emitted read length up to a whole word. The reader response
1504 # is word-granular and 'valid_bytes' still carries the real trailing byte
1505 # count, so the reassembled element count is unchanged; this just keeps
1506 # every read word-aligned for single-flit HostMem transports that reject
1507 # sub-word read lengths.
1508 if word_shift == 0 or not round_read_lengths:
1509 chunk_len_out = chunk_len
1510 else:
1511 chunk_words = (chunk_len + UInt(length_width)(word_bytes - 1)
1512 ).as_bits()[word_shift:].as_uint(words_width)
1513 chunk_len_out = BitsSignal.concat(
1514 [chunk_words.as_bits(), Bits(word_shift)(0)]).as_uint(length_width)
1515
1516 req_out_ch, req_out_ready = req_channel_type.wrap(
1517 req_struct({
1518 "address": cur_addr,
1519 "length": chunk_len_out,
1520 "tag": tag_reg,
1521 }), emit_busy)
1522 ports.req_out = req_out_ch
1523 chunk_xact = emit_busy & req_out_ready
1524 emit_done = chunk_xact & last_chunk
1525
1526 # With more than one request allowed in flight, accept the next request
1527 # in the cycle the current one issues its final chunk so single-chunk
1528 # requests stream at one per cycle. (With one, 'inflight_full' blocks
1529 # that case anyway.)
1530 if max_outstanding > 1:
1531 emitter_free = ~emit_busy | emit_done
1532 else:
1533 emitter_free = ~emit_busy
1534 req_ready = Wire(Bits(1))
1535 req_payload, req_valid = ports.req_in.unwrap(req_ready)
1536 req_ready.assign(emitter_free & ~inflight_full)
1537 accept = req_valid & req_ready
1538
1539 emit_busy.assign(
1540 ControlReg(clk, rst, [accept], [emit_done], name="emit_busy_reg"))
1541
1542 # cur_addr / remaining: load on accept; on each non-final chunk advance by
1543 # a full chunk. (After the final chunk they are dead until the next
1544 # accept, so the constant-step update is exact.) Datapath only: no reset.
1545 cur_addr_incr = (cur_addr +
1546 UInt(addr_width)(max_chunk_bytes)).as_uint(addr_width)
1547 cur_addr.assign(
1548 Mux(accept, cur_addr_incr,
1549 req_payload.address).reg(clk,
1550 ce=accept | chunk_xact,
1551 name="cur_addr_reg"))
1552 remaining_dec = (remaining - max_chunk).as_uint(length_width)
1553 remaining.assign(
1554 Mux(accept, remaining_dec,
1555 req_payload.length).reg(clk,
1556 ce=accept | chunk_xact,
1557 name="remaining_reg"))
1558 # The next chunk is the last iff what is left after this one fits in a
1559 # chunk: remaining - max_chunk <= max_chunk.
1560 two_chunks = min(2 * max_chunk_bytes, (1 << length_width) - 1)
1561 last_chunk.assign(
1562 Mux(accept, remaining <= UInt(length_width)(two_chunks),
1563 req_payload.length <= max_chunk).reg(clk,
1564 ce=accept | chunk_xact,
1565 name="last_chunk_reg"))
1566 tag_reg.assign(req_payload.tag.reg(clk, ce=accept, name="tag_reg_r"))
1567
1568 # --- Per-request response metadata. Elements need not tile evenly into
1569 # words: with len_m1 = length - 1, the request spans (len_m1 >> shift) + 1
1570 # words and its final word holds (len_m1 % word_bytes) + 1 bytes. ---
1571 len_m1 = (req_payload.length -
1572 UInt(length_width)(1)).as_uint(length_width)
1573 if word_shift == 0:
1574 words_m1 = len_m1.as_uint(words_width)
1575 final_vb = UInt(vb_width)(0)
1576 else:
1577 words_m1 = len_m1.as_bits()[word_shift:].as_uint(words_width)
1578 final_vb = len_m1.as_bits()[:word_shift].as_uint(vb_width)
1579 meta_in = meta_struct({
1580 "words_m1": words_m1,
1581 "final_vb": final_vb,
1582 "single": req_payload.length <= UInt(length_width)(word_bytes),
1583 })
1584
1585 # Metadata store between accept and the response framer. A request
1586 # accepted while the store is empty and the framer is free bypasses the
1587 # store (see 'bypass' below).
1588 meta_push = Wire(Bits(1), name="meta_push")
1589 meta_pop = Wire(Bits(1), name="meta_pop")
1590 if max_outstanding == 1:
1591 # At most one request in flight: a single holding register suffices.
1592 meta_head = meta_in.reg(clk, ce=meta_push, name="meta_reg")
1593 meta_empty = ~ControlReg(
1594 clk, rst, [meta_push], [meta_pop], name="meta_valid")
1595 else:
1596 meta_fifo = SeqFIFO(meta_struct, max_outstanding, clk, rst)
1597 meta_fifo.push(meta_in, meta_push)
1598 meta_head = meta_fifo.pop(meta_pop)
1599 meta_empty = meta_fifo.empty
1600
1601 # --- Response framing: re-derive the burst-final 'last' and the byte
1602 # count of the (possibly partial) final word. The current burst's
1603 # metadata is held in registers loaded from the metadata store head:
1604 # 'cur_words' counts the words after the current one and 'cur_last'
1605 # flags the current word as the burst's final word. ---
1606 cur_valid = Wire(Bits(1), name="cur_valid")
1607 cur_words = Wire(UInt(words_width), name="cur_words")
1608 cur_last = Wire(Bits(1), name="cur_last")
1609 cur_vb = Wire(UInt(vb_width), name="cur_vb")
1610
1611 # The response handshake is not gated on 'cur_valid': a request's
1612 # metadata is in the framer by the cycle its first chunk is issued, so
1613 # no response word can precede it. Leaving 'resp_in' ready untouched
1614 # also keeps an idle splitter ready, which `esi.TaggedDemux` (whose input
1615 # ready is the AND of every output's ready) relies on.
1616 resp_ready = Wire(Bits(1))
1617 resp_payload, resp_valid = ports.resp_in.unwrap(resp_ready)
1618 resp_out_ch, resp_out_ready = resp_out_channel_type.wrap(
1619 resp_out_struct({
1620 "tag":
1621 resp_payload.tag,
1622 "data":
1623 resp_payload.data,
1624 "valid_bytes":
1625 Mux(cur_last,
1626 UInt(vb_width)(word_bytes - 1), cur_vb),
1627 "last":
1628 cur_last,
1629 }), resp_valid)
1630 ports.resp_out = resp_out_ch
1631 resp_ready.assign(resp_out_ready)
1632 resp_xact = resp_valid & resp_out_ready
1633 burst_done = resp_xact & cur_last
1634
1635 # Load the next burst's metadata when idle or as the current burst ends,
1636 # so back-to-back bursts need no bubble: from the store head if it holds
1637 # anything, otherwise straight from a request accepted this cycle.
1638 framer_free = ~cur_valid | burst_done
1639 load_head = ~meta_empty & framer_free
1640 bypass = accept & meta_empty & framer_free
1641 load = load_head | bypass
1642 meta_pop.assign(load_head)
1643 meta_push.assign(accept & ~bypass)
1644 next_meta = Mux(load_head, meta_in, meta_head)
1645 cur_valid.assign(
1646 ControlReg(clk, rst, [load], [burst_done], name="cur_valid_reg"))
1647 cur_words_dec = (cur_words - UInt(words_width)(1)).as_uint(words_width)
1648 cur_words.assign(
1649 Mux(load, cur_words_dec,
1650 next_meta["words_m1"]).reg(clk,
1651 ce=load | resp_xact,
1652 name="cur_words_reg"))
1653 cur_last.assign(
1654 Mux(load, cur_words == UInt(words_width)(1),
1655 next_meta["single"]).reg(clk,
1656 ce=load | resp_xact,
1657 name="cur_last_reg"))
1658 cur_vb.assign(next_meta["final_vb"].reg(clk, ce=load, name="cur_vb_reg"))
1659
1660 # --- In-flight request accounting: accepted, final word not yet
1661 # delivered. 'inflight_full' is registered so 'req_in' ready has no
1662 # combinational dependence on the response path. ---
1663 inflight = Wire(UInt(inflight_width), name="inflight")
1664 one = UInt(inflight_width)(1)
1665 inflight_inc = (inflight + one).as_uint(inflight_width)
1666 inflight_dec = (inflight - one).as_uint(inflight_width)
1667 inflight_next = Mux(accept, Mux(burst_done, inflight, inflight_dec),
1668 Mux(burst_done, inflight_inc, inflight))
1669 inflight.assign(
1670 inflight_next.reg(clk, rst, rst_value=0, name="inflight_reg"))
1671 inflight_full.assign(
1672 (inflight_next == UInt(inflight_width)(max_outstanding)).reg(
1673 clk, rst, rst_value=0, name="inflight_full_reg"))
1674
1675 return HostMemReadReqSplitterImpl

2000 ready_for_client.assign(~upstream_valid)

2003 [counter.out.as_bits(),
2004 Bits(address_padding_bits)(0)]).as_uint()

2012 name="address_reg")

2029 else:

2032 upstream_channel, upstrm_ready_sig = TaggedWriteGearboxImpl.out.type.wrap(
2033 {
2034 "address": address,
2035 "tag": tag,
2036 "data": upstream_data_bits,
2037 "valid_bytes": valid_bytes,
2038 "last": last,
2039 }, upstream_valid)

2041 ports.out = upstream_channel
