134class CallServiceCallback(Module):
135 """Hardware-initiated call into the host via `CallService`.
136
137 The trigger is an MMIO write at offset ``0x10`` (whose write data forms
138 the ``payload``) so the driver can deterministically time when the
139 callback fires. The callback returns no payload (``Bits(0)`` is the void
140 encoding); the host-side handler increments a counter so the driver can
141 assert it actually ran.
142 """
143
144 clk = Clock()
145 rst = Reset()
146
147 @generator
148 def construct(ports):
149 clk = ports.clk
150 rst = ports.rst
151
152 mmio_bundle = esi.MMIO.read_write(appid=AppID("trigger"))
153 data_resp_chan = Wire(Channel(Bits(64)))
154 cmd_chan = mmio_bundle.unpack(data=data_resp_chan)["cmd"]
155
156 # Snoop the cmd-channel handshake (xact = valid & ready) combinationally
157 # without consuming it. The echo response below drives the actual
158 # handshake; back-pressure on the CallService never reaches the MMIO bus
159 # because the latch overwrites in place.
160 xact, cmd = cmd_chan.snoop_xact()
161
162 # Echo write data back as the read response.
163 data_resp_chan.assign(cmd_chan.transform(lambda c: Bits(64)(c.data)))
164
165 # trigger_xact: a cmd was accepted this cycle AND its offset is 0x10.
166 # Use it both as the clock enable for the payload latch and as the
167 # assert for the callback's valid flag.
168 is_trigger = (cmd.offset == UInt(32)(0x10))
169 trigger_xact = xact & is_trigger
170
171 # A new trigger overwrites any previous-but-unconsumed payload
172 # (one-outstanding semantics, same as a Mailbox).
173 data_reg = cmd.data.as_uint(32).reg(clk, rst, ce=trigger_xact)
174 notify_args = NotifyArgs(tag=UInt(8)(0xA5), payload=data_reg)
175
176 # ControlReg holds the callback's valid bit: assert on trigger_xact,
177 # clear when the consumer takes the message. ControlReg gives asserts
178 # priority on same-cycle ties, so a new trigger landing exactly on the
179 # consumption cycle correctly keeps valid asserted for the next message.
180 cb_consumed = Wire(Bits(1))
181 cb_valid = ControlReg(clk,
182 rst,
183 asserts=[trigger_xact],
184 resets=[cb_consumed])
185 cb_chan, cb_ready = Channel(NotifyArgs).wrap(notify_args, cb_valid)
186 cb_consumed.assign(cb_valid & cb_ready)
187 esi.CallService.call(AppID("callback"), cb_chan, Bits(0))

195class TypedReadChannelStruct(Module):
196 """To-host channel of `EventStruct`: TypedReadPort polling of a struct.
197
198 Hardware pushes a small bounded sequence of distinct events on reset
199 release so the driver can read N items and check exact values. Each event
200 has ``ts = i+1`` and ``val = -(i+1)`` so an off-by-one or sign bug shows
201 up immediately.
202 """
203
204 clk = Clock()
205 rst = Reset()
206
207 num_events = Constant(UInt(8), 4)
208
209 @generator
210 def construct(ports):
211 clk = ports.clk
212 rst = ports.rst
213
214 # One counter that advances exactly when the to-host channel handshakes
215 # (valid & ready). The valid signal is ``count < num_events`` so the
216 # stream goes silent after num_events have been delivered.
217 ready_wire = Wire(Bits(1))
218 increment_wire = Wire(Bits(1))
219 counter = Counter(8)(clk=clk,
220 rst=rst,
221 clear=Bits(1)(0),
222 increment=increment_wire,
223 instance_name="event_counter")
224 valid = counter.out < TypedReadChannelStruct.num_events.value
225 increment_wire.assign(valid & ready_wire)
226
227 one_based = (counter.out + UInt(8)(1)).as_uint(8)
228 ts = one_based.as_uint(64)
229 # val = -ts as si32 -- small enough to fit and reveals sign-extension
230 # bugs on the host side.
231 one_based_u32 = one_based.as_uint(32)
232 neg = (UInt(32)(0) - one_based_u32).as_sint(32)
233 event = EventStruct(ts=ts, val=neg)
234 out_chan, out_ready = Channel(EventStruct).wrap(event, valid)
235 ready_wire.assign(out_ready)
236 esi.ChannelService.to_host(AppID("data"), out_chan)

604class CallbackWindowedList(Module):
605 """HW-initiated callback whose argument is a windowed list with header.
606
607 Combines the callback pattern (``CallService.call``) with the serial-burst
608 windowed list payload. An MMIO write at offset ``0x10`` arms one burst;
609 the HW then sends the same ``{tag=0xCAFE, items=[10,20,30,40]}`` pattern
610 used by the channel probes into the host callback. The host handler
611 verifies the payload and the callback returns void (``Bits(0)``).
612 """
613
614 clk = Clock()
615 rst = Reset()
616
617 @generator
618 def construct(ports):
619 clk = ports.clk
620 rst = ports.rst
621
622 # MMIO trigger: write to offset 0x10 arms one burst.
623 trigger_bundle = esi.MMIO.read_write(appid=AppID("trigger"))
624 resp_chan = Wire(Channel(Bits(64)))
625 cmd_chan = trigger_bundle.unpack(data=resp_chan)["cmd"]
626 cmd_xact, cmd = cmd_chan.snoop_xact()
627 resp_chan.assign(cmd_chan.transform(lambda c: Bits(64)(c.data)))
628 trigger_xact = cmd_xact & (cmd.offset == UInt(32)(0x10))
629
630 n_items = len(_WINDOW_PROBE_ITEMS)
631 burst_end = Wire(Bits(1))
632
633 armed = ControlReg(clk, rst, asserts=[trigger_xact], resets=[burst_end])
634
635 par_ready = Wire(Bits(1))
636 handshake = armed & par_ready
637 idx_counter = Counter(2)(clk=clk,
638 rst=rst,
639 clear=burst_end,
640 increment=handshake,
641 instance_name="cb_window_idx")
642 idx = idx_counter.out
643 last_bit = (idx == UInt(2)(n_items - 1))
644 burst_end.assign(handshake & last_bit)
645
646 item_bits = Array(
647 Bits(32), len(_WINDOW_PROBE_ITEMS))(_WINDOW_PROBE_ITEMS)[idx.as_bits()]
648
649 parallel_window_type = Window.default_of(_window_probe_struct)
650 par_struct = parallel_window_type.lowered_type({
651 "tag": Bits(16)(_WINDOW_PROBE_TAG),
652 "items": item_bits,
653 "last": last_bit,
654 })
655 par_window = parallel_window_type.wrap(par_struct)
656 parallel_chan, parallel_ready = Channel(parallel_window_type).wrap(
657 par_window, armed)
658 par_ready.assign(parallel_ready)
659
660 p2s = ListWindowToSerial(parallel_window_type, _WINDOW_PROBE_BULK_WIDTH,
661 _WINDOW_PROBE_ITEMS_PER_FRAME,
