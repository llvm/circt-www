120/// A contiguous, non-owning view of bytes within a SegmentedMessageData.
121/// Valid only while the owning SegmentedMessageData is alive.
122struct Segment {
123 const uint8_t *data;
124 size_t size;
125 /// The HostMem region (from services::HostMem::allocate()) which holds
126 /// [data, data + size), if any. Several segments may share a region.
127 /// Non-owning: the SegmentedMessageData keeps the region alive; backends may
128 /// be able to obtain ownership via SegmentedMessageData::take().
129 services::HostMemRegion *region = nullptr;
130
131 std::span<const uint8_t> span() const { return {data, size}; }
132 bool empty() const { return size == 0; }
133
134 /// Device address of this segment's bytes, if `region` is set and contains
135 /// them; std::nullopt otherwise.
136 std::optional<uint64_t> getDeviceAddress() const;
137};

139/// Abstract multi-segment message. Generated types subclass this to expose
140/// header + list segments without flattening into a contiguous buffer.
141///
142/// MessageData is the canonical flat, one-segment implementation of this
143/// interface. Other subclasses represent naturally segmented layouts.
144///
145/// Subclasses MUST own all data that their segments point to. Read and write
146/// APIs can hold the message across async boundaries / retries. Subclasses
147/// which hold data in HostMem regions must keep those regions alive for the
148/// lifetime of the message (e.g. by owning them as unique_ptr<HostMemRegion>)
149/// and point their segments' `region` at them. Subclasses which own their
150/// regions exclusively may override take() to hand them to the backend.
151class SegmentedMessageData {
152public:
153 virtual ~SegmentedMessageData() = default;
154
155 /// Number of segments in the message.
156 virtual size_t numSegments() const = 0;
157 /// Get a segment by index. Throws if the segment has been take()n.
158 virtual Segment segment(size_t idx) const = 0;
159
160 /// Optionally called by a backend once it has transmitted segment `segIdx`,
161 /// to obtain ownership of the HostMem region backing it (e.g. to re-use the
162 /// region or return it to a pool).
163 ///
164 /// Returns nullptr, leaving the segment accessible, if the segment isn't
165 /// backed by a region or the message doesn't transfer region ownership.
166 /// Otherwise, marks the segment as taken -- after which segment(segIdx) (and
167 /// thus anything which reads the whole message, such as toMessageData()) and
168 /// further calls to take(segIdx) throw -- and returns the region if no other
169 /// untaken segment references it, or nullptr if some still do. Since several
170 /// segments may share a region, a backend which calls take() on each segment
171 /// as it is transmitted receives each region exactly once, after the last
172 /// segment using it.
173 ///
174 /// Since SegmentedMessageDataCursor reads segments via segment(), a backend
175 /// must not take a segment which a cursor over the message can still reach,
176 /// and must not take a segment it may need to retransmit. Any cursor
177 /// (including one created after the take()) throws when it reaches a taken
178 /// segment. Segment copies of a taken segment obtained before the take() are
179 /// invalid: their `data` may point into a region which has been re-used.
180 ///
181 /// The default implementation returns nullptr.
182 virtual std::unique_ptr<services::HostMemRegion> take(size_t segIdx);
183
184 /// Total size across all segments.
185 size_t totalSize() const;
186 /// True if totalSize() == 0.
187 bool empty() const;
188
189 /// Flatten all segments into a standard MessageData.
190 virtual MessageData toMessageData() const;
191};

193/// A concrete flat message backed by a single vector of bytes.
194///
195/// This is also a one-segment SegmentedMessageData so flat and segmented
196/// messages can flow through the same internal APIs.
197class MessageData : public SegmentedMessageData {
198public:
199 /// Adopts the data vector buffer.
200 MessageData() = default;
201 MessageData(std::span<const uint8_t> data)
202 : data(data.data(), data.data() + data.size()) {}
203 MessageData(std::vector<uint8_t> &data) : data(std::move(data)) {}
204 MessageData(std::vector<uint8_t> &&data) : data(std::move(data)) {}
205 MessageData(const uint8_t *data, size_t size) : data(data, data + size) {}
206 ~MessageData() = default;
207
208 const uint8_t *getBytes() const { return data.data(); }
209
210 /// Get the data as a vector of bytes.
211 const std::vector<uint8_t> &getData() const { return data; }
212
213 /// Implicit conversion to a vector/span of bytes, to play nice with other
214 /// APIs that accept bytearray-like things.
215 operator const std::vector<uint8_t> &() const { return data; }
216 operator std::span<const uint8_t>() const { return data; }
217
218 /// Move the data out of this object.
219 std::vector<uint8_t> takeData() { return std::move(data); }
220
221 /// Get the size of the data in bytes.
222 size_t getSize() const { return data.size(); }
223 size_t size() const { return getSize(); }
224
225 /// Returns true if this message contains no data.
226 bool empty() const { return data.empty(); }
227
228 /// Cast to a type. Throws if the size of the data does not match the size of
229 /// the message. The lifetime of the resulting pointer is tied to the lifetime
230 /// of this object.
231 template <typename T>
232 const T *as() const {
233 if (data.size() != sizeof(T))
234 throw std::runtime_error("Data size does not match type size. Size is " +
235 std::to_string(data.size()) + ", expected " +
236 std::to_string(sizeof(T)) + ".");
237 return reinterpret_cast<const T *>(data.data());
238 }
239
240 /// Cast from a type to its raw bytes.
241 template <typename T>
242 static MessageData from(T &t) {
243 return MessageData(reinterpret_cast<const uint8_t *>(&t), sizeof(T));
244 }
245
246 /// Convert the data to a hex string.
247 std::string toHex() const;
248
249 size_t numSegments() const override { return 1; }
250 Segment segment(size_t idx) const override {
251 if (idx != 0)
252 throw std::out_of_range("MessageData only has one segment");
253 return {data.data(), data.size()};
254 }
255 MessageData toMessageData() const override { return *this; }
256
257private:
258 std::vector<uint8_t> data;
259};

261/// Cursor for incremental consumption of a SegmentedMessageData.
262/// Tracks position across segment boundaries. Backends store this alongside
263/// a unique_ptr<SegmentedMessageData> for partial writes.
264///
265/// The cursor reads segments through SegmentedMessageData::segment() on every
266/// access and caches nothing, so it throws upon reaching a segment which has
267/// been take()n -- regardless of whether it was created before or after the
268/// take().
269///
270/// Deliberately a separate class (not embedded in SegmentedMessageData) so
271/// that generated types have no hidden members — their layout matches the
272/// hardware wire format exactly.
273class SegmentedMessageDataCursor {
274public:
275 SegmentedMessageDataCursor(const SegmentedMessageData &msg) : msg(msg) {}
276
277 /// Contiguous span from current position to end of current segment.
278 std::span<const uint8_t> remaining() const;
279
280 /// The unconsumed part of the current segment, with its `region`. `data` and
281 /// `size` match remaining().
282 Segment remainingSegment() const;
283
284 /// Advance by `n` bytes, crossing segment boundaries as needed.
285 void advance(size_t n);
286
287 /// True when all segments have been consumed.
288 bool done() const;
289
290private:
291 const SegmentedMessageData &msg;
292 size_t segIdx = 0;
293 size_t offset = 0;
294};
