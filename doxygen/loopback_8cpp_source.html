167struct SerialCoordData {
168 SerialCoordData(uint32_t x, uint32_t y) : _pad_head(0), y(y), x(x) {}
169 uint16_t _pad_head;
170 uint32_t y;
171 uint32_t x;
172};
173static_assert(sizeof(SerialCoordData) == sizeof(SerialCoordHeader),
174 "Size mismatch");
175#pragma pack(pop)

177// Note: this application is intended to test hardware. As such, we need
178// to be able to send batches. So this is not the typical way one would define a
179// message struct. It's closer to a streaming style.
180//
181// If constructed with a HostMemAllocator, the coordinates are stored in a
182// HostMem region (owned by the message) which backends can DMA directly from
183// and take() once sent.
184struct SerialCoordInput : SegmentedMessageData {
185private:
186 SerialCoordHeader header;
187 std::vector<SerialCoordData> coords;
188 SerialCoordHeader footer;
189 // Set iff the coordinates live in a HostMem region rather than `coords`.
190 bool coordsInRegion = false;
191 // Null after the region has been taken.
192 std::unique_ptr<services::HostMemRegion> coordsRegion;
193 size_t coordsBytes;
194
195public:
196 SerialCoordInput(uint32_t xTrans, uint32_t yTrans,
197 std::vector<SerialCoordData> &coords,
198 services::HostMemAllocator *allocator = nullptr)
199 : coordsBytes(coords.size() * sizeof(SerialCoordData)) {
200 header.coordsCount = (uint16_t)coords.size();
201 header.xTranslation = xTrans;
202 header.yTranslation = yTrans;
203 footer.coordsCount = 0;
204 if (allocator && coordsBytes > 0)
205 coordsRegion = allocator->allocate(coordsBytes, {});
206 if (coordsRegion) {
207 std::memcpy(coordsRegion->getPtr(), coords.data(), coordsBytes);
208 coordsRegion->flush();
209 coordsInRegion = true;
210 } else {
211 // No allocator, or the allocation failed: keep the data in the vector.
212 this->coords = coords;
213 }
214 }
215
216 size_t numSegments() const override { return 3; }
217 Segment segment(size_t idx) const override {
218 if (idx == 0)
219 return {reinterpret_cast<const uint8_t *>(&header), sizeof(header)};
220 else if (idx == 1) {
221 if (!coordsInRegion)
222 return {reinterpret_cast<const uint8_t *>(coords.data()), coordsBytes};
223 if (!coordsRegion)
224 throw std::runtime_error(
225 "SerialCoordInput: coordinates segment has been taken");
226 return {static_cast<const uint8_t *>(coordsRegion->getPtr()), coordsBytes,
227 coordsRegion.get()};
228 } else if (idx == 2)
229 return {reinterpret_cast<const uint8_t *>(&footer), sizeof(footer)};
230 else
231 throw std::out_of_range("SerialCoordInput: invalid segment index");
232 }
233
234 std::unique_ptr<services::HostMemRegion> take(size_t segIdx) override {
235 if (segIdx >= numSegments())
236 throw std::out_of_range("SerialCoordInput: invalid segment index");
237 if (segIdx != 1 || !coordsInRegion)
238 return nullptr;
239 if (!coordsRegion)
240 throw std::runtime_error(
241 "SerialCoordInput: coordinates segment has been taken");
242 return std::move(coordsRegion);
243 }
244};

262static int serialCoordTranslate(Accelerator *accel,
263 services::HostMemAllocator *allocator,
264 const char *probeName) {
265 size_t numCoords = 100;
266 uint32_t xTrans = 10, yTrans = 20;
267
268 // Generate random coordinates.
269 std::mt19937 rng(0xDEADBEEF);
270 std::uniform_int_distribution<uint32_t> dist(0, 1000000);
271 std::vector<Coord> inputCoords;
272 inputCoords.reserve(numCoords);
273 for (uint32_t i = 0; i < numCoords; ++i)
274 inputCoords.push_back({dist(rng), dist(rng)});
275
276 auto child = accel->getChildren().find(AppID("coord_translator_serial"));
277 if (child == accel->getChildren().end())
278 throw std::runtime_error("Serial coord translate test: no "
279 "'coord_translator_serial' child found");
280
281 auto &ports = child->second->getPorts();
282 auto portIter = ports.find(AppID("translate_coords_serial"));
283 if (portIter == ports.end())
284 throw std::runtime_error(
285 "Serial coord translate test: no 'translate_coords_serial' port found");
286
287 auto *func = portIter->second.getAs<services::FuncService::Function>();
288 if (!func)
289 throw std::runtime_error(
290 "Serial coord translate test: port is not a FuncService::Function");
291
292 // Drive the raw arg port through TypedWritePort so the segmented batch is
293 // packed correctly; drain the raw result port directly. We bypass
294 // TypedFunction here because `SerialCoordOutputFrame` is a hand-written
295 // union (not a real ESI type) and we don't want the typed result decoder
296 // to backpressure on the N+2 reply frames.
297 TypedWritePort<SerialCoordInput, /*SkipTypeCheck=*/true> argPort(
298 func->getRawWrite("arg"));
299 ChannelPort::ConnectOptions argOpts(/*bufferSize=*/std::nullopt,
300 /*translateMessage=*/false);
301 argPort.connect(argOpts);
302 ReadChannelPort &rawResult = func->getRawRead("result");
303 // The reply contains numCoords + 2 raw frames (one header + N data + one
304 // footer). Use an unbounded buffer (bufferSize=0) so the polling queue
305 // doesn't backpressure the backend mid-stream.
306 ChannelPort::ConnectOptions resultOpts(/*bufferSize=*/0,
307 /*translateMessage=*/false);
308 rawResult.connect(resultOpts);
309
310 std::vector<SerialCoordData> coords;
311 for (auto &c : inputCoords)
312 coords.emplace_back(c.x, c.y);
313 auto batch =
314 std::make_unique<SerialCoordInput>(xTrans, yTrans, coords, allocator);
315 if (allocator) {
316 // The coordinates segment should reference the HostMem region holding it.
317 Segment coordsSeg = batch->segment(1);
318 if (!coordsSeg.region || !coordsSeg.getDeviceAddress())
319 throw std::runtime_error(
320 "Serial coord translate test: coordinates not in a HostMem region");
321 if (batch->take(0) != nullptr)
322 throw std::runtime_error(
323 "Serial coord translate test: header segment has a region");
324 }
325 // Transfers ownership of the batch (and any HostMem region it owns) to the
326 // port; `batch` must not be used afterwards.
327 argPort.write(batch);
328
329 // The bulk-list reply is one header frame + numCoords data frames + one
330 // footer frame. TODO: list results aren't decoded here; we just drain the
331 // frames so the backend doesn't backpressure / hang.
332 MessageData drained;
333 for (size_t i = 0; i < numCoords + 2; ++i)
334 rawResult.read(drained);
335 std::cout << probeName << " ok\n";
336 return 0;
337}

339// Check HostMemAllocator failure and SerialCoordInput::take() semantics on a
340// message which is never sent.
341static void checkHostMemTake(services::HostMemAllocator &allocator) {
342 if (allocator.allocate(0, {}) != nullptr)
343 throw std::runtime_error("HostMem take test: allocate(0) should fail");
344
345 std::vector<SerialCoordData> coords = {{1, 2}, {3, 4}, {5, 6}};
346 SerialCoordInput msg(10, 20, coords, &allocator);
347 Segment coordsSeg = msg.segment(1);
348 if (!coordsSeg.region)
349 throw std::runtime_error(
350 "HostMem take test: coordinates not in a HostMem region");
351
352 // Take the region from the message (as a DMA engine would once the segment
353 // has been transmitted, to re-use it or return it to a pool).
354 std::unique_ptr<services::HostMemRegion> region = msg.take(1);
355 if (region.get() != coordsSeg.region)
356 throw std::runtime_error("HostMem take test: take() returned the wrong "
357 "region");
358 if (std::memcmp(region->getPtr(), coords.data(),
359 coords.size() * sizeof(SerialCoordData)) != 0)
360 throw std::runtime_error("HostMem take test: region contents corrupted");
361
362 // The taken segment is no longer accessible.
363 auto throws = [](auto fn) {
364 try {
365 fn();
366 } catch (const std::runtime_error &) {
367 return true;
368 }
369 return false;
370 };
371 if (!throws([&] { msg.segment(1); }) || !throws([&] { msg.take(1); }))
372 throw std::runtime_error(
373 "HostMem take test: taken segment is still accessible");
374 if (msg.segment(0).size != sizeof(SerialCoordHeader))
375 throw std::runtime_error("HostMem take test: header segment inaccessible");
376}
