36@dataclass(frozen=True)
37class CppPortKind:
38 """Per-port-kind constants for generating a module's typed port slots.
39
40 Holds everything that varies between port kinds in one record.
41 `member_template` (typed channel/func ports) and `member_ref` (service /
42 bundle ports) are mutually exclusive. `find_style` / `indexed_find` select
43 the `connect()` resolution snippet for scalar and indexed ports
44 respectively; `alias_kind` selects which `using` aliases to emit for the
45 typed template parameters.
46 """
47 service_type: Optional[
48 str] # findPortAs<...> service type (None = plain bundle)
49 find_style: str # 'ptr' | 'read_chan' | 'write_chan' | 'deref' | 'bundle'
50 indexed_find: str # 'as_ptr' | 'chan_read' | 'chan_write' | 'bundle_addr'
51 param_type: str # ctor parameter type
52 param_suffix: str # '_port' | '_chan' | '_svc'
53 member_template: Optional[
54 str] # typed member, e.g. "esi::TypedFunction<{p}Args, {p}Result>"
55 member_ref: Optional[str] # non-typed member / ctor ref type
56 indexed_elem: Optional[
57 str] # IndexedPorts<...> element (None => use member_template)
58 alias_kind: Optional[str] # None | 'func' (Args/Result) | 'chan' (Data)
59 connectable: bool # whether generated connect() calls .connect()
60
61 def member_type(self, alias_prefix: Optional[str] = None) -> str:
62 """Return the C++ member type string for this kind (no member name).
63
64 For typed ports (function/callback/to-host/from-host channels)
65 `alias_prefix` is required; the returned template parameters are written
66 using the alias names (`<prefix>Args`, `<prefix>Result`, `<prefix>Data`)
67 that should be emitted at module-class scope. For non-typed ports (MMIO
68 regions, telemetry metrics, plain bundles) `alias_prefix` is ignored and
69 the runtime reference/pointer type is returned directly.
70 """
71 if self.member_template is not None:
72 assert alias_prefix is not None, (
73 "alias_prefix is required for typed ports to avoid emitting long "
74 "mangled type names inline (which would also collide across "
75 "modules' `using` declarations)")
76 return self.member_template.format(p=alias_prefix)
77 # Every non-typed kind defines a reference member type.
78 assert self.member_ref is not None
79 return self.member_ref
80
81 def indexed_elem_type(self, alias_prefix: Optional[str] = None) -> str:
82 """Return the storage type used inside an `IndexedPorts<T>` for this kind.
83
84 Typed ports use the same `TypedFunction<...>` / `TypedReadPort<...>` etc.
85 that `member_type` produces. MMIO regions, telemetry metrics, and plain
86 bundle ports are stored as raw pointers because `std::map<int, T&>` is
87 ill-formed.
88 """
89 if self.indexed_elem is not None:
90 return self.indexed_elem
91 return self.member_type(alias_prefix=alias_prefix)

93 def scalar_find_code(self, v: str, appid_expr: str) -> str:
94 """Render the connect()-body snippet that resolves one scalar port into the
95 local `v`. Pure function of the kind so it can be golden-tested without a
96 live accelerator (see `test_port_find_code_golden`)."""
97 if self.find_stylefind_style == "ptr":
98 return (f"auto *{v} =\n"
99 f" esi::findPortAsOrThrow<{self.service_type}>(\n"
100 f" rawModule, {appid_expr});")
101 if self.find_stylefind_style == "read_chan":
102 return (f"auto &{v} =\n"
103 f" esi::findPortAsOrThrow<{self.service_type}>(\n"
104 f' rawModule, {appid_expr})->getRawRead("data");')
105 if self.find_stylefind_style == "write_chan":
106 return (f"auto &{v} =\n"
107 f" esi::findPortAsOrThrow<{self.service_type}>(\n"
108 f' rawModule, {appid_expr})->getRawWrite("data");')
109 if self.find_stylefind_style == "deref":
110 return (f"auto &{v} =\n"
111 f" *esi::findPortAsOrThrow<{self.service_type}>(\n"
112 f" rawModule, {appid_expr});")
113 if self.find_stylefind_style == "bundle":
114 # plain BundlePort
115 return f"auto &{v} = esi::findPortOrThrow(rawModule, {appid_expr});"
116 raise ValueError(f"unknown find_style: {self.find_style!r}")

118 def indexed_find_code(self, map_var: str, appid_expr: str) -> str:

139 f' svc->getRawWrite("data"));')

141 return (f" {map_var}.try_emplace(\n"
142 f" static_cast<int>(idx),\n"
143 f" &esi::findPortOrThrow(rawModule, {appid_expr}));")
144 raise ValueError(f"unknown indexed_find: {self.indexed_find!r}")
