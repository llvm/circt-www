39def connect(destination, source):
40 """A convenient way to use BackedgeBuilder."""
41 if not isinstance(destination, OpOperand):
42 raise TypeError(
43 f"cannot connect to destination of type {type(destination)}. "
44 "Must be OpOperand.")
45 value = get_value(source)
46 if value is None:
47 raise TypeError(f"cannot connect from source of type {type(source)}")
48
49 index = destination.index
50 destination.operation.operands[index] = value
51 if destination.backedge_owner and \
52 index in destination.backedge_owner.backedges:
53 destination.backedge_owner.backedges[index].erase()
54 del destination.backedge_owner.backedges[index]

57def var_to_attribute(obj, none_on_fail: bool = False) -> ir.Attribute:
58 """Create an MLIR attribute from a Python object for a few common cases."""
59 if isinstance(obj, ir.Attribute):
60 return obj
61 if isinstance(obj, bool):
62 return ir.BoolAttr.get(obj)
63 if isinstance(obj, int):
64 attrTy = ir.IntegerType.get_signless(64)
65 return ir.IntegerAttr.get(attrTy, obj)
66 if isinstance(obj, str):
67 return ir.StringAttr.get(obj)
68 if isinstance(obj, list):
69 arr = [var_to_attribute(x, none_on_fail) for x in obj]
70 if all(arr):
71 return ir.ArrayAttr.get(arr)
72 return None
73 if none_on_fail:
74 return None
75 raise TypeError(f"Cannot convert type '{type(obj)}' to MLIR attribute")

157# conversation regarding how to achieve this is ongoing and I expect it to be a

159def attribute_to_var(attr):
160
161 if attr is None:
162 return None
163 if not isinstance(attr, ir.Attribute):
164 raise TypeError("attribute_to_var only accepts MLIR Attributes")
165
166 # If it's not the root type, assume it's already been downcasted and don't do
167 # the expensive probing below.
168 if attr.__class__ != ir.Attribute and hasattr(attr, "value"):
169 return attr.value
170
171 from .dialects import hw, om
172 if isinstance(attr, ir.BoolAttr):
173 return ir.BoolAttr(attr).value
174 if isinstance(attr, ir.IntegerAttr):
175 return ir.IntegerAttr(attr).value
176 if hw.InnerSymAttr.isinstance(attr):
177 return ir.StringAttr(hw.InnerSymAttr(attr).symName).value
178 if isinstance(attr, ir.StringAttr):
179 return ir.StringAttr(attr).value
180 if isinstance(attr, ir.FlatSymbolRefAttr):
181 return ir.FlatSymbolRefAttr(attr).value
182 if isinstance(attr, ir.TypeAttr):
183 return ir.TypeAttr(attr).value
184 if isinstance(attr, ir.ArrayAttr):
185 arr = ir.ArrayAttr(attr)
186 return [attribute_to_var(x) for x in arr]
187 if isinstance(attr, ir.DictAttr):
188 dict = ir.DictAttr(attr)
189 return {i.name: attribute_to_var(i.attr) for i in dict}
190 if om.ReferenceAttr.isinstance(attr):
191 return attribute_to_var(om.ReferenceAttr(attr).inner_ref)
192 if hw.InnerRefAttr.isinstance(attr):
193 ref = hw.InnerRefAttr(attr)
194 return (ir.StringAttr(ref.module).value, ir.StringAttr(ref.name).value)
195 if om.ListAttr.isinstance(attr):
196 return list(map(attribute_to_var, om.ListAttr(attr)))
197 if om.OMIntegerAttr.isinstance(attr):
198 return int(str(om.OMIntegerAttr(attr)))
199 if om.PathAttr.isinstance(attr):
200 return om.PathAttr(attr).value
201
202 raise TypeError(f"Cannot convert {repr(attr)} to python value")

279 errors = []

322class NamedValueOpView:
323 """Helper class to incrementally construct an instance of an operation that
324 names its operands and results"""
325
326 def __init__(self,
327 cls,
328 data_type=None,
329 input_port_mapping=None,
330 pre_args=None,
331 post_args=None,
332 needs_result_type=False,
333 **kwargs):
334 # Set defaults
335 if input_port_mapping is None:
336 input_port_mapping = {}
337 if pre_args is None:
338 pre_args = []
339 if post_args is None:
340 post_args = []
341
342 # Set result_indices to name each result.
343 result_names = self.result_names()
344 result_indices = {}
345 for i in range(len(result_names)):
346 result_indices[result_names[i]] = i
347
348 # Set operand_indices to name each operand. Give them an initial value,
349 # either from input_port_mapping or a default value.
350 backedges = {}
351 operand_indices = {}
352 operand_values = []
353 operand_names = self.operand_names()
354 for i in range(len(operand_names)):
355 arg_name = operand_names[i]
356 operand_indices[arg_name] = i
357 if arg_name in input_port_mapping:
358 value = get_value(input_port_mapping[arg_name])
359 operand = value
360 else:
361 backedge = self.create_default_value(i, data_type, arg_name)
362 backedges[i] = backedge
363 operand = backedge.result
364 operand_values.append(operand)
365
366 # Some ops take a list of operand values rather than splatting them out.
367 if isinstance(data_type, list):
368 operand_values = [operand_values]
369
370 # In many cases, result types are inferred, and we do not need to pass
371 # data_type to the underlying constructor. It must be provided to
372 # NamedValueOpView in cases where we need to build backedges, but should
373 # generally not be passed to the underlying constructor in this case. There
374 # are some oddball ops that must pass it, even when building backedges, and
375 # these set needs_result_type=True.
376 if data_type is not None and (needs_result_type or len(backedges) == 0):
377 pre_args.insert(0, data_type)
378
379 self.opview = cls(*pre_args, *operand_values, *post_args, **kwargs)
380 self.operand_indices = operand_indices
381 self.result_indices = result_indices
382 self.backedges = backedges
383
384 def __getattr__(self, name):
385 # Check for the attribute in the arg name set.
386 if "operand_indices" in dir(self) and name in self.operand_indices:
387 index = self.operand_indices[name]
388 value = self.opview.operands[index]
389 return OpOperand(self.opview.operation, index, value, self)
390
391 # Check for the attribute in the result name set.
392 if "result_indices" in dir(self) and name in self.result_indices:
393 index = self.result_indices[name]
394 value = self.opview.results[index]
395 return OpOperand(self.opview.operation, index, value, self)
396
397 # Forward "attributes" attribute from the operation.
398 if name == "attributes":
399 return self.opview.operation.attributes
400
401 # If we fell through to here, the name isn't a result.
402 raise AttributeError(f"unknown port name {name}")
403
404 def create_default_value(self, index, data_type, arg_name):
405 return BackedgeBuilder.create(data_type, arg_name, self)

407 @property
408 def operation(self):
409 """Get the operation associated with this builder."""
410 return self.opview.operation
