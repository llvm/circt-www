46def test_channel_mmio_manifest_tracks_client_space():
47 """The manifest window must not be a fixed reservation: it follows whatever
48 the clients actually use so the BAR stays as small as possible."""
49
50 small, = _Bundles(_Bundle("read", {"size": 8})).to_client_reqs
51 _, small_loc = ChannelMMIO.build_table(_Bundles(small))
52 assert small_loc == 0x200

59def test_channel_mmio_places_manifest_after_clients():
60 table, manifest_loc = ChannelMMIO.build_table(
61 _Bundles(_Bundle("read"), _Bundle("read", {"size": 0x1203})))
62 last_base = list(table)[-1]
63 last_size = table[last_base][0]
64
65 assert manifest_loc >= last_base + last_size
66 # A power-of-two base means 'address has a bit set above the clients' is the
67 # whole decode: no comparator, and the region can't overflow.
68 assert manifest_loc & (manifest_loc - 1) == 0
