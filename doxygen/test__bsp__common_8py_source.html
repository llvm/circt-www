41def test_channel_mmio_manifest_tracks_client_space():
42 """The manifest window must not be a fixed reservation: it follows whatever
43 the clients actually use so the BAR stays as small as possible."""
44
45 small, = _Bundles(_Bundle("read", {"size": 8})).to_client_reqs
46 _, small_loc = ChannelMMIO.build_table(_Bundles(small))
47 assert small_loc == 0x200
48
49 big = _Bundle("read", {"size": 1 << 20})
50 _, big_loc = ChannelMMIO.build_table(_Bundles(big))
51 assert big_loc == 1 << 21

54def test_channel_mmio_places_manifest_after_clients():
55 table, manifest_loc = ChannelMMIO.build_table(
56 _Bundles(_Bundle("read"), _Bundle("read", {"size": 0x1203})))
57 last_base = list(table)[-1]
58 last_size = table[last_base][0]
59
60 assert manifest_loc >= last_base + last_size
61 # A power-of-two base means 'address has a bit set above the clients' is the
62 # whole decode: no comparator, and the region can't overflow.
63 assert manifest_loc & (manifest_loc - 1) == 0
