22def _run(conn: AcceleratorConnection, bus_width: int):
23 # Exercise discovery without a caller-populated manifest type table.
24 hostmem = conn.get_service_hostmem()
25 hostmem.start()
26 acc = conn.build_accelerator()
27 allocations = []
28 writers = []
29 for width in [13, bus_width, bus_width + 24, bus_width * 2 + 24, 16384]:
30 dut = acc.children[esiaccel.AppID("wr", width)]
31 req, ack, gate = [
32 dut.ports[esiaccel.AppID(name)] for name in ("req", "ack", "gate")
33 ]
34 for port in (req, ack, gate):
35 port.connect()
36 size = (width + 7) // 8
37 expected = []
38 for i in range(3):
39 mem = hostmem.allocate(size + 16)
40 allocations.append(mem)
41 ctypes.memset(mem.ptr, 0xA5, size + 16)
42 data = bytes((j * 37 + 11 + i) & 0xFF for j in range(size))
43 value = int.from_bytes(data, "little") & ((1 << width) - 1)
44 data = value.to_bytes(size, "little")
45 req.write({"address": mem.ptr + 3, "tag": 0, "data": data})
46 expected.append((mem, data))
47 writers.append((ack, gate, expected))
48
49 # Release completions after issuing all clients' writes.
50 for ack, gate, expected in writers:
51 gate.write(1)
52 for mem, data in expected:
53 ack.read().result(timeout=30)
54 assert ctypes.string_at(mem.ptr,
55 len(data) + 16) == (b"\xA5" * 3 + data +
56 b"\xA5" * 13)
57
58 req = acc.ports[esiaccel.AppID("list_req")]
59 ack = acc.ports[esiaccel.AppID("list_ack")]
60 req.connect()
61 ack.connect()
62 for count in [1, 7, 100]:
63 mem = hostmem.allocate(count * 3 + 16)
64 allocations.append(mem)
65 ctypes.memset(mem.ptr, 0xA5, count * 3 + 16)
66 data = bytes((i * 29 + 7) & 0xFF for i in range(count * 3))
67 for i in range(count):
68 req.write({
69 "address": mem.ptr + 3,
70 "data": data[i * 3:i * 3 + 3],
71 "last": int(i == count - 1),
72 })
73 for _ in range(count):
74 ack.read().result(timeout=30)
75 assert ctypes.string_at(mem.ptr, count * 3 + 16) == (b"\xA5" * 3 + data +
76 b"\xA5" * 13)
77
78 req = acc.ports[esiaccel.AppID("probe_req")]
79 resp = acc.ports[esiaccel.AppID("probe_resp")]
80 req.connect()
81 resp.connect()
82 data = bytes((i * 13 + 3) & 0xFF for i in range(2048))
83 req.write({"address": 19, "tag": 17, "data": data})
84 beat_bytes = bus_width // 8
85 cap = max(256, beat_bytes)
86 for offset in range(0, len(data), beat_bytes):

88 assert _int(beat["address"]) == 19 + offset
89 assert _int(beat["tag"]) == 17

91 assert _int(beat["data"]) == int.from_bytes(
